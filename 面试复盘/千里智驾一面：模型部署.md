class Cache {
public:
    explicit Cache(size_t cap) : cap_(cap) {}
    void put(const std::string &k, std::shared_ptr<Data> v) {
        std::lock_guard<std::mutex> lk(mu_);
        map_[k] = v;
        order_.push_back(k);
        if (order_.size() > cap_) {
            map_.erase(order_.front());
            order_.pop_front();
        }
    }
    const std::shared_ptr<Data> &get(const std::string &k) {
        std::lock_guard<std::mutex> lk(mu_);
        auto it = map_.find(k);
        if (it == map_.end()) return nullptr;
        return it->second;
    }

private:
    size_t cap_;
    std::mutex mu_;
    std::unordered_map<std::string, std::shared_ptr<Data>> map_;
    std::deque<std::string> order_;
};
