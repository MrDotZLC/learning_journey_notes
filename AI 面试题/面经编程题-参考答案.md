### 1. 快速排序

#### 1.1 核心思路

选取一个 **pivot**，将数组分为「小于 pivot」和「大于 pivot」两部分，递归处理两侧。 平均时间复杂度 $O(n \log n)$，最坏（已排序 + 固定 pivot）退化为 $O(n^2)$。 三路划分（Dutch National Flag）处理大量重复元素时可避免退化。

#### 1.2 标准双指针 Lomuto / Hoare 分区

```cpp
#include <vector>
#include <algorithm>  // std::swap

// ── Hoare 分区方案（左右双指针，交换次数更少）────────────────────────────
// 返回最终 pivot 落定后左半段的最右下标（不保证 pivot 精确在该位置）
// [lo, hi] 均为闭区间
int partition(std::vector<int>& arr, int lo, int hi) {
    // 三数取中：避免已排序序列退化为 O(n²)
    int mid = lo + (hi - lo) / 2;
    if (arr[lo] > arr[mid]) std::swap(arr[lo], arr[mid]);
    if (arr[lo] > arr[hi]) std::swap(arr[lo], arr[hi]);
    if (arr[mid] > arr[hi]) std::swap(arr[mid], arr[hi]);
    // 此时 arr[mid] 为三者中值，将其放到 hi-1 作为 pivot
    std::swap(arr[mid], arr[hi - 1]);
    int pivot = arr[hi - 1];

    int i = lo, j = hi - 1;
    while (true) {
        while (arr[++i] < pivot) {}   // 找到左侧 ≥ pivot 的元素
        while (arr[--j] > pivot) {}   // 找到右侧 ≤ pivot 的元素
        if (i >= j) break;
        std::swap(arr[i], arr[j]);    // 交换使其各归其位
    }
    std::swap(arr[i], arr[hi - 1]);   // pivot 归位
    return i;                          // pivot 最终下标
}

void quickSort(std::vector<int>& arr, int lo, int hi) {
    if (hi - lo < 1) return;          // 区间长度 ≤ 1，无需排序
    int p = partition(arr, lo, hi);
    quickSort(arr, lo, p - 1);        // 递归左半段
    quickSort(arr, p + 1, hi);        // 递归右半段
}

// 对外接口
void quickSort(std::vector<int>& arr) {
    if (arr.size() > 1)
        quickSort(arr, 0, static_cast<int>(arr.size()) - 1);
}
```

#### 1.3 三路划分（处理大量重复元素）

```cpp
// 将 [lo, hi] 划分为三段：< pivot | == pivot | > pivot
// 递归只处理两端，中间段（等于 pivot）直接跳过
void quickSort3Way(std::vector<int>& arr, int lo, int hi) {
    if (lo >= hi) return;

    int pivot = arr[lo + (hi - lo) / 2];
    int lt = lo;      // arr[lo..lt-1]  < pivot
    int gt = hi;      // arr[gt+1..hi]  > pivot
    int i  = lo;      // arr[lt..i-1]  == pivot（已处理）

    while (i <= gt) {
        if      (arr[i] < pivot) std::swap(arr[lt++], arr[i++]);
        else if (arr[i] > pivot) std::swap(arr[i],    arr[gt--]);
        else                     ++i;                  // 等于 pivot，直接跳过
    }
    quickSort3Way(arr, lo, lt - 1);   // 递归 < pivot 段
    quickSort3Way(arr, gt + 1, hi);   // 递归 > pivot 段
}
```

### 2. 堆排序

#### 2.1 核心思路

利用 **最大堆** 性质：堆顶永远是当前区间最大值。

1. **建堆**（Build Heap）：从最后一个非叶节点向上逐一 `siftDown`，$O(n)$。
2. **排序**：反复将堆顶与末尾交换，缩小堆范围并 `siftDown`，$O(n \log n)$。

节点 $i$（0-indexed）的子节点为 $2i+1$（左）和 $2i+2$（右）。

```cpp
#include <vector>

// ── siftDown：将下标 i 的元素下沉到正确位置 ─────────────────────────────
// heap_size：当前堆的有效大小（堆尾已交换到数组末尾，不参与堆操作）
void siftDown(std::vector<int>& arr, int i, int heap_size) {
    while (true) {
        int largest = i;
        int left  = 2 * i + 1;
        int right = 2 * i + 2;

        // 找出 i、左孩子、右孩子中的最大值
        if (left  < heap_size && arr[left]  > arr[largest]) largest = left;
        if (right < heap_size && arr[right] > arr[largest]) largest = right;

        if (largest == i) break;           // 已满足堆性质，退出

        std::swap(arr[i], arr[largest]);   // 将当前节点下沉
        i = largest;                        // 继续向下检查
    }
}

void heapSort(std::vector<int>& arr) {
    int n = static_cast<int>(arr.size());

    // 阶段 1：建最大堆
    // 最后一个非叶节点下标为 n/2 - 1，从此向上逐一 siftDown
    for (int i = n / 2 - 1; i >= 0; --i)
        siftDown(arr, i, n);

    // 阶段 2：逐步将堆顶（最大值）交换到数组末尾
    for (int end = n - 1; end > 0; --end) {
        std::swap(arr[0], arr[end]);   // 堆顶放到已排好的区域
        siftDown(arr, 0, end);         // 对缩小后的堆重新维护堆性质
    }
}
```

### 3. 归并排序

#### 3.1 核心思路

**分治**：递归将数组二分，再将两个有序子数组合并（Merge）。 时间复杂度严格 $O(n \log n)$，空间 $O(n)$（需要辅助数组）。 是 **稳定排序**，适合链表排序和外排序。

```cpp
#include <vector>

// ── merge：将 arr[lo..mid] 和 arr[mid+1..hi] 两段有序子数组合并 ─────────
void merge(std::vector<int>& arr, int lo, int mid, int hi,
           std::vector<int>& tmp) {
    // 将当前区间拷贝到临时数组，避免原地覆盖
    for (int k = lo; k <= hi; ++k) tmp[k] = arr[k];

    int i = lo;       // 左半段游标
    int j = mid + 1;  // 右半段游标

    for (int k = lo; k <= hi; ++k) {
        if      (i > mid)              arr[k] = tmp[j++];  // 左段用尽
        else if (j > hi)               arr[k] = tmp[i++];  // 右段用尽
        else if (tmp[i] <= tmp[j])     arr[k] = tmp[i++];  // 取较小值（≤ 保证稳定）
        else                           arr[k] = tmp[j++];
    }
}

void mergeSort(std::vector<int>& arr, int lo, int hi,
               std::vector<int>& tmp) {
    if (lo >= hi) return;              // 只剩一个元素，已有序

    int mid = lo + (hi - lo) / 2;
    mergeSort(arr, lo, mid, tmp);      // 排序左半段
    mergeSort(arr, mid + 1, hi, tmp);  // 排序右半段
    merge(arr, lo, mid, hi, tmp);      // 合并两段
}

void mergeSort(std::vector<int>& arr) {
    std::vector<int> tmp(arr.size());
    mergeSort(arr, 0, static_cast<int>(arr.size()) - 1, tmp);
}
```

### 4. 拓扑排序

#### 4.1 核心思路

针对 **有向无环图（DAG）**，输出一个线性顺序使得每条边 $u \to v$ 中 $u$ 出现在 $v$ 之前。 两种标准实现：

|方法|原理|检测环|
|---|---|---|
|Kahn（BFS）|维护入度表，入度为 0 则入队|最终输出节点数 < 总节点数|
|DFS 后序|DFS 完成时压栈，逆序即为拓扑序|检测到灰色节点（back edge）|

#### 4.2 Kahn 算法（BFS，推荐面试写法）

```cpp
#include <vector>
#include <queue>
#include <stdexcept>

// 输入：n 个节点（0-indexed），edges[i] = {from, to}
// 返回：拓扑序列；若图含环则抛出异常
std::vector<int> topoSort(int n, const std::vector<std::pair<int,int>>& edges) {
    std::vector<std::vector<int>> adj(n);    // 邻接表
    std::vector<int> indegree(n, 0);          // 每个节点的入度

    for (auto& [u, v] : edges) {
        adj[u].push_back(v);
        ++indegree[v];                         // v 的入度 +1
    }

    // 将所有入度为 0 的节点加入队列（可以作为起点）
    std::queue<int> q;
    for (int i = 0; i < n; ++i)
        if (indegree[i] == 0) q.push(i);

    std::vector<int> order;
    order.reserve(n);

    while (!q.empty()) {
        int u = q.front(); q.pop();
        order.push_back(u);

        for (int v : adj[u]) {
            --indegree[v];                     // 移除 u 后，v 的入度减 1
            if (indegree[v] == 0) q.push(v);  // 入度归零，可以处理
        }
    }

    if (static_cast<int>(order.size()) != n)
        throw std::runtime_error("Graph has a cycle");

    return order;
}
```

#### 4.3 DFS 后序反转

```cpp
#include <vector>
#include <stack>
#include <stdexcept>

// color: 0=白(未访问), 1=灰(递归栈中), 2=黑(已完成)
void dfs(int u, std::vector<std::vector<int>>& adj,
         std::vector<int>& color, std::stack<int>& stk) {
    color[u] = 1;                               // 标记为"处理中"
    for (int v : adj[u]) {
        if (color[v] == 1)
            throw std::runtime_error("Cycle detected");  // 发现回边 → 有环
        if (color[v] == 0)
            dfs(v, adj, color, stk);
    }
    color[u] = 2;                               // 所有后继已处理完
    stk.push(u);                                // 后序压栈
}

std::vector<int> topoSortDFS(int n, const std::vector<std::pair<int,int>>& edges) {
    std::vector<std::vector<int>> adj(n);
    for (auto& [u, v] : edges) adj[u].push_back(v);

    std::vector<int> color(n, 0);
    std::stack<int> stk;

    for (int i = 0; i < n; ++i)
        if (color[i] == 0)
            dfs(i, adj, color, stk);

    std::vector<int> order;
    while (!stk.empty()) {
        order.push_back(stk.top());
        stk.pop();
    }
    return order;   // 弹栈顺序即为拓扑序
}
```

### 5. LRU Cache

#### 5.1 核心思路

**Least Recently Used**：容量满时淘汰最久未访问的条目。

经典实现：**双向链表 + 哈希表**。

- 链表维护访问顺序（头部最新，尾部最旧）。
- 哈希表 `key → 链表节点指针` 实现 $O(1)$ 查找。
- `get` / `put` 均摊 $O(1)$。

```cpp
#include <unordered_map>
#include <list>
#include <stdexcept>

class LRUCache {
public:
    explicit LRUCache(int capacity) : cap_(capacity) {}

    // 查询 key；不存在返回 -1，存在则将节点移到链表头（最新位置）
    int get(int key) {
        auto it = map_.find(key);
        if (it == map_.end()) return -1;

        // 将命中节点移到链表头部（表示最近访问）
        moveToFront(it->second);
        return it->second->second;   // 返回 value
    }

    // 插入或更新 key-value
    void put(int key, int value) {
        auto it = map_.find(key);
        if (it != map_.end()) {
            // key 已存在：更新值并移到头部
            it->second->second = value;
            moveToFront(it->second);
            return;
        }

        // 容量已满：移除链表尾部节点（最久未访问）
        if (static_cast<int>(map_.size()) >= cap_) {
            auto lru = std::prev(list_.end());  // 尾部节点
            map_.erase(lru->first);              // 从哈希表中删除
            list_.erase(lru);                    // 从链表中删除
        }

        // 插入新节点到链表头部
        list_.emplace_front(key, value);
        map_[key] = list_.begin();
    }

private:
    using KV   = std::pair<int, int>;
    using Iter = std::list<KV>::iterator;

    int                           cap_;
    std::list<KV>                 list_;   // front=最新, back=最旧
    std::unordered_map<int, Iter> map_;    // key → 链表节点迭代器

    // 将节点 it 移到链表头部（splice 不会使其他迭代器失效）
    void moveToFront(Iter it) {
        list_.splice(list_.begin(), list_, it);
    }
};
```

### 6. 快速幂

#### 6.1 核心思路

反复平方法（Exponentiation by Squaring）：

$$ x^n = \begin{cases} 1 & n = 0 \ (x^{n/2})^2 & n \text{ 为偶数} \ x \cdot (x^{(n-1)/2})^2 & n \text{ 为奇数} \end{cases} $$

时间复杂度 $O(\log n)$，空间 $O(1)$（迭代版）。

```cpp
#include <cstdint>

// ── 整数快速幂（模运算版，防溢出）─────────────────────────────────────────
// 计算 base^exp mod modulus
long long fastPow(long long base, long long exp, long long modulus) {
    long long result = 1;
    base %= modulus;              // 先对底数取模，避免初始溢出

    while (exp > 0) {
        if (exp & 1)              // 当前位为 1：将当前 base 乘入结果
            result = result * base % modulus;
        base = base * base % modulus;  // base 自平方，对应指数右移一位
        exp >>= 1;                // 指数右移
    }
    return result;
}

// ── 浮点快速幂（用于 pow(x, n) 类题目，需处理负指数）───────────────────
double myPow(double x, int n) {
    // 用 long long 防止 n = INT_MIN 取反时溢出
    long long N = n;
    if (N < 0) { x = 1.0 / x; N = -N; }

    double result = 1.0;
    while (N > 0) {
        if (N & 1) result *= x;
        x *= x;
        N >>= 1;
    }
    return result;
}
```

### 7. 求平方根

#### 7.1 核心思路

两种方法并存：

|方法|特点|
|---|---|
|二分查找|整数平方根，精确，$O(\log n)$|
|牛顿迭代|浮点平方根，二次收敛，$O(\log(1/\varepsilon))$|

**牛顿迭代**：求 $f(x) = x^2 - S = 0$ 的根，迭代公式：

$$ x_{k+1} = \frac{1}{2}\left(x_k + \frac{S}{x_k}\right) $$

每次迭代有效位数翻倍（二次收敛）。

```cpp
#include <cmath>    // std::abs（浮点版）
#include <cstdint>

// ── 方法 1：二分查找（整数平方根，向下取整）──────────────────────────────
// 返回 ⌊√n⌋
int sqrtBinarySearch(int n) {
    if (n < 2) return n;
    int lo = 1, hi = n / 2;    // 结果一定 ≤ n/2（n ≥ 4 时）
    while (lo <= hi) {
        long long mid = lo + (hi - lo) / 2;
        long long sq  = mid * mid;
        if      (sq == n) return static_cast<int>(mid);
        else if (sq < n)  lo = static_cast<int>(mid) + 1;
        else              hi = static_cast<int>(mid) - 1;
    }
    return hi;   // hi 是最后一个满足 hi² ≤ n 的值
}

// ── 方法 2：牛顿迭代（浮点平方根）────────────────────────────────────────
// 精度由 eps 控制
double sqrtNewton(double S, double eps = 1e-9) {
    if (S < 0) return -1.0;    // 非法输入
    if (S == 0) return 0.0;

    double x = S;              // 初始猜测值（也可用 S/2 或更好的估计）
    while (true) {
        double next = 0.5 * (x + S / x);   // 牛顿迭代公式
        if (std::abs(next - x) < eps)       // 收敛判断
            break;
        x = next;
    }
    return x;
}
```

### 8. 大数乘法

#### 8.1 核心思路

用字符串模拟竖式乘法。两个长度分别为 $m$、$n$ 的数相乘，结果最多 $m+n$ 位。

关键观察：$\text{num1}[i] \times \text{num2}[j]$ 的结果，**个位**落在结果数组的 $[i+j+1]$，**十位**落在 $[i+j]$。

```cpp
#include <string>
#include <algorithm>

// 输入：两个非负整数字符串（不含前导零，除了 "0" 本身）
// 输出：乘积字符串
std::string multiply(const std::string& num1, const std::string& num2) {
    int m = static_cast<int>(num1.size());
    int n = static_cast<int>(num2.size());

    // 结果最多 m+n 位，初始化为 0
    std::vector<int> pos(m + n, 0);

    // 从低位到高位逐位相乘（从字符串末尾开始）
    for (int i = m - 1; i >= 0; --i) {
        for (int j = n - 1; j >= 0; --j) {
            int mul  = (num1[i] - '0') * (num2[j] - '0');
            int p1   = i + j;       // 十位落点
            int p2   = i + j + 1;   // 个位落点
            int sum  = mul + pos[p2];

            pos[p2] = sum % 10;     // 个位存入结果
            pos[p1] += sum / 10;    // 进位加到高位
        }
    }

    // 将 pos 数组转换为字符串，跳过前导零
    std::string result;
    for (int digit : pos) {
        if (result.empty() && digit == 0) continue;  // 跳过前导零
        result += static_cast<char>('0' + digit);
    }

    return result.empty() ? "0" : result;
}
```

### 9. Hamming Weight（位 1 的个数）

#### 9.1 核心思路

统计一个 32 位无符号整数的二进制表示中 `1` 的个数（popcount）。

三种常见实现：

```cpp
#include <cstdint>

// ── 方法 1：逐位检查，O(32) ──────────────────────────────────────────────
int hammingWeightNaive(uint32_t n) {
    int count = 0;
    while (n) {
        count += n & 1;   // 检查最低位
        n >>= 1;
    }
    return count;
}

// ── 方法 2：Brian Kernighan 技巧，O(popcount) ────────────────────────────
// n & (n-1) 每次消去最低位的 1，循环次数等于 1 的个数
int hammingWeightKernighan(uint32_t n) {
    int count = 0;
    while (n) {
        n &= (n - 1);   // 清除最低位的 1（例：1010 → 1000）
        ++count;
    }
    return count;
}

// ── 方法 3：SWAR（并行位计数），O(1)，面试展示用 ─────────────────────────
// 原理：将 32 位分为多组，组内并行累加，最后汇总
int hammingWeightSWAR(uint32_t n) {
    n = n - ((n >> 1) & 0x55555555u);          // 每 2 位：计算该 2 位内的 1 数量
    n = (n & 0x33333333u) + ((n >> 2) & 0x33333333u);  // 每 4 位合并
    n = (n + (n >> 4)) & 0x0f0f0f0fu;          // 每 8 位合并
    return static_cast<int>((n * 0x01010101u) >> 24);   // 横向累加所有字节
}

// ── 方法 4：编译器内建（实际生产代码首选）────────────────────────────────
int hammingWeightBuiltin(uint32_t n) {
    return __builtin_popcount(n);   // GCC/Clang 直接映射到 POPCNT 指令
}
```

### 10. K-coverage Intervals

#### 10.1 问题定义

给定 $n$ 个区间 $[l_i, r_i]$，找出所有被 **至少 $k$ 个区间覆盖** 的点（或连续区间段）。

#### 10.2 差分数组解法，$O(n \log n)$

```cpp
#include <vector>
#include <algorithm>
#include <utility>

// 返回所有被至少 k 个区间覆盖的连续段，每段用 [start, end] 表示（闭区间）
// 输入区间为整数闭区间 [l, r]
std::vector<std::pair<int,int>>
kCoverageIntervals(const std::vector<std::pair<int,int>>& intervals, int k) {
    // 差分数组：在 l 处 +1，在 r+1 处 -1
    // 扫描线扫过时，前缀和即为当前点被覆盖的区间数
    std::vector<std::pair<int,int>> events;
    events.reserve(2 * intervals.size());

    for (auto& [l, r] : intervals) {
        events.emplace_back(l,     +1);   // 区间开始
        events.emplace_back(r + 1, -1);   // 区间结束（开区间端点）
    }

    // 按坐标排序；同坐标时先处理 +1（保证端点处计数正确）
    std::sort(events.begin(), events.end(), [](const auto& a, const auto& b) {
        return a.first != b.first ? a.first < b.first : a.second > b.second;
    });

    std::vector<std::pair<int,int>> result;
    int coverage = 0;
    int segStart = -1;   // 当前 ≥k 覆盖段的起点，-1 表示未开始

    for (auto& [pos, delta] : events) {
        int prevCoverage = coverage;
        coverage += delta;

        // 从 < k 变为 ≥ k：段开始
        if (prevCoverage < k && coverage >= k) {
            segStart = pos;
        }
        // 从 ≥ k 变为 < k：段结束
        if (prevCoverage >= k && coverage < k) {
            result.emplace_back(segStart, pos - 1);
        }
    }

    return result;
}
```

### 11. 按指定概率采样

#### 11.1 问题定义

给定 $n$ 个选项，概率分别为 $p_0, p_1, \ldots, p_{n-1}$（$\sum p_i = 1$），每次采样返回下标 $i$，使得 $P(\text{返回} i) = p_i$。

#### 11.2 前缀和 + 二分查找，$O(n)$ 构建，$O(\log n)$ 查询

```cpp
#include <vector>
#include <random>
#include <algorithm>
#include <stdexcept>

class WeightedSampler {
public:
    // weights[i] 为第 i 项的权重（不要求归一化）
    explicit WeightedSampler(const std::vector<double>& weights)
        : gen_(std::random_device{}())
    {
        if (weights.empty())
            throw std::invalid_argument("Empty weights");

        // 构建前缀和数组（累积分布函数 CDF）
        prefix_.resize(weights.size());
        prefix_[0] = weights[0];
        for (size_t i = 1; i < weights.size(); ++i) {
            prefix_[i] = prefix_[i-1] + weights[i];
            if (weights[i] < 0)
                throw std::invalid_argument("Negative weight");
        }
        // prefix_.back() 为总权重，uniform 分布上界
        dist_ = std::uniform_real_distribution<double>(0.0, prefix_.back());
    }

    // 返回采样下标
    int sample() {
        double r = dist_(gen_);    // 在 [0, total_weight) 均匀采样
        // lower_bound 找到第一个 prefix_[i] >= r 的位置，即所属区间
        auto it = std::lower_bound(prefix_.begin(), prefix_.end(), r);
        return static_cast<int>(it - prefix_.begin());
    }

private:
    std::vector<double>                        prefix_;  // 前缀权重和
    std::mt19937                               gen_;     // Mersenne Twister PRNG
    std::uniform_real_distribution<double>    dist_;
};
```

#### 11.3 别名方法（Alias Method），$O(n)$ 构建，$O(1)$ 查询

```cpp
// Alias Method：适合高频采样场景
// 将 n 个概率"重分配"为 n 个等面积柱，每柱最多两个原始分类
class AliasSampler {
public:
    explicit AliasSampler(std::vector<double> probs)
        : n_(probs.size()), gen_(std::random_device{}()),
          prob_(probs.size()), alias_(probs.size()),
          coin_(0.0, 1.0), dice_(0, static_cast<int>(probs.size()) - 1)
    {
        double n = static_cast<double>(n_);
        // 将概率归一化为均匀分布的倍数（每格期望面积 = 1）
        for (auto& p : probs) p *= n;

        std::vector<int> small, large;
        for (int i = 0; i < n_; ++i)
            (probs[i] < 1.0 ? small : large).push_back(i);

        // 将"小格"用"大格"填满
        while (!small.empty() && !large.empty()) {
            int s = small.back(); small.pop_back();
            int l = large.back(); large.pop_back();
            prob_[s]  = probs[s];
            alias_[s] = l;
            probs[l]  = (probs[l] + probs[s]) - 1.0;   // 减去分给 s 的部分
            (probs[l] < 1.0 ? small : large).push_back(l);
        }
        // 剩余数值接近 1.0，浮点误差导致分类不准确，强制置 1
        for (int i : small) prob_[i] = 1.0;
        for (int i : large) prob_[i] = 1.0;
    }

    int sample() {
        int i = dice_(gen_);                    // 随机选一列
        return (coin_(gen_) < prob_[i]) ? i : alias_[i];  // 按列内概率决定取原或别名
    }

private:
    int                                     n_;
    std::mt19937                            gen_;
    std::vector<double>                     prob_;   // 每列原概率
    std::vector<int>                        alias_;  // 每列的别名（填充列）
    std::uniform_real_distribution<double>  coin_;
    std::uniform_int_distribution<int>      dice_;
};
```

### 12. 循环 Buffer / 循环队列

#### 12.1 核心思路

固定大小数组实现的 FIFO 队列，头尾指针循环利用空间。 区分**满**与**空**：常见方案有两种——

1. **浪费一格**：`(tail + 1) % cap == head` 为满。
2. **额外计数器**（本文采用）：用 `size_` 变量，无歧义。

```cpp
#include <vector>
#include <stdexcept>

template<typename T>
class CircularBuffer {
public:
    explicit CircularBuffer(int capacity)
        : buf_(capacity), cap_(capacity), head_(0), tail_(0), size_(0) {}

    // 尾部入队
    void push(const T& val) {
        if (full())
            throw std::overflow_error("CircularBuffer is full");
        buf_[tail_] = val;
        tail_ = (tail_ + 1) % cap_;   // 尾指针循环前进
        ++size_;
    }

    // 头部出队，返回队首元素
    T pop() {
        if (empty())
            throw std::underflow_error("CircularBuffer is empty");
        T val = buf_[head_];
        head_ = (head_ + 1) % cap_;   // 头指针循环前进
        --size_;
        return val;
    }

    // 查看队首，不出队
    T& front() {
        if (empty()) throw std::underflow_error("CircularBuffer is empty");
        return buf_[head_];
    }

    bool empty() const { return size_ == 0; }
    bool full()  const { return size_ == cap_; }
    int  size()  const { return size_; }
    int  capacity() const { return cap_; }

private:
    std::vector<T> buf_;   // 底层存储
    int cap_;              // 总容量
    int head_;             // 队首下标（下一个出队位置）
    int tail_;             // 队尾下标（下一个入队位置）
    int size_;             // 当前元素数量
};
```

#### 12.2 无锁单生产者单消费者（SPSC）版

```cpp
#include <vector>
#include <atomic>
#include <optional>

// 无锁 SPSC 循环队列：生产者写 tail，消费者写 head，无需互斥锁
// 注意：cap 必须为 2 的幂，利用位掩码替代取模（更快）
template<typename T>
class SPSCQueue {
public:
    explicit SPSCQueue(size_t cap)
        : buf_(cap), mask_(cap - 1), head_(0), tail_(0) {
        // cap 必须是 2 的幂
        if ((cap & mask_) != 0)
            throw std::invalid_argument("Capacity must be a power of 2");
    }

    // 生产者调用：入队，失败返回 false（队满）
    bool push(const T& val) {
        size_t tail = tail_.load(std::memory_order_relaxed);
        size_t next = (tail + 1) & mask_;
        // 队满判断：下一个 tail 追上 head
        if (next == head_.load(std::memory_order_acquire)) return false;
        buf_[tail] = val;
        tail_.store(next, std::memory_order_release);   // release 保证写入可见
        return true;
    }

    // 消费者调用：出队，失败返回 nullopt（队空）
    std::optional<T> pop() {
        size_t head = head_.load(std::memory_order_relaxed);
        if (head == tail_.load(std::memory_order_acquire)) return std::nullopt;
        T val = buf_[head];
        head_.store((head + 1) & mask_, std::memory_order_release);
        return val;
    }

private:
    std::vector<T>         buf_;
    size_t                 mask_;                      // cap - 1，用于位与取模
    std::atomic<size_t>    head_;                      // 消费者端写
    char                   pad_[64 - sizeof(size_t)];  // 避免 head/tail 共享缓存行
    std::atomic<size_t>    tail_;                      // 生产者端写
};
```

### 1. 生产者-消费者

#### 1.1 核心思路

经典并发模型：生产者向共享缓冲区写数据，消费者从中读取。 核心问题：

- **互斥**：同一时刻只有一个线程访问缓冲区（`std::mutex`）。
- **同步**：缓冲区满时生产者阻塞；缓冲区空时消费者阻塞（`std::condition_variable`）。

两个条件变量 `not_full` / `not_empty` 精确唤醒，避免虚假唤醒（`while` 而非 `if`）。

```cpp
#include <queue>
#include <mutex>
#include <condition_variable>
#include <thread>
#include <iostream>
#include <optional>

template<typename T>
class BoundedQueue {
public:
    explicit BoundedQueue(size_t cap) : cap_(cap) {}

    // 生产者调用：阻塞直到有空位
    void push(T val) {
        std::unique_lock<std::mutex> lk(mu_);
        // 用 while 防止虚假唤醒（spurious wakeup）
        not_full_.wait(lk, [this]{ return q_.size() < cap_ || stopped_; });
        if (stopped_) return;
        q_.push(std::move(val));
        not_empty_.notify_one();   // 通知至少一个消费者
    }

    // 消费者调用：阻塞直到有数据；返回 nullopt 表示已关闭且队列为空
    std::optional<T> pop() {
        std::unique_lock<std::mutex> lk(mu_);
        not_empty_.wait(lk, [this]{ return !q_.empty() || stopped_; });
        if (q_.empty()) return std::nullopt;   // 关闭且无剩余数据
        T val = std::move(q_.front());
        q_.pop();
        not_full_.notify_one();    // 通知至少一个生产者
        return val;
    }

    // 关闭队列，唤醒所有阻塞线程
    void stop() {
        {
            std::lock_guard<std::mutex> lk(mu_);
            stopped_ = true;
        }
        not_full_.notify_all();
        not_empty_.notify_all();
    }

private:
    std::queue<T>           q_;
    size_t                  cap_;
    bool                    stopped_ = false;
    std::mutex              mu_;
    std::condition_variable not_full_;
    std::condition_variable not_empty_;
};

// ── 使用示例 ─────────────────────────────────────────────────────────────
void producerConsumerDemo() {
    BoundedQueue<int> bq(4);   // 容量为 4 的有界队列

    // 生产者线程：生产 10 个数字后关闭队列
    std::thread producer([&]{
        for (int i = 0; i < 10; ++i) {
            bq.push(i);
            std::cout << "[Producer] pushed " << i << "\n";
        }
        bq.stop();
    });

    // 消费者线程：持续消费直到队列关闭且为空
    std::thread consumer([&]{
        while (auto val = bq.pop()) {
            std::cout << "[Consumer] got " << *val << "\n";
        }
    });

    producer.join();
    consumer.join();
}
```

#### 1.2 多消费者扩展

上述 `BoundedQueue` 天然支持多生产者多消费者（MPMC）：

- `notify_one()` 保证每次只唤醒一个等待线程，避免惊群（thundering herd）。
- `notify_all()` 仅在 `stop()` 时调用，确保所有线程能退出。

### 2. 多线程循环按顺序打印递增数字

#### 2.1 问题定义

$N$ 个线程，线程 $i$（$0$-indexed）负责打印满足 $n \equiv i \pmod{N}$ 的数字，要求严格按 $1, 2, 3, \ldots$ 的顺序打印，总共打印到 $M$。

核心挑战：强制线程间的**全局有序执行**。

#### 2.2 方案一：条件变量 + 轮转令牌

```cpp
#include <thread>
#include <mutex>
#include <condition_variable>
#include <vector>
#include <iostream>

// N 个线程按序打印 1..M
// 使用"令牌"变量 turn 标记当前轮到哪个线程
void printInOrder(int N, int M) {
    std::mutex              mu;
    std::condition_variable cv;
    int                     turn    = 0;   // 当前应执行的线程编号
    int                     current = 1;   // 下一个要打印的数字

    std::vector<std::thread> threads;
    for (int id = 0; id < N; ++id) {
        threads.emplace_back([&, id] {
            while (true) {
                std::unique_lock<std::mutex> lk(mu);
                // 等待直到轮到自己，且还有数字可打印
                cv.wait(lk, [&]{ return turn == id || current > M; });
                if (current > M) break;

                std::cout << "Thread-" << id << " prints: " << current++ << "\n";
                turn = (turn + 1) % N;   // 令牌传给下一个线程
                cv.notify_all();          // 唤醒所有等待线程，让下一个线程接力
            }
        });
    }

    for (auto& t : threads) t.join();
}
```

#### 2.3 方案二：信号量链（更低唤醒开销）

```cpp
#include <thread>
#include <vector>
#include <semaphore>   // C++20
#include <iostream>

// 每个线程有自己的信号量，形成环形"接力棒"
// thread[i] 完成后 release thread[(i+1)%N] 的信号量
void printInOrderSemaphore(int N, int M) {
    // 初始化：thread-0 的信号量为 1（先行），其余为 0（等待）
    std::vector<std::counting_semaphore<1>> sems;
    sems.reserve(N);
    for (int i = 0; i < N; ++i)
        sems.emplace_back(i == 0 ? 1 : 0);

    int current = 1;   // 共享计数器（由信号量保证互斥，无需额外锁）
    std::vector<std::thread> threads;

    for (int id = 0; id < N; ++id) {
        threads.emplace_back([&, id] {
            while (true) {
                sems[id].acquire();               // 等待自己的令牌
                if (current > M) {
                    sems[(id + 1) % N].release(); // 唤醒下一个使其也能退出
                    break;
                }
                std::cout << "Thread-" << id << ": " << current++ << "\n";
                sems[(id + 1) % N].release();     // 传递令牌给下一个线程
            }
        });
    }

    for (auto& t : threads) t.join();
}
```

### 3. 手撕内存池（支持 `new[]` / `delete[]`）

#### 3.1 核心思路

标准 `new`/`delete` 每次调用系统分配器（`malloc`/`free`），有较高的系统调用开销和内存碎片风险。 **内存池**预先申请一大块内存，在其上手动管理分配/释放：

- **固定大小内存池**：所有对象大小相同，用侵入式空闲链表（free list）管理空闲块，分配/释放均 $O(1)$。
- 支持 `operator new[]` / `operator delete[]`：重载类级别运算符，池内管理数组分配。

```cpp
#include <cstddef>
#include <cstdlib>
#include <stdexcept>
#include <new>

// ── 固定对象大小内存池 ────────────────────────────────────────────────────
// 用法：替换特定类型的 new/delete，消除 malloc 开销
// 模板参数 ObjSize：每个对象的字节数；Capacity：池中最大对象数
template<size_t ObjSize, size_t Capacity>
class FixedMemoryPool {
    // 保证对象足够容纳一个指针（用于空闲链表节点）
    static constexpr size_t BlockSize = ObjSize > sizeof(void*) ? ObjSize : sizeof(void*);

public:
    FixedMemoryPool() {
        // 将原始内存切割成 Capacity 个等大块，串成链表
        // 每块的前 sizeof(void*) 字节存储下一个空闲块的指针
        for (size_t i = 0; i < Capacity - 1; ++i) {
            // 将当前块的起始位置解释为指针，指向下一块
            *reinterpret_cast<void**>(&arena_[i * BlockSize]) =
                &arena_[(i + 1) * BlockSize];
        }
        // 最后一块的 next 指针为 nullptr（链表尾）
        *reinterpret_cast<void**>(&arena_[(Capacity - 1) * BlockSize]) = nullptr;
        free_list_ = &arena_[0];   // 空闲链表头指向第一块
    }

    // 分配一个对象大小的内存块，O(1)
    void* allocate() {
        if (!free_list_)
            throw std::bad_alloc();       // 池已耗尽

        void* block = free_list_;         // 取出链表头
        free_list_ = *reinterpret_cast<void**>(free_list_);  // 头指针后移
        return block;
    }

    // 归还内存块到池，O(1)
    void deallocate(void* ptr) noexcept {
        // 将归还的块插回链表头（LIFO，有利于缓存局部性）
        *reinterpret_cast<void**>(ptr) = free_list_;
        free_list_ = ptr;
    }

    // 查询剩余可用块数（调试用）
    size_t available() const noexcept {
        size_t count = 0;
        void* p = free_list_;
        while (p) { ++count; p = *reinterpret_cast<void**>(p); }
        return count;
    }

private:
    alignas(std::max_align_t) char arena_[BlockSize * Capacity];  // 原始内存
    void* free_list_;   // 空闲链表头
};

// ── 将内存池集成到类，支持 new[] / delete[] ──────────────────────────────
// 每个 Foo 对象大小 16 字节，池容量 1024
struct Foo {
    int  data[4];   // 占 16 字节

    // 重载 operator new：从池中分配
    static void* operator new(size_t size) {
        if (size != sizeof(Foo)) return ::operator new(size);  // 非标准大小走系统分配
        return pool_.allocate();
    }

    // 重载 operator delete：归还到池
    static void operator delete(void* ptr) noexcept {
        pool_.deallocate(ptr);
    }

    // 重载 operator new[]：数组分配（需要额外存储元素数量，用于析构）
    static void* operator new[](size_t total_size) {
        // total_size = sizeof(Foo) * n + 可能的对齐填充（实现相关）
        // 简单方案：直接走系统分配器（数组 new 语义复杂，生产中用容器代替）
        return ::operator new(total_size);
    }

    // 重载 operator delete[]
    static void operator delete[](void* ptr) noexcept {
        ::operator delete(ptr);
    }

private:
    static FixedMemoryPool<sizeof(Foo), 1024> pool_;
};

// 类静态成员定义（放在 .cpp 中）
FixedMemoryPool<sizeof(Foo), 1024> Foo::pool_{};
```

#### 3.2 通用多 Size 分级内存池（面试进阶）

```cpp
#include <array>
#include <vector>
#include <cstddef>
#include <cstdlib>

// 类似 tcmalloc 的思路：按对象大小分级，每级维护独立空闲链表
// 此处简化为 8 个 Size Class：8, 16, 32, 64, 128, 256, 512, 1024 字节
class TieredMemoryPool {
    static constexpr size_t kNumClasses = 8;
    // 每级对应的块大小
    static constexpr size_t kSizes[kNumClasses] = {8,16,32,64,128,256,512,1024};

public:
    void* allocate(size_t size) {
        int cls = sizeClass(size);
        if (cls < 0) return std::malloc(size);   // 超大对象走系统分配

        if (free_lists_[cls]) {
            // 从对应级别的空闲链表取出一块
            void* block = free_lists_[cls];
            free_lists_[cls] = *reinterpret_cast<void**>(block);
            return block;
        }
        // 链表为空：从系统申请一批（批量摊还系统调用开销）
        return refill(cls);
    }

    void deallocate(void* ptr, size_t size) noexcept {
        int cls = sizeClass(size);
        if (cls < 0) { std::free(ptr); return; }
        // 归还：插回对应级别链表头
        *reinterpret_cast<void**>(ptr) = free_lists_[cls];
        free_lists_[cls] = ptr;
    }

private:
    void* free_lists_[kNumClasses] = {};   // 每级空闲链表头

    // 找到能容纳 size 的最小 Size Class，找不到返回 -1
    static int sizeClass(size_t size) {
        for (int i = 0; i < kNumClasses; ++i)
            if (size <= kSizes[i]) return i;
        return -1;
    }

    // 向系统申请一批块（默认 16 块），串成链表返回头部
    void* refill(int cls) {
        constexpr int kBatch = 16;
        size_t block_size = kSizes[cls];
        char* mem = static_cast<char*>(std::malloc(block_size * kBatch));
        if (!mem) throw std::bad_alloc();

        // 将 [1, kBatch-1] 串成链表
        for (int i = 0; i < kBatch - 1; ++i)
            *reinterpret_cast<void**>(mem + i * block_size) =
                mem + (i + 1) * block_size;
        *reinterpret_cast<void**>(mem + (kBatch-1) * block_size) = free_lists_[cls];
        free_lists_[cls] = mem + block_size;   // [1..] 进链表

        return mem;   // 返回第 0 块给调用方
    }
};
```

### 4. 智能指针

#### 4.1 `unique_ptr` 实现

**核心语义**：独占所有权，不可拷贝，可移动，析构时自动释放。 关键：禁用拷贝构造/赋值，实现移动构造/赋值，支持自定义 Deleter。

```cpp
#include <utility>    // std::exchange, std::forward
#include <functional> // std::default_delete（参考）

// ── MyUniquePtr<T, Deleter> ───────────────────────────────────────────────
template<typename T, typename Deleter = std::default_delete<T>>
class MyUniquePtr {
public:
    // 构造：接管裸指针（不隐式转换，explicit 防止意外赋值）
    explicit MyUniquePtr(T* ptr = nullptr, Deleter d = Deleter{})
        : ptr_(ptr), deleter_(std::move(d)) {}

    // 析构：调用 deleter 释放资源
    ~MyUniquePtr() {
        if (ptr_) deleter_(ptr_);
    }

    // ── 禁止拷贝（独占所有权）────────────────────────────────────────────
    MyUniquePtr(const MyUniquePtr&)            = delete;
    MyUniquePtr& operator=(const MyUniquePtr&) = delete;

    // ── 移动构造：转移所有权，来源置空 ──────────────────────────────────
    MyUniquePtr(MyUniquePtr&& other) noexcept
        : ptr_(std::exchange(other.ptr_, nullptr)),
          deleter_(std::move(other.deleter_)) {}

    // ── 移动赋值：先释放自身，再接管来源 ────────────────────────────────
    MyUniquePtr& operator=(MyUniquePtr&& other) noexcept {
        if (this != &other) {
            if (ptr_) deleter_(ptr_);                   // 释放当前持有资源
            ptr_     = std::exchange(other.ptr_, nullptr);
            deleter_ = std::move(other.deleter_);
        }
        return *this;
    }

    // ── 访问接口 ──────────────────────────────────────────────────────────
    T* get()           const noexcept { return ptr_; }
    T& operator*()     const          { return *ptr_; }
    T* operator->()    const noexcept { return ptr_; }
    explicit operator bool() const noexcept { return ptr_ != nullptr; }

    // 释放所有权（返回裸指针，调用方负责删除）
    T* release() noexcept { return std::exchange(ptr_, nullptr); }

    // 重置：释放当前资源，持有新指针
    void reset(T* ptr = nullptr) noexcept {
        if (ptr_) deleter_(ptr_);
        ptr_ = ptr;
    }

private:
    T*      ptr_;
    Deleter deleter_;
};

// make_unique（C++14 风格）
template<typename T, typename... Args>
MyUniquePtr<T> makeUnique(Args&&... args) {
    return MyUniquePtr<T>(new T(std::forward<Args>(args)...));
}
```

#### 4.2 `shared_ptr` 实现

**核心语义**：共享所有权，引用计数。引用计数归零时释放对象；`weak_ptr` 用弱引用，不影响计数。

```cpp
#include <atomic>
#include <utility>
#include <stdexcept>

// ── 控制块：存储引用计数 ────────────────────────────────────────────────
struct ControlBlock {
    std::atomic<int> strong_count{1};   // shared_ptr 数量
    std::atomic<int> weak_count{1};     // weak_ptr 数量（+1 表示 shared 本身的弱引用）

    virtual void destroyObject() noexcept = 0;  // 销毁被管理对象
    virtual void destroySelf()   noexcept = 0;  // 销毁控制块自身
    virtual ~ControlBlock() = default;
};

template<typename T>
struct ControlBlockImpl : ControlBlock {
    T* ptr;
    explicit ControlBlockImpl(T* p) : ptr(p) {}
    void destroyObject() noexcept override { delete ptr; ptr = nullptr; }
    void destroySelf()   noexcept override { delete this; }
};

// ── 前向声明 ──────────────────────────────────────────────────────────────
template<typename T> class MyWeakPtr;

// ── MySharedPtr<T> ────────────────────────────────────────────────────────
template<typename T>
class MySharedPtr {
    friend class MyWeakPtr<T>;
public:
    // 默认构造：空指针
    MySharedPtr() noexcept : ptr_(nullptr), cb_(nullptr) {}

    // 接管裸指针，创建控制块
    explicit MySharedPtr(T* ptr)
        : ptr_(ptr), cb_(ptr ? new ControlBlockImpl<T>(ptr) : nullptr) {}

    // 拷贝构造：共享所有权，strong_count +1
    MySharedPtr(const MySharedPtr& other) noexcept
        : ptr_(other.ptr_), cb_(other.cb_) {
        if (cb_) cb_->strong_count.fetch_add(1, std::memory_order_relaxed);
    }

    // 移动构造：转移所有权，来源置空（不改变引用计数）
    MySharedPtr(MySharedPtr&& other) noexcept
        : ptr_(std::exchange(other.ptr_, nullptr)),
          cb_ (std::exchange(other.cb_,  nullptr)) {}

    // 析构：strong_count -1；归零时销毁对象；weak_count 也归零时销毁控制块
    ~MySharedPtr() { release(); }

    // 拷贝赋值
    MySharedPtr& operator=(const MySharedPtr& other) noexcept {
        if (this != &other) {
            release();
            ptr_ = other.ptr_;
            cb_  = other.cb_;
            if (cb_) cb_->strong_count.fetch_add(1, std::memory_order_relaxed);
        }
        return *this;
    }

    // 移动赋值
    MySharedPtr& operator=(MySharedPtr&& other) noexcept {
        if (this != &other) {
            release();
            ptr_ = std::exchange(other.ptr_, nullptr);
            cb_  = std::exchange(other.cb_,  nullptr);
        }
        return *this;
    }

    T* get()           const noexcept { return ptr_; }
    T& operator*()     const          { return *ptr_; }
    T* operator->()    const noexcept { return ptr_; }
    explicit operator bool() const noexcept { return ptr_ != nullptr; }

    int use_count() const noexcept {
        return cb_ ? cb_->strong_count.load(std::memory_order_relaxed) : 0;
    }

private:
    T*            ptr_;
    ControlBlock* cb_;

    // 供 weak_ptr::lock() 调用，从控制块直接构造（不再 new）
    MySharedPtr(T* ptr, ControlBlock* cb) : ptr_(ptr), cb_(cb) {
        if (cb_) cb_->strong_count.fetch_add(1, std::memory_order_relaxed);
    }

    void release() noexcept {
        if (!cb_) return;
        // strong_count 减 1，若归零则销毁对象
        if (cb_->strong_count.fetch_sub(1, std::memory_order_acq_rel) == 1) {
            cb_->destroyObject();
            // weak_count 减 1（对应 strong 持有的那个弱引用），若归零则销毁控制块
            if (cb_->weak_count.fetch_sub(1, std::memory_order_acq_rel) == 1)
                cb_->destroySelf();
        }
        ptr_ = nullptr;
        cb_  = nullptr;
    }
};

// ── MyWeakPtr<T> ──────────────────────────────────────────────────────────
template<typename T>
class MyWeakPtr {
public:
    MyWeakPtr() noexcept : ptr_(nullptr), cb_(nullptr) {}

    // 从 shared_ptr 构造：weak_count +1（不影响 strong_count）
    MyWeakPtr(const MySharedPtr<T>& sp) noexcept
        : ptr_(sp.ptr_), cb_(sp.cb_) {
        if (cb_) cb_->weak_count.fetch_add(1, std::memory_order_relaxed);
    }

    ~MyWeakPtr() { release(); }

    MyWeakPtr(const MyWeakPtr& other) noexcept : ptr_(other.ptr_), cb_(other.cb_) {
        if (cb_) cb_->weak_count.fetch_add(1, std::memory_order_relaxed);
    }

    MyWeakPtr& operator=(const MyWeakPtr& other) noexcept {
        if (this != &other) {
            release();
            ptr_ = other.ptr_; cb_ = other.cb_;
            if (cb_) cb_->weak_count.fetch_add(1, std::memory_order_relaxed);
        }
        return *this;
    }

    bool expired() const noexcept {
        return !cb_ || cb_->strong_count.load(std::memory_order_acquire) == 0;
    }

    // 尝试提升为 shared_ptr；若对象已销毁则返回空 shared_ptr
    MySharedPtr<T> lock() const noexcept {
        if (expired()) return MySharedPtr<T>{};
        // 用 CAS 原子地将 strong_count 从非零值 +1，防止竞态
        int cnt = cb_->strong_count.load(std::memory_order_relaxed);
        while (cnt > 0) {
            if (cb_->strong_count.compare_exchange_weak(
                    cnt, cnt + 1,
                    std::memory_order_acq_rel,
                    std::memory_order_relaxed))
                return MySharedPtr<T>(ptr_, cb_);  // 使用私有构造，不再 +1
        }
        return MySharedPtr<T>{};   // 对象已销毁（strong_count 已降至 0）
    }

private:
    T*            ptr_;
    ControlBlock* cb_;

    void release() noexcept {
        if (!cb_) return;
        if (cb_->weak_count.fetch_sub(1, std::memory_order_acq_rel) == 1)
            cb_->destroySelf();   // 最后一个 weak_ptr 销毁时回收控制块
        ptr_ = nullptr; cb_ = nullptr;
    }
};

// make_shared（合并对象和控制块为一次 new，减少内存碎片）
template<typename T, typename... Args>
MySharedPtr<T> makeShared(Args&&... args) {
    return MySharedPtr<T>(new T(std::forward<Args>(args)...));
    // 注：真正的 make_shared 将对象嵌入控制块一次性分配，此处为简化版
}
```

#### 4.3 常见面试追问

|问题|答案|
|---|---|
|`shared_ptr` 引用计数为何用 `atomic`？|多线程同时拷贝/析构时并发修改计数，需要原子操作保证不撕裂|
|为何 `shared_ptr` 本身不是线程安全的？|引用计数是原子的，但 `ptr_` 的读写不是；同时读写同一 `shared_ptr` 对象需外部加锁|
|`weak_ptr` 解决什么问题？|打破循环引用（A→B→A 导致内存泄漏）|
|`make_shared` vs `new` 传入 `shared_ptr`？|`make_shared` 一次分配合并对象和控制块，减少 1 次 `malloc`，且异常安全|
|为何 `unique_ptr` 不需要原子计数？|独占所有权，同一时刻只有一个所有者，无并发竞争|

### 1. 比较两个 `float` 是否相等

#### 1.1 核心思路

IEEE 754 浮点数不可用 `==` 直接比较。两种方案：

- **绝对误差**：$|a - b| < \varepsilon$，适合接近 0 的数。
- **相对误差**：$\dfrac{|a-b|}{\max(|a|,|b|)} < \varepsilon$，适合量级较大的数。
- **ULP（Units in the Last Place）**：比较二进制表示的距离，最通用。

```cpp
#include <cmath>
#include <cstdint>
#include <cstring>
#include <algorithm>
#include <limits>

// ── 方法 1：绝对误差（适合接近 0 的场景）────────────────────────────────
bool floatEqAbsolute(float a, float b, float eps = 1e-6f) {
    return std::fabs(a - b) < eps;
}

// ── 方法 2：相对 + 绝对组合（推荐通用方案）──────────────────────────────
// rel_eps：相对误差阈值；abs_eps：兜底绝对误差（防止除以接近 0 的数）
bool floatEqRelative(float a, float b,
                     float rel_eps = 1e-5f,
                     float abs_eps = 1e-8f) {
    float diff = std::fabs(a - b);
    if (diff < abs_eps) return true;   // 绝对意义上足够近
    float largest = std::max(std::fabs(a), std::fabs(b));
    return diff < rel_eps * largest;   // 相对误差检查
}

// ── 方法 3：ULP 比较（最精确，符合 IEEE 754 语义）────────────────────────
// 将 float 的二进制表示转为有符号整数，差值即为 ULP 距离
bool floatEqULP(float a, float b, int max_ulp = 4) {
    // NaN 不等于任何值（包括自身）
    if (std::isnan(a) || std::isnan(b)) return false;
    // 符号不同且均非零时，不相等（+0 和 -0 的 bit 表示不同但应视为相等）
    if (std::signbit(a) != std::signbit(b))
        return a == b;   // 仅 +0 == -0 返回 true

    // 通过 memcpy 将 float 的位模式无 UB 地转为 int32
    int32_t ia, ib;
    std::memcpy(&ia, &a, sizeof(float));
    std::memcpy(&ib, &b, sizeof(float));

    return std::abs(ia - ib) <= max_ulp;
}
```

### 2. 线性插值

#### 2.1 公式

在两点 $(x_0, y_0)$、$(x_1, y_1)$ 之间，对给定 $x$ 的插值：

$$ y = y_0 + \frac{x - x_0}{x_1 - x_0}(y_1 - y_0) = \text{lerp}(y_0,\ y_1,\ t), \quad t = \frac{x - x_0}{x_1 - x_0} $$

```cpp
// ── 基本 lerp（C++20 std::lerp 的等价实现）───────────────────────────────
// lerp(a, b, t)：当 t=0 返回 a，t=1 返回 b，t∈(0,1) 线性插值
// 精度写法：a + t*(b-a) 在 t=1 时有误差；下式在端点处精确
inline float lerp(float a, float b, float t) {
    return (1.0f - t) * a + t * b;   // 等价 a*(1-t) + b*t，保证 t=0/1 精确
}

// ── 根据 x 坐标插值 ──────────────────────────────────────────────────────
float linearInterp(float x0, float y0, float x1, float y1, float x) {
    if (x0 == x1) return y0;   // 避免除以零
    float t = (x - x0) / (x1 - x0);
    return lerp(y0, y1, t);
}
```

### 3. 双线性插值

#### 3.1 公式

在矩形网格四角 $(x_0,y_0),(x_1,y_0),(x_0,y_1),(x_1,y_1)$ 已知函数值 $Q_{00},Q_{10},Q_{01},Q_{11}$ 的条件下，对内部点 $(x,y)$ 插值：

$$ f(x,y) \approx \frac{1}{(x_1-x_0)(y_1-y_0)} \Big[(x_1-x)(y_1-y),Q_{00} + (x-x_0)(y_1-y),Q_{10}

- (x_1-x)(y-y_0),Q_{01} + (x-x_0)(y-y_0),Q_{11}\Big] $$

等价为先在 $x$ 方向做两次线性插值，再在 $y$ 方向做一次：

$$ R_0 = \text{lerp}(Q_{00}, Q_{10},\ t_x), \quad R_1 = \text{lerp}(Q_{01}, Q_{11},\ t_x), \quad f = \text{lerp}(R_0, R_1,\ t_y) $$

```cpp
// Q00=左下, Q10=右下, Q01=左上, Q11=右上
// tx = (x - x0)/(x1 - x0), ty = (y - y0)/(y1 - y0)
float bilinearInterp(float Q00, float Q10, float Q01, float Q11,
                     float tx,  float ty) {
    float R0 = lerp(Q00, Q10, tx);   // 下边沿插值（y = y0）
    float R1 = lerp(Q01, Q11, tx);   // 上边沿插值（y = y1）
    return lerp(R0, R1, ty);          // 在 y 方向合并
}

// 更常见的图像采样形式：输入 (u, v) ∈ [0, W-1] × [0, H-1]（非整数坐标）
float sampleBilinear(const float* img, int W, int H, float u, float v) {
    // 取整得到四个相邻像素坐标，clamp 防越界
    int x0 = std::max(0, std::min((int)u,     W - 1));
    int x1 = std::max(0, std::min((int)u + 1, W - 1));
    int y0 = std::max(0, std::min((int)v,     H - 1));
    int y1 = std::max(0, std::min((int)v + 1, H - 1));

    float tx = u - (int)u;   // 小数部分
    float ty = v - (int)v;

    float Q00 = img[y0 * W + x0];
    float Q10 = img[y0 * W + x1];
    float Q01 = img[y1 * W + x0];
    float Q11 = img[y1 * W + x1];

    return bilinearInterp(Q00, Q10, Q01, Q11, tx, ty);
}
```

## 4. 向量外积（Cross Product）

### 4.1 公式

三维向量 $\mathbf{a} = (a_x, a_y, a_z)$，$\mathbf{b} = (b_x, b_y, b_z)$：

$$ \mathbf{a} \times \mathbf{b} = \begin{vmatrix} \mathbf{i} & \mathbf{j} & \mathbf{k} \ a_x & a_y & a_z \ b_x & b_y & b_z \end{vmatrix} = (a_y b_z - a_z b_y,; a_z b_x - a_x b_z,; a_x b_y - a_y b_x) $$

结果向量垂直于 $\mathbf{a}$、$\mathbf{b}$ 所在平面，模长 $= |\mathbf{a}||\mathbf{b}|\sin\theta$。

```cpp
#include <array>
#include <cmath>

using Vec3 = std::array<float, 3>;

Vec3 cross(const Vec3& a, const Vec3& b) {
    return {
        a[1]*b[2] - a[2]*b[1],   // i 分量
        a[2]*b[0] - a[0]*b[2],   // j 分量
        a[0]*b[1] - a[1]*b[0]    // k 分量
    };
}

float dot(const Vec3& a, const Vec3& b) {
    return a[0]*b[0] + a[1]*b[1] + a[2]*b[2];
}

float norm(const Vec3& a) {
    return std::sqrt(dot(a, a));
}
```

## 5. 矩阵链乘：$(A_{1\times256} \cdot B_{256\times128}) \cdot C_{128\times256}$

### 5.1 最优括号顺序分析

设矩阵维度 $A_{1\times256}$，$B_{256\times128}$，$C_{128\times256}$。

**方案一**：$(A \cdot B) \cdot C$

$$ \text{cost}_1 = 1 \times 256 \times 128 + 1 \times 128 \times 256 = 32768 + 32768 = 65536 $$

**方案二**：$A \cdot (B \cdot C)$

$$ \text{cost}_2 = 256 \times 128 \times 256 + 1 \times 256 \times 256 = 8388608 + 65536 \gg \text{cost}_1 $$

**结论**：先算 $A \cdot B$，代价 $65536$，远优于先算 $B \cdot C$（$8M+$）。

```cpp
#include <vector>
#include <cassert>
#include <cstring>

using Matrix = std::vector<std::vector<float>>;

// 矩阵乘法：C = A(m×k) · B(k×n)，返回 m×n 矩阵
Matrix matMul(const Matrix& A, const Matrix& B) {
    int m = static_cast<int>(A.size());
    int k = static_cast<int>(B.size());
    int n = static_cast<int>(B[0].size());
    assert(static_cast<int>(A[0].size()) == k);

    Matrix C(m, std::vector<float>(n, 0.0f));
    for (int i = 0; i < m; ++i)
        for (int p = 0; p < k; ++p)       // 循环顺序 i-p-j 对缓存友好（A行连续）
            for (int j = 0; j < n; ++j)
                C[i][j] += A[i][p] * B[p][j];
    return C;
}

// 链乘：先 (A·B)，再结果·C
Matrix chainMul(const Matrix& A, const Matrix& B, const Matrix& C) {
    auto AB = matMul(A, B);    // A(1×256) · B(256×128) → AB(1×128)
    return matMul(AB, C);      // AB(1×128) · C(128×256) → result(1×256)
}
```

## 6. 交叉熵损失（Cross-Entropy Loss）

### 6.1 公式

对于 $C$ 类分类，单样本损失：

$$ \mathcal{L} = -\sum_{c=1}^{C} y_c \log(\hat{p}_c) $$

其中 $y_c \in {0,1}$ 为 one-hot 标签，$\hat{p}_c = \text{Softmax}(\mathbf{z})_c$。 数值稳定写法（log-sum-exp）：

$$ \log \hat{p}_c = z_c - \log!\sum_{j} e^{z_j} = z_c - \left(z_{\max} + \log!\sum_{j} e^{z_j - z_{\max}}\right) $$

```cpp
#include <vector>
#include <cmath>
#include <algorithm>
#include <numeric>

// logits：模型原始输出（未经 softmax）；label：真实类别下标
// 返回单样本交叉熵损失（数值稳定版）
float crossEntropyLoss(const std::vector<float>& logits, int label) {
    // 数值稳定：减去最大值后再做 exp，防止上溢
    float max_logit = *std::max_element(logits.begin(), logits.end());

    float log_sum_exp = 0.0f;
    for (float z : logits)
        log_sum_exp += std::exp(z - max_logit);
    log_sum_exp = std::log(log_sum_exp) + max_logit;   // log Σ exp(z_j)

    // loss = -z_label + log_sum_exp（等价于 -log softmax(label)）
    return log_sum_exp - logits[label];
}

// Batch 版本：对多个样本取平均
float batchCrossEntropyLoss(const std::vector<std::vector<float>>& batch_logits,
                             const std::vector<int>& labels) {
    float total = 0.0f;
    for (size_t i = 0; i < batch_logits.size(); ++i)
        total += crossEntropyLoss(batch_logits[i], labels[i]);
    return total / static_cast<float>(batch_logits.size());
}
```

## 7. 线性回归 SGD

### 7.1 模型与梯度

模型：$\hat{y} = \mathbf{w}^\top \mathbf{x} + b$

MSE 损失：$\mathcal{L} = \dfrac{1}{2N}\sum_{i=1}^{N}(\hat{y}_i - y_i)^2$

梯度（单样本）：

$$ \frac{\partial \mathcal{L}}{\partial \mathbf{w}} = (\hat{y} - y),\mathbf{x}, \qquad \frac{\partial \mathcal{L}}{\partial b} = (\hat{y} - y) $$

SGD 更新（学习率 $\eta$）：

$$ \mathbf{w} \leftarrow \mathbf{w} - \eta(\hat{y}-y)\mathbf{x}, \qquad b \leftarrow b - \eta(\hat{y}-y) $$

```cpp
#include <vector>
#include <random>
#include <numeric>
#include <cmath>

class LinearRegression {
public:
    explicit LinearRegression(int features, float lr = 0.01f)
        : w_(features, 0.0f), b_(0.0f), lr_(lr) {}

    // 前向推理：ŷ = w·x + b
    float predict(const std::vector<float>& x) const {
        float y = b_;
        for (size_t i = 0; i < w_.size(); ++i)
            y += w_[i] * x[i];
        return y;
    }

    // 单样本 SGD 更新
    void sgdStep(const std::vector<float>& x, float y_true) {
        float y_hat = predict(x);
        float err   = y_hat - y_true;   // 残差 (ŷ - y)

        // 梯度下降：w -= lr * err * x
        for (size_t i = 0; i < w_.size(); ++i)
            w_[i] -= lr_ * err * x[i];
        b_ -= lr_ * err;
    }

    // 训练 epochs 轮（每轮随机打乱顺序）
    void fit(const std::vector<std::vector<float>>& X,
             const std::vector<float>& Y, int epochs = 100) {
        std::mt19937 rng(42);
        std::vector<int> idx(X.size());
        std::iota(idx.begin(), idx.end(), 0);

        for (int e = 0; e < epochs; ++e) {
            std::shuffle(idx.begin(), idx.end(), rng);  // 随机打乱，减少梯度相关性
            for (int i : idx)
                sgdStep(X[i], Y[i]);
        }
    }

    const std::vector<float>& weights() const { return w_; }
    float bias() const { return b_; }

private:
    std::vector<float> w_;
    float              b_;
    float              lr_;
};
```

## 8. Softmax

### 8.1 公式与数值稳定性

$$ \text{Softmax}(z_i) = \frac{e^{z_i}}{\sum_{j=1}^{C} e^{z_j}} $$

数值稳定写法：令 $m = \max_j z_j$，则：

$$ \text{Softmax}(z_i) = \frac{e^{z_i - m}}{\sum_{j} e^{z_j - m}} $$

减去 $m$ 不改变结果（分子分母同乘 $e^{-m}$），但避免 $e^{z_i}$ 上溢。

```cpp
#include <vector>
#include <cmath>
#include <algorithm>
#include <numeric>

// in-place Softmax（数值稳定版）
void softmax(std::vector<float>& x) {
    // 1. 找最大值，做数值稳定偏移
    float max_val = *std::max_element(x.begin(), x.end());

    // 2. exp(x_i - max)
    float sum = 0.0f;
    for (float& v : x) {
        v    = std::exp(v - max_val);
        sum += v;
    }

    // 3. 归一化
    for (float& v : x) v /= sum;
}

// 返回新 vector 的版本
std::vector<float> softmaxCopy(const std::vector<float>& logits) {
    std::vector<float> out(logits);
    softmax(out);
    return out;
}
```

## 9. Online Softmax

### 9.1 核心思路

**问题**：标准 Softmax 需要两次遍历（第一次找 max，第二次算 exp 和归一化）。 **Online Softmax** 在一次遍历中同时维护最大值 $m$ 和修正后的归一化因子 $d$，实现单 Pass 计算。这是 Flash Attention 的数学基础。

**维护不变式**：处理到元素 $x_j$ 时，

$$ m_j = \max_{i \le j} x_i, \qquad d_j = \sum_{i \le j} e^{x_i - m_j} $$

当新元素 $x_{j+1}$ 到来时，若 $x_{j+1} > m_j$，需修正旧的 $d_j$：

$$ m_{j+1} = x_{j+1}, \qquad d_{j+1} = d_j \cdot e^{m_j - m_{j+1}} + 1 $$

否则（$x_{j+1} \le m_j$）：

$$ m_{j+1} = m_j, \qquad d_{j+1} = d_j + e^{x_{j+1} - m_j} $$

合并写法：

$$ m_{j+1} = \max(m_j, x_{j+1}), \qquad d_{j+1} = d_j \cdot e^{m_j - m_{j+1}} + e^{x_{j+1} - m_{j+1}} $$

```cpp
#include <vector>
#include <cmath>
#include <limits>

// Online Softmax：单次遍历，输出 softmax 概率向量
// 同时输出 log_sum_exp（供 log-softmax 等场景使用）
std::vector<float> onlineSoftmax(const std::vector<float>& x) {
    float m = -std::numeric_limits<float>::infinity();   // 当前最大值
    float d = 0.0f;   // 修正后的 exp 累加和（对应当前 m 的归一化因子）

    // Pass 1：单次遍历，同时维护 m 和 d
    for (float xi : x) {
        float new_m = std::max(m, xi);
        // 旧的 d 是基于 m 计算的，新 m 变化需要修正：乘以 e^(m - new_m)
        d = d * std::exp(m - new_m) + std::exp(xi - new_m);
        m = new_m;
    }

    // Pass 2：输出每个元素的 softmax 值
    std::vector<float> out(x.size());
    for (size_t i = 0; i < x.size(); ++i)
        out[i] = std::exp(x[i] - m) / d;

    return out;
}

// 完全单 Pass 版本（同时输出结果，需要额外存 exp 中间值）
std::vector<float> onlineSoftmaxSinglePass(const std::vector<float>& x) {
    int n = static_cast<int>(x.size());
    std::vector<float> exp_vals(n);

    float m = -std::numeric_limits<float>::infinity();
    float d = 0.0f;

    for (int i = 0; i < n; ++i) {
        float new_m = std::max(m, x[i]);
        // 修正之前存储的 exp 值（基准从 m 改为 new_m，需同比缩放）
        if (new_m != m) {
            float scale = std::exp(m - new_m);
            for (int j = 0; j < i; ++j) exp_vals[j] *= scale;
            d *= scale;
        }
        exp_vals[i] = std::exp(x[i] - new_m);
        d += exp_vals[i];
        m  = new_m;
    }

    for (float& v : exp_vals) v /= d;
    return exp_vals;
}
```

## 10. Layer Norm

### 10.1 公式

对长度为 $d$ 的向量 $\mathbf{x}$（一个样本的隐层激活）：

$$ \mu = \frac{1}{d}\sum_{i=1}^{d} x_i, \qquad \sigma^2 = \frac{1}{d}\sum_{i=1}^{d}(x_i - \mu)^2 $$

$$ \hat{x}_i = \frac{x_i - \mu}{\sqrt{\sigma^2 + \varepsilon}}, \qquad y_i = \gamma_i \hat{x}_i + \beta_i $$

其中 $\gamma, \beta$ 为可学习的缩放和偏移（Scale & Shift），$\varepsilon$ 防除零。

```cpp
#include <vector>
#include <cmath>
#include <stdexcept>

// in-place Layer Norm
// gamma, beta：可学习参数（shape 与 x 相同）；eps：数值稳定项
void layerNorm(std::vector<float>& x,
               const std::vector<float>& gamma,
               const std::vector<float>& beta,
               float eps = 1e-5f) {
    int d = static_cast<int>(x.size());
    if (d == 0) return;

    // 1. 计算均值 μ
    float mean = 0.0f;
    for (float v : x) mean += v;
    mean /= static_cast<float>(d);

    // 2. 计算方差 σ²（两次遍历，数值更稳定；Welford 算法也可）
    float var = 0.0f;
    for (float v : x) {
        float diff = v - mean;
        var += diff * diff;
    }
    var /= static_cast<float>(d);

    // 3. 归一化 + 仿射变换
    float inv_std = 1.0f / std::sqrt(var + eps);
    for (int i = 0; i < d; ++i)
        x[i] = gamma[i] * ((x[i] - mean) * inv_std) + beta[i];
}

// RMSNorm（无均值中心化，更高效，LLaMA 使用）
// y_i = (x_i / RMS(x)) * γ_i, RMS(x) = sqrt(1/d * Σ x_i²)
void rmsNorm(std::vector<float>& x,
             const std::vector<float>& gamma,
             float eps = 1e-6f) {
    int d = static_cast<int>(x.size());
    float rms = 0.0f;
    for (float v : x) rms += v * v;
    rms = std::sqrt(rms / static_cast<float>(d) + eps);
    float inv_rms = 1.0f / rms;
    for (int i = 0; i < d; ++i)
        x[i] = x[i] * inv_rms * gamma[i];
}
```

## 11. Attention 计算

### 11.1 公式

给定查询矩阵 $Q \in \mathbb{R}^{T_q \times d_k}$，键 $K \in \mathbb{R}^{T_k \times d_k}$，值 $V \in \mathbb{R}^{T_k \times d_v}$：

$$ \text{Attention}(Q,K,V) = \text{Softmax}!\left(\frac{QK^\top}{\sqrt{d_k}}\right) V $$

```cpp
#include <vector>
#include <cmath>
#include <algorithm>

using Mat = std::vector<std::vector<float>>;

// 矩阵乘法 C = A(m×k) · B(k×n)
Mat matMulBasic(const Mat& A, const Mat& B) {
    int m = A.size(), k = B.size(), n = B[0].size();
    Mat C(m, std::vector<float>(n, 0.0f));
    for (int i = 0; i < m; ++i)
        for (int p = 0; p < k; ++p)
            for (int j = 0; j < n; ++j)
                C[i][j] += A[i][p] * B[p][j];
    return C;
}

// 矩阵转置
Mat transpose(const Mat& A) {
    int m = A.size(), n = A[0].size();
    Mat B(n, std::vector<float>(m));
    for (int i = 0; i < m; ++i)
        for (int j = 0; j < n; ++j)
            B[j][i] = A[i][j];
    return B;
}

// 行 Softmax（in-place）
void rowSoftmax(Mat& A) {
    for (auto& row : A) {
        float mx = *std::max_element(row.begin(), row.end());
        float sum = 0.0f;
        for (float& v : row) { v = std::exp(v - mx); sum += v; }
        for (float& v : row)   v /= sum;
    }
}

// Scaled Dot-Product Attention
// Q: [Tq, dk], K: [Tk, dk], V: [Tk, dv]
// 返回: [Tq, dv]
Mat scaledDotProductAttention(const Mat& Q, const Mat& K, const Mat& V,
                               bool causal_mask = false) {
    int Tq = Q.size(), dk = Q[0].size(), Tk = K.size();
    float scale = 1.0f / std::sqrt(static_cast<float>(dk));

    // 1. 计算 scores = Q · K^T / sqrt(dk)，shape [Tq, Tk]
    Mat KT     = transpose(K);
    Mat scores = matMulBasic(Q, KT);
    for (auto& row : scores)
        for (float& v : row) v *= scale;

    // 2. 因果掩码（自回归场景）：position j > i 的位置置为 -inf
    if (causal_mask) {
        for (int i = 0; i < Tq; ++i)
            for (int j = i + 1; j < Tk; ++j)
                scores[i][j] = -std::numeric_limits<float>::infinity();
    }

    // 3. Softmax（逐行）
    rowSoftmax(scores);

    // 4. 加权求和：output = scores · V，shape [Tq, dv]
    return matMulBasic(scores, V);
}
```

## 12. Linear Attention

### 12.1 核心思路

标准 Attention 复杂度 $O(T^2 d)$，Linear Attention 通过核函数近似将复杂度降至 $O(Td^2)$。

**核技巧**：将 $\text{softmax}(QK^\top)$ 近似为 $\phi(Q)\phi(K)^\top$，其中 $\phi(\cdot)$ 为特征映射（常用 ELU+1 或随机特征）。

利用矩阵结合律：

$$ \text{Linear-Attention}(Q,K,V) = \frac{\phi(Q),[\phi(K)^\top V]}{\phi(Q),[\phi(K)^\top \mathbf{1}]} $$

先算 $\phi(K)^\top V$（$d \times d_v$ 矩阵），消去序列维度 $T$。

```cpp
#include <vector>
#include <cmath>
#include <algorithm>

// 特征映射 φ(x)_i = max(x_i, 0) + 1（ELU+1，保证非负）
std::vector<float> featureMap(const std::vector<float>& x) {
    std::vector<float> out(x.size());
    for (size_t i = 0; i < x.size(); ++i)
        out[i] = (x[i] >= 0.0f) ? (x[i] + 1.0f) : (std::exp(x[i]));
    return out;
}

// Linear Attention（简化版，单头）
// Q,K,V 均为 [T, d] 二维向量
Mat linearAttention(const Mat& Q, const Mat& K, const Mat& V) {
    int T = Q.size(), d = Q[0].size(), dv = V[0].size();

    // 1. 对 Q, K 逐行应用特征映射
    Mat phiQ(T), phiK(T);
    for (int i = 0; i < T; ++i) {
        phiQ[i] = featureMap(Q[i]);
        phiK[i] = featureMap(K[i]);
    }

    // 2. 计算 S = φ(K)^T · V，shape [d, dv]（与 T 无关）
    Mat S(d, std::vector<float>(dv, 0.0f));
    for (int t = 0; t < T; ++t)
        for (int i = 0; i < d;  ++i)
            for (int j = 0; j < dv; ++j)
                S[i][j] += phiK[t][i] * V[t][j];

    // 3. 计算归一化因子 z = φ(K)^T · 1，shape [d]
    std::vector<float> z(d, 0.0f);
    for (int t = 0; t < T; ++t)
        for (int i = 0; i < d; ++i)
            z[i] += phiK[t][i];

    // 4. 输出：output[t] = φ(Q[t]) · S / (φ(Q[t]) · z)
    Mat out(T, std::vector<float>(dv, 0.0f));
    for (int t = 0; t < T; ++t) {
        float norm = 0.0f;
        for (int i = 0; i < d; ++i) norm += phiQ[t][i] * z[i];
        if (norm < 1e-6f) norm = 1e-6f;   // 防除零
        for (int j = 0; j < dv; ++j) {
            float val = 0.0f;
            for (int i = 0; i < d; ++i) val += phiQ[t][i] * S[i][j];
            out[t][j] = val / norm;
        }
    }
    return out;
}
```

## 13. MoE（Mixture of Experts）前向推理

### 13.1 架构概述

$$ \text{MoE}(\mathbf{x}) = \sum_{i=1}^{K} g_i(\mathbf{x}), E_i(\mathbf{x}) $$

- **Router（门控网络）**：$g(\mathbf{x}) = \text{TopK-Softmax}(W_g \mathbf{x})$，仅激活 $K$ 个 Expert（通常 $K=2$）。
- **Expert**：每个 Expert 为独立的 FFN 子网络。
- **稀疏激活**：每个 token 只经过 $K/N$ 的参数量，推理计算量大幅降低。

```cpp
#include <vector>
#include <algorithm>
#include <numeric>
#include <cmath>

// 模拟单个 Expert（线性层 + ReLU）
std::vector<float> expertForward(const std::vector<float>& x,
                                  const Mat& W1, const Mat& W2) {
    // W1: [hidden_dim, d_model], W2: [d_model, hidden_dim]
    int hidden = W1.size(), d = x.size();
    std::vector<float> h(hidden, 0.0f);

    // 上投影 + ReLU
    for (int i = 0; i < hidden; ++i) {
        for (int j = 0; j < d; ++j) h[i] += W1[i][j] * x[j];
        h[i] = std::max(0.0f, h[i]);   // ReLU
    }

    // 下投影
    std::vector<float> out(d, 0.0f);
    for (int j = 0; j < d; ++j)
        for (int i = 0; i < hidden; ++i)
            out[j] += W2[j][i] * h[i];
    return out;
}

// MoE 前向：TopK 路由 + Expert 加权求和
// x: [d_model]; router_w: [num_experts, d_model]; experts: num_experts 个 FFN
std::vector<float> moeForward(
    const std::vector<float>& x,
    const Mat& router_w,                   // 路由权重矩阵
    const std::vector<std::pair<Mat,Mat>>& experts,  // 每个 Expert 的 (W1, W2)
    int top_k = 2)
{
    int num_experts = static_cast<int>(experts.size());
    int d = static_cast<int>(x.size());

    // 1. 计算路由 logits = router_w · x
    std::vector<float> logits(num_experts, 0.0f);
    for (int e = 0; e < num_experts; ++e)
        for (int j = 0; j < d; ++j)
            logits[e] += router_w[e][j] * x[j];

    // 2. TopK 选择：找最大的 top_k 个 Expert
    std::vector<int> indices(num_experts);
    std::iota(indices.begin(), indices.end(), 0);
    std::partial_sort(indices.begin(), indices.begin() + top_k, indices.end(),
                      [&](int a, int b){ return logits[a] > logits[b]; });
    indices.resize(top_k);

    // 3. Softmax 归一化所选 Expert 的权重（只对 top_k 个做 softmax）
    float max_logit = -std::numeric_limits<float>::infinity();
    for (int i : indices) max_logit = std::max(max_logit, logits[i]);
    float sum = 0.0f;
    for (int i : indices) { logits[i] = std::exp(logits[i] - max_logit); sum += logits[i]; }
    for (int i : indices)   logits[i] /= sum;

    // 4. 加权聚合各 Expert 输出
    std::vector<float> out(d, 0.0f);
    for (int i : indices) {
        auto expert_out = expertForward(x, experts[i].first, experts[i].second);
        for (int j = 0; j < d; ++j)
            out[j] += logits[i] * expert_out[j];
    }
    return out;
}
```

## 14. Flow Matching Model 采样

### 14.1 核心思路

Flow Matching 学习一个速度场 $v_\theta(\mathbf{x}, t)$，使得从噪声 $\mathbf{x}_0 \sim \mathcal{N}(0,I)$ 积分 ODE 到 $t=1$ 得到数据样本：

$$ \frac{d\mathbf{x}}{dt} = v_\theta(\mathbf{x}_t,\ t) $$

采样过程即求解此 ODE：常用 **Euler 法**（一阶）或 **Heun 法**（二阶，精度更高）。

**Euler 法**：$\mathbf{x}_{t+h} = \mathbf{x}_t + h \cdot v_\theta(\mathbf{x}_t, t)$

**Heun 法（Predictor-Corrector）**：

$$ \tilde{\mathbf{x}}_{t+h} = \mathbf{x}_t + h \cdot v_\theta(\mathbf{x}_t, t) $$ $$ \mathbf{x}_{t+h} = \mathbf{x}_t + \frac{h}{2}\left[v_\theta(\mathbf{x}_t, t) + v_\theta(\tilde{\mathbf{x}}_{t+h}, t+h)\right] $$

```cpp
#include <vector>
#include <functional>
#include <random>

using VelocityFn = std::function<std::vector<float>(
    const std::vector<float>& x, float t)>;

// ── Euler 采样器 ──────────────────────────────────────────────────────────
// velocity_fn: 模型预测的速度场 v_θ(x, t)
// x0: 初始噪声；steps: 积分步数（越多越准确）
std::vector<float> flowMatchingSampleEuler(
    VelocityFn velocity_fn,
    std::vector<float> x,      // 初始噪声 x_0
    int steps = 50)
{
    float dt = 1.0f / static_cast<float>(steps);  // 等步长，t 从 0 积分到 1

    for (int i = 0; i < steps; ++i) {
        float t = i * dt;
        auto v = velocity_fn(x, t);   // 查询当前速度
        for (size_t j = 0; j < x.size(); ++j)
            x[j] += dt * v[j];        // Euler 步进
    }
    return x;
}

// ── Heun 采样器（二阶，精度更高，每步需调用 2 次模型）────────────────────
std::vector<float> flowMatchingSampleHeun(
    VelocityFn velocity_fn,
    std::vector<float> x,
    int steps = 20)
{
    float dt = 1.0f / static_cast<float>(steps);

    for (int i = 0; i < steps; ++i) {
        float t  = i * dt;
        float t1 = t + dt;

        // Predictor：Euler 预测下一步
        auto v0 = velocity_fn(x, t);
        std::vector<float> x_pred(x.size());
        for (size_t j = 0; j < x.size(); ++j)
            x_pred[j] = x[j] + dt * v0[j];

        // Corrector：用预测值再算一次速度，取均值
        auto v1 = velocity_fn(x_pred, t1);
        for (size_t j = 0; j < x.size(); ++j)
            x[j] += 0.5f * dt * (v0[j] + v1[j]);   // 梯形公式
    }
    return x;
}
```

## 15. 支持 Broadcast 的 4D Tensor Elementwise Mul

### 15.1 Broadcast 规则

对齐后，各维度满足：相等，或其中一个为 1（自动扩展）。

```cpp
#include <vector>
#include <array>
#include <stdexcept>
#include <numeric>

using Shape4 = std::array<int, 4>;

// 计算广播后的输出 shape；不可广播时抛出异常
Shape4 broadcastShape(const Shape4& a, const Shape4& b) {
    Shape4 out;
    for (int i = 0; i < 4; ++i) {
        if (a[i] == b[i])         out[i] = a[i];
        else if (a[i] == 1)       out[i] = b[i];
        else if (b[i] == 1)       out[i] = a[i];
        else throw std::invalid_argument("Shapes not broadcastable");
    }
    return out;
}

// 将线性下标 idx（基于 out_shape）转换为在 src_shape 中的线性下标
// Broadcast 维度（src_shape[d]=1）在 src 中始终取 0
int broadcastIndex(int idx, const Shape4& out_shape, const Shape4& src_shape) {
    int src_idx = 0;
    int src_stride = 1;
    for (int d = 3; d >= 0; --d) {
        int pos     = idx % out_shape[d];    // 当前维度的坐标
        idx        /= out_shape[d];
        int src_pos = (src_shape[d] == 1) ? 0 : pos;  // Broadcast 维度强制取 0
        src_idx    += src_pos * src_stride;
        src_stride *= src_shape[d];
    }
    return src_idx;
}

// 4D Broadcast Elementwise 乘法
// 返回 out_shape 的 flat vector
std::vector<float> broadcastMul(const std::vector<float>& A, const Shape4& shapeA,
                                  const std::vector<float>& B, const Shape4& shapeB) {
    Shape4 out_shape = broadcastShape(shapeA, shapeB);
    int total = 1;
    for (int d : out_shape) total *= d;

    std::vector<float> out(total);
    for (int i = 0; i < total; ++i) {
        int ia = broadcastIndex(i, out_shape, shapeA);
        int ib = broadcastIndex(i, out_shape, shapeB);
        out[i] = A[ia] * B[ib];
    }
    return out;
}
```

## 16. Embedding Sparse Feature Pooling

### 16.1 问题定义

给定索引数组 $A$（长度 $M$）和嵌入向量 $B$（形状 $[M, d]$），计算：

$$ C[i] = \sum_{j:, A[j]=i} B[j], \quad i = 0, 1, \ldots, N-1 $$

即对每个 Embedding ID $i$，将所有属于它的行求和（稀疏 Reduce）。

```cpp
#include <vector>
#include <stdexcept>

// A: 索引数组，长度 M；B: 嵌入矩阵，[M, dim]；N: 目标嵌入数
// 返回 C: [N, dim]，C[i] = sum of B[j] for all j where A[j] == i
std::vector<std::vector<float>> embeddingPooling(
    const std::vector<int>& A,
    const std::vector<std::vector<float>>& B,
    int N)
{
    int M   = static_cast<int>(A.size());
    int dim = B.empty() ? 0 : static_cast<int>(B[0].size());

    // 输出矩阵，初始化为 0
    std::vector<std::vector<float>> C(N, std::vector<float>(dim, 0.0f));

    for (int j = 0; j < M; ++j) {
        int i = A[j];
        if (i < 0 || i >= N)
            throw std::out_of_range("Index out of range");
        // 累加第 j 行的嵌入向量到 C[i]
        for (int d = 0; d < dim; ++d)
            C[i][d] += B[j][d];
    }
    return C;
}

// 平坦化版本（嵌入矩阵以 flat vector 存储，行优先）
std::vector<float> embeddingPoolingFlat(
    const std::vector<int>& A,
    const std::vector<float>& B_flat,   // [M * dim]
    int M, int N, int dim)
{
    std::vector<float> C(N * dim, 0.0f);
    for (int j = 0; j < M; ++j) {
        int i = A[j];
        for (int d = 0; d < dim; ++d)
            C[i * dim + d] += B_flat[j * dim + d];
    }
    return C;
}
```

### 18.3 双重检查锁（DCLP，C++11 atomic 版，需跨编译器兼容时使用）

```cpp
#include <atomic>
#include <mutex>

class SingletonDCLP {
public:
    static SingletonDCLP* getInstance() {
        // 第一次检查（无锁，避免每次都加锁）
        SingletonDCLP* p = instance_.load(std::memory_order_acquire);
        if (!p) {
            std::lock_guard<std::mutex> lk(mu_);
            // 第二次检查（加锁后再次确认，防止多线程同时通过第一次检查）
            p = instance_.load(std::memory_order_relaxed);
            if (!p) {
                p = new SingletonDCLP();
                instance_.store(p, std::memory_order_release);  // release 保证初始化可见
            }
        }
        return p;
    }

    SingletonDCLP(const SingletonDCLP&)            = delete;
    SingletonDCLP& operator=(const SingletonDCLP&) = delete;

private:
    SingletonDCLP() = default;
    static std::atomic<SingletonDCLP*> instance_;
    static std::mutex                  mu_;
};

std::atomic<SingletonDCLP*> SingletonDCLP::instance_{nullptr};
std::mutex                  SingletonDCLP::mu_;
```

## 1. 向量加（Vector Add）

### 1.1 核心思路

最基础的 CUDA kernel：每个线程处理一个元素，无需线程间通信。 重点考察：Grid/Block 维度计算、越界保护、`cudaMalloc` / `cudaMemcpy` 流程。

```cuda
#include <cuda_runtime.h>
#include <cstdio>

// ── Kernel：每线程处理一个元素 ────────────────────────────────────────────
__global__ void vecAddKernel(const float* __restrict__ A,
                              const float* __restrict__ B,
                              float*       __restrict__ C,
                              int N) {
    // 全局线程 ID
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    // 越界保护：N 不一定是 blockDim.x 的整数倍
    if (idx < N) C[idx] = A[idx] + B[idx];
}

// ── Host 封装 ─────────────────────────────────────────────────────────────
void vecAdd(const float* h_A, const float* h_B, float* h_C, int N) {
    float *d_A, *d_B, *d_C;
    size_t bytes = N * sizeof(float);

    cudaMalloc(&d_A, bytes);
    cudaMalloc(&d_B, bytes);
    cudaMalloc(&d_C, bytes);

    cudaMemcpy(d_A, h_A, bytes, cudaMemcpyHostToDevice);
    cudaMemcpy(d_B, h_B, bytes, cudaMemcpyHostToDevice);

    // 每 Block 256 线程；Block 数向上取整
    int blockSize = 256;
    int gridSize  = (N + blockSize - 1) / blockSize;
    vecAddKernel<<<gridSize, blockSize>>>(d_A, d_B, d_C, N);

    cudaMemcpy(h_C, d_C, bytes, cudaMemcpyDeviceToHost);

    cudaFree(d_A); cudaFree(d_B); cudaFree(d_C);
}
```

## 2. Vector Dot Product

### 2.1 核心思路

点积 = 逐元素乘积之和，需要全局归约。 两阶段策略：

1. **Kernel 1**：每个 Block 内部做 shared memory 归约，将 `blockDim.x` 个元素压缩成 1 个分块和，写入 `partial[]`。
2. **Kernel 2**（或 atomicAdd）：将所有分块和再次归约。

```cuda
#include <cuda_runtime.h>

// ── Block 内 Shared Memory 归约（后续多题复用）────────────────────────────
// 将 smem[0..blockDim.x-1] 归约为 smem[0]
__device__ void blockReduce(volatile float* smem, int tid) {
    // 步长从 blockDim.x/2 开始折半
    for (int s = blockDim.x / 2; s > 32; s >>= 1) {
        if (tid < s) smem[tid] += smem[tid + s];
        __syncthreads();   // 保证本轮写入对所有线程可见
    }
    // s <= 32 时同一 Warp 内的线程天然同步（Warp-synchronous），
    // 但加 volatile 防止编译器优化掉写入
    if (tid < 32) {
        smem[tid] += smem[tid + 32];
        smem[tid] += smem[tid + 16];
        smem[tid] += smem[tid +  8];
        smem[tid] += smem[tid +  4];
        smem[tid] += smem[tid +  2];
        smem[tid] += smem[tid +  1];
    }
}

// ── Kernel：逐元素乘积 + Block 内归约 ────────────────────────────────────
__global__ void dotProductKernel(const float* __restrict__ A,
                                  const float* __restrict__ B,
                                  float*       __restrict__ partial,
                                  int N) {
    extern __shared__ float smem[];   // 动态 shared memory，大小 = blockDim.x * 4 字节
    int tid = threadIdx.x;
    int idx = blockIdx.x * blockDim.x + tid;

    // 每个线程加载并计算乘积；越界补 0
    smem[tid] = (idx < N) ? A[idx] * B[idx] : 0.0f;
    __syncthreads();

    blockReduce(smem, tid);

    // 只有线程 0 写出本 Block 的归约结果
    if (tid == 0) partial[blockIdx.x] = smem[0];
}

// ── Host 封装（两阶段归约）────────────────────────────────────────────────
float dotProduct(const float* d_A, const float* d_B, int N) {
    const int BLOCK = 256;
    int grid = (N + BLOCK - 1) / BLOCK;

    float *d_partial;
    cudaMalloc(&d_partial, grid * sizeof(float));

    // 第一阶段：每 Block 归约成一个分块和
    dotProductKernel<<<grid, BLOCK, BLOCK * sizeof(float)>>>(d_A, d_B, d_partial, N);

    // 第二阶段：将所有分块和在一个 Block 内再次归约
    // grid 通常 << BLOCK，可以直接用单 Block 处理
    dotProductKernel<<<1, BLOCK, BLOCK * sizeof(float)>>>(d_partial, d_partial, d_partial, grid);
    // 注意：此处 A==B==d_partial，第二阶段求的是 Σ partial[i]*partial[i]，
    // 需改为专用归约 kernel。正确做法：

    // ── 正确的单 Block 求和 kernel ───────────────────────────────────────
    // （见下方 reduceSum 章节的 singleBlockReduceKernel）

    float result;
    cudaMemcpy(&result, d_partial, sizeof(float), cudaMemcpyDeviceToHost);
    cudaFree(d_partial);
    return result;
}
```

## 3. Reduce Sum（Shared Memory 归约 + 分层 Kernel）

### 3.1 归约模式演进

|版本|问题|改进|
|---|---|---|
|V1 基础折半|Bank conflict（相邻线程访问相邻地址）|—|
|V2 交错寻址|Warp divergence（if tid%2==0 分支）|—|
|V3 连续寻址|消除 divergence|消除 Warp divergence|
|V4 首次加载时归约|一半线程空闲|Grid 减半，每线程加载 2 元素|
|V5 Warp Unroll|最后 32 线程仍有 `__syncthreads`|Warp 内 unroll|
|V6 完全 Unroll|模板 blockSize，编译期展开|最高性能|

```cuda
#include <cuda_runtime.h>

// ── V3：连续寻址，无 Bank Conflict，无 Warp Divergence ────────────────────
__global__ void reduceSumV3(const float* __restrict__ in,
                             float*       __restrict__ out,
                             int N) {
    extern __shared__ float smem[];
    int tid = threadIdx.x;
    int idx = blockIdx.x * blockDim.x + tid;

    smem[tid] = (idx < N) ? in[idx] : 0.0f;
    __syncthreads();

    // 从大步长到小步长折半
    for (int s = blockDim.x / 2; s > 0; s >>= 1) {
        if (tid < s)
            smem[tid] += smem[tid + s];
        __syncthreads();
    }

    if (tid == 0) out[blockIdx.x] = smem[0];
}

// ── V4：每线程加载 2 元素（Grid 减半，提高 SM 占用率）────────────────────
__global__ void reduceSumV4(const float* __restrict__ in,
                             float*       __restrict__ out,
                             int N) {
    extern __shared__ float smem[];
    int tid  = threadIdx.x;
    // 每 Block 处理 2*blockDim.x 个元素
    int idx  = blockIdx.x * (blockDim.x * 2) + tid;

    // 第一次归约发生在加载时：两个元素直接相加
    float val = 0.0f;
    if (idx     < N) val += in[idx];
    if (idx + blockDim.x < N) val += in[idx + blockDim.x];
    smem[tid] = val;
    __syncthreads();

    for (int s = blockDim.x / 2; s > 0; s >>= 1) {
        if (tid < s) smem[tid] += smem[tid + s];
        __syncthreads();
    }

    if (tid == 0) out[blockIdx.x] = smem[0];
}

// ── V5：Warp Unroll（最后 32 线程用 __shfl_down_sync 代替 smem）──────────
__device__ float warpReduceSum(float val) {
    // __shfl_down_sync：Warp 内寄存器直接交换，无需 smem，无需 __syncthreads
    for (int offset = 16; offset > 0; offset >>= 1)
        val += __shfl_down_sync(0xFFFFFFFF, val, offset);
    return val;   // 只有 lane 0 的结果有效
}

__global__ void reduceSumV5(const float* __restrict__ in,
                             float*       __restrict__ out,
                             int N) {
    extern __shared__ float smem[];
    int tid  = threadIdx.x;
    int idx  = blockIdx.x * (blockDim.x * 2) + tid;

    float val = 0.0f;
    if (idx              < N) val += in[idx];
    if (idx + blockDim.x < N) val += in[idx + blockDim.x];

    // smem 归约到 Warp 边界（32 个 Warp 的第一个元素）
    smem[tid] = val;
    __syncthreads();

    for (int s = blockDim.x / 2; s > 32; s >>= 1) {
        if (tid < s) smem[tid] += smem[tid + s];
        __syncthreads();
    }

    // 最后 Warp（tid < 32）用寄存器 shuffle 完成
    if (tid < 32) {
        val = smem[tid];
        val = warpReduceSum(val);
    }

    if (tid == 0) out[blockIdx.x] = val;
}

// ── 分层 Host 封装（递归调用直到结果为单个值）────────────────────────────
float reduceSum(const float* d_in, int N) {
    const int BLOCK = 256;
    int grid = (N + BLOCK * 2 - 1) / (BLOCK * 2);

    float *d_out;
    cudaMalloc(&d_out, grid * sizeof(float));

    reduceSumV4<<<grid, BLOCK, BLOCK * sizeof(float)>>>(d_in, d_out, N);

    float result;
    if (grid == 1) {
        // 只剩一个 Block 的结果，直接拷出
        cudaMemcpy(&result, d_out, sizeof(float), cudaMemcpyDeviceToHost);
    } else {
        // 递归对分块和再做一次归约
        result = reduceSum(d_out, grid);
    }

    cudaFree(d_out);
    return result;
}
```

## 4. Block Reduce（可复用原语）

### 4.1 核心思路

将 Block 内的值归约为单个值，供上层 kernel 直接调用。 使用 `__shfl_down_sync` 做 Warp 级归约，再用 shared memory 跨 Warp 合并。

```cuda
#include <cuda_runtime.h>

// ── Warp 级求和（32 线程 → 1 个结果，只有 lane 0 有效）────────────────────
__device__ __forceinline__ float warpReduceSum(float val) {
    #pragma unroll
    for (int offset = 16; offset > 0; offset >>= 1)
        val += __shfl_down_sync(0xFFFFFFFF, val, offset);
    return val;
}

// ── Block 级求和（最多 1024 线程 → 1 个结果，只有 tid==0 有效）───────────
// 要求：blockDim.x 为 32 的整数倍
__device__ float blockReduceSum(float val) {
    // 静态 shared memory：最多 32 个 Warp（blockDim.x 最大 1024）
    __shared__ float warp_sums[32];

    int lane    = threadIdx.x % 32;   // 当前线程在 Warp 内的编号
    int warp_id = threadIdx.x / 32;   // 当前 Warp 的编号

    // 第一步：每个 Warp 内部归约
    val = warpReduceSum(val);

    // 第二步：每个 Warp 的 lane 0 将结果写入 shared memory
    if (lane == 0) warp_sums[warp_id] = val;
    __syncthreads();

    // 第三步：第一个 Warp 读取所有 Warp 的部分和，再做一次 Warp 归约
    int num_warps = (blockDim.x + 31) / 32;
    val = (threadIdx.x < num_warps) ? warp_sums[threadIdx.x] : 0.0f;
    if (warp_id == 0) val = warpReduceSum(val);

    return val;   // 只有 tid == 0 的结果是 Block 内所有元素的总和
}

// ── 使用示例：每 Block 归约后 atomicAdd 到全局结果 ───────────────────────
__global__ void reduceSumAtomic(const float* __restrict__ in,
                                 float*       __restrict__ result,
                                 int N) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    float val = (idx < N) ? in[idx] : 0.0f;

    float block_sum = blockReduceSum(val);

    // 只有 tid==0 做 atomicAdd，避免多线程竞争
    if (threadIdx.x == 0)
        atomicAdd(result, block_sum);
}
```

## 5. 大矩阵 Reduce：$(10^6, 128) \to (1, 128)$

### 5.1 核心思路

矩阵形状 $[R, C] = [10^6, 128]$，对每一列做规约（Col-wise Reduce），输出 $[1, 128]$。

两种方案：

- **方案 A（列并行）**：每列分配一组 Block，各列独立归约。Grid = $(C, \lceil R / (2 \cdot \text{BLOCK}) \rceil)$。
- **方案 B（行并行，缓存友好）**：每行分配一个线程，原子累加到输出。读取连续，但 `atomicAdd` 有竞争。

方案 A 更适合大 $R$，下面实现方案 A。

```cuda
#include <cuda_runtime.h>

// ── Kernel：每个 Block 处理 (col_id, 一段行) 的归约 ──────────────────────
// gridDim.x = C（列数），gridDim.y = 分块数
// 每个 Block 沿列方向归约 blockDim.x * 2 行
__global__ void colReduceKernel(const float* __restrict__ in,
                                 float*       __restrict__ partial,
                                 int R, int C) {
    extern __shared__ float smem[];
    int col = blockIdx.x;                            // 当前处理的列
    int tid = threadIdx.x;
    // 每 Block 覆盖 2*blockDim.x 行，按 blockIdx.y 分段
    int row0 = blockIdx.y * (blockDim.x * 2) + tid;

    // 加载并首次相加（两行合并为一）
    float val = 0.0f;
    if (row0          < R) val += in[row0          * C + col];
    if (row0 + blockDim.x < R) val += in[(row0 + blockDim.x) * C + col];
    smem[tid] = val;
    __syncthreads();

    // Block 内折半归约
    for (int s = blockDim.x / 2; s > 0; s >>= 1) {
        if (tid < s) smem[tid] += smem[tid + s];
        __syncthreads();
    }

    // partial[gridDim.y * col + blockIdx.y]：第 col 列的第 blockIdx.y 个分块和
    if (tid == 0)
        partial[blockIdx.y * C + col] = smem[0];
}

// ── Host 封装 ─────────────────────────────────────────────────────────────
// d_in:  [R, C] row-major；d_out: [1, C]
void colReduceMatrix(const float* d_in, float* d_out, int R, int C) {
    const int BLOCK = 256;
    int gridY = (R + BLOCK * 2 - 1) / (BLOCK * 2);   // 行方向分块数

    float* d_partial;
    cudaMalloc(&d_partial, gridY * C * sizeof(float));

    // 第一阶段：gridX=C, gridY=行分块数
    dim3 grid(C, gridY), block(BLOCK);
    colReduceKernel<<<grid, block, BLOCK * sizeof(float)>>>(d_in, d_partial, R, C);

    if (gridY == 1) {
        // 直接就是结果
        cudaMemcpy(d_out, d_partial, C * sizeof(float), cudaMemcpyDeviceToDevice);
    } else {
        // 第二阶段：对 d_partial[gridY, C] 再次列归约
        colReduceMatrix(d_partial, d_out, gridY, C);
    }

    cudaFree(d_partial);
}
```

```cuda
#include <cuda_runtime.h>
#include <vector>
#include <numeric>

// 复用 reduceSumV4 + warpReduceSum（见第 3 节）

// ── 完整主机端流程 ─────────────────────────────────────────────────────────
float sumOneMillion(const std::vector<float>& h_data) {
```

## 7. 前缀和（Prefix Sum / Scan）

### 7.1 核心思路

**独占前缀和（Exclusive Scan）**：$\text{out}[i] = \sum_{j=0}^{i-1} \text{in}[j]$，$\text{out}[0] = 0$。

两阶段算法（适合任意大小）：

1. **Up-sweep（Reduce 阶段）**：构建部分和树。
2. **Down-sweep（Scan 阶段）**：分发前缀和。

Block 内用 Kogge-Stone 或 Blelloch 算法；跨 Block 用分阶段扫描。

```cuda
#include <cuda_runtime.h>

// ── Block 内 Blelloch 独占扫描（原地，blockDim.x 必须为 2 的幂）─────────
// 处理 smem[0..n-1]，n = blockDim.x * 2
__device__ void blellochScanBlock(volatile float* smem, int n) {
    int tid = threadIdx.x;

    // Up-sweep：构建部分和树
    for (int stride = 1; stride < n; stride <<= 1) {
        int idx = (tid + 1) * stride * 2 - 1;   // 右孩子下标
        if (idx < n) smem[idx] += smem[idx - stride];
        __syncthreads();
    }

    // 根节点置零（独占扫描）
    if (tid == 0) smem[n - 1] = 0.0f;
    __syncthreads();

    // Down-sweep：分发前缀
    for (int stride = n / 2; stride >= 1; stride >>= 1) {
        int idx = (tid + 1) * stride * 2 - 1;
        if (idx < n) {
            float tmp      = smem[idx - stride];
            smem[idx - stride] = smem[idx];        // 左孩子 ← 父节点
            smem[idx]          = smem[idx] + tmp;  // 右孩子 ← 父节点 + 左孩子
        }
        __syncthreads();
    }
}

// ── 单 Block 前缀和 Kernel（N ≤ 2 * blockDim.x）─────────────────────────
__global__ void prefixSumKernel(const float* __restrict__ in,
                                 float*       __restrict__ out,
                                 int N) {
    extern __shared__ float smem[];   // 大小 = 2 * blockDim.x
    int tid = threadIdx.x;
    int n   = blockDim.x * 2;        // 每 Block 处理元素数

    // 加载（越界补 0）
    smem[2 * tid]     = (2 * tid     < N) ? in[2 * tid]     : 0.0f;
    smem[2 * tid + 1] = (2 * tid + 1 < N) ? in[2 * tid + 1] : 0.0f;
    __syncthreads();

    blellochScanBlock(smem, n);

    // 写出结果
    if (2 * tid     < N) out[2 * tid]     = smem[2 * tid];
    if (2 * tid + 1 < N) out[2 * tid + 1] = smem[2 * tid + 1];
}

// ── 多 Block 前缀和（三阶段，适合任意 N）────────────────────────────────
// 阶段 1：每 Block 内部独立扫描，保存 Block 末尾和到 block_sums[]
// 阶段 2：对 block_sums[] 做前缀和（递归调用或单 Block 扫描）
// 阶段 3：将 block_sums[blockIdx.x] 加到 Block 内所有元素

__global__ void scanPhase1(const float* in, float* out,
                            float* block_sums, int N) {
    extern __shared__ float smem[];
    int tid  = threadIdx.x;
    int n    = blockDim.x * 2;
    int base = blockIdx.x * n;

    smem[2*tid]   = (base + 2*tid   < N) ? in[base + 2*tid]   : 0.0f;
    smem[2*tid+1] = (base + 2*tid+1 < N) ? in[base + 2*tid+1] : 0.0f;
    __syncthreads();

    blellochScanBlock(smem, n);   // Block 内扫描（独占）

    if (2*tid   < N - base) out[base + 2*tid]   = smem[2*tid];
    if (2*tid+1 < N - base) out[base + 2*tid+1] = smem[2*tid+1];

    // 记录本 Block 的原始总和（= 最后一个元素的值 + 最后一个输入）
    if (tid == 0)
        block_sums[blockIdx.x] = smem[n-1] +
            ((base + n - 1 < N) ? in[base + n - 1] : 0.0f);
}

__global__ void scanPhase3(float* out, const float* block_prefix, int N) {
    int base = blockIdx.x * blockDim.x * 2;
    float add = block_prefix[blockIdx.x];   // 本 Block 之前所有元素的总和
    int idx   = base + threadIdx.x;
    if (idx < N)     out[idx]     += add;
    if (idx + blockDim.x < N) out[idx + blockDim.x] += add;
}
```

## 8. 行 TOP-1

### 8.1 核心思路

对矩阵 $[R, C]$ 的每一行求最大值及其列下标。 每行分配一个 Block，Block 内做 argmax 规约。

```cuda
#include <cuda_runtime.h>
#include <cfloat>   // FLT_MIN

// ── Warp 级 argmax ────────────────────────────────────────────────────────
__device__ void warpArgmax(float& val, int& idx) {
    for (int offset = 16; offset > 0; offset >>= 1) {
        float other_val = __shfl_down_sync(0xFFFFFFFF, val, offset);
        int   other_idx = __shfl_down_sync(0xFFFFFFFF, idx, offset);
        if (other_val > val) { val = other_val; idx = other_idx; }
    }
}

// ── Kernel：每个 Block 处理一行，输出该行的 max 值和 argmax 列下标 ────────
__global__ void rowTop1Kernel(const float* __restrict__ in,
                               float* __restrict__ out_val,
                               int*   __restrict__ out_idx,
                               int R, int C) {
    // 每行一个 Block
    int row = blockIdx.x;
    if (row >= R) return;

    int tid = threadIdx.x;
    const float* row_ptr = in + row * C;

    // 每线程遍历自己负责的列（步长 = blockDim.x）
    float best_val = -FLT_MAX;
    int   best_idx = -1;
    for (int col = tid; col < C; col += blockDim.x) {
        float v = row_ptr[col];
        if (v > best_val) { best_val = v; best_idx = col; }
    }

    // Shared memory 存储每线程的局部最优
    __shared__ float s_val[256];
    __shared__ int   s_idx[256];
    s_val[tid] = best_val;
    s_idx[tid] = best_idx;
    __syncthreads();

    // Block 内折半 argmax 归约
    for (int s = blockDim.x / 2; s > 32; s >>= 1) {
        if (tid < s && s_val[tid + s] > s_val[tid]) {
            s_val[tid] = s_val[tid + s];
            s_idx[tid] = s_idx[tid + s];
        }
        __syncthreads();
    }

    // 最后 Warp 用 shuffle 完成
    if (tid < 32) {
        float v = s_val[tid];
        int   i = s_idx[tid];
        warpArgmax(v, i);
        if (tid == 0) {
            out_val[row] = v;
            out_idx[row] = i;
        }
    }
}

// ── Host 封装 ─────────────────────────────────────────────────────────────
// d_in: [R, C]；d_out_val/d_out_idx: [R]
void rowTop1(const float* d_in, float* d_out_val, int* d_out_idx, int R, int C) {
    // Block 大小取 min(C 向上取到 32 倍数, 256)
    int block = std::min(256, ((C + 31) / 32) * 32);
    rowTop1Kernel<<<R, block>>>(d_in, d_out_val, d_out_idx, R, C);
}
```

## 1. GEMM（通用矩阵乘法）

### 1.1 核心思路

$C = A \cdot B$，其中 $A \in \mathbb{R}^{M \times K}$，$B \in \mathbb{R}^{K \times N}$，$C \in \mathbb{R}^{M \times N}$。

朴素实现：每个线程计算 $C$ 的一个元素，需 $K$ 次乘加，全局内存访问次数 $O(MNK)$，带宽瓶颈严重。

**Tiled GEMM（Shared Memory 分块）**：将 $A$、$B$ 按 `TILE_SIZE × TILE_SIZE` 分块，协作加载到 Shared Memory，减少全局内存访问。

理想情况下，全局内存流量从 $2MNK$ 降至 $2MNK / \text{TILE_SIZE}$（Arithmetic Intensity 提升 `TILE_SIZE` 倍）。

```cuda
#include <cuda_runtime.h>

#define TILE_SIZE 16   // 每个 Tile 的边长（16×16 = 256 线程，恰好一个 Block）

// ── V1：朴素实现（无 Shared Memory，每线程计算 C[row][col]）────────────────
__global__ void gemmNaive(const float* __restrict__ A,
                           const float* __restrict__ B,
                           float*       __restrict__ C,
                           int M, int N, int K) {
    int row = blockIdx.y * blockDim.y + threadIdx.y;
    int col = blockIdx.x * blockDim.x + threadIdx.x;
    if (row >= M || col >= N) return;

    float acc = 0.0f;
    for (int k = 0; k < K; ++k)
        acc += A[row * K + k] * B[k * N + col];
    C[row * N + col] = acc;
}

// ── V2：Tiled GEMM（Shared Memory 分块，减少全局内存访问）──────────────────
__global__ void gemmTiled(const float* __restrict__ A,
                           const float* __restrict__ B,
                           float*       __restrict__ C,
                           int M, int N, int K) {
    // 每个 Block 负责计算 C 的一个 TILE_SIZE × TILE_SIZE 子块
    __shared__ float tileA[TILE_SIZE][TILE_SIZE];  // A 的分块缓存
    __shared__ float tileB[TILE_SIZE][TILE_SIZE];  // B 的分块缓存

    int row = blockIdx.y * TILE_SIZE + threadIdx.y;  // C 中的全局行
    int col = blockIdx.x * TILE_SIZE + threadIdx.x;  // C 中的全局列

    float acc = 0.0f;

    // 沿 K 维度分块迭代
    int numTiles = (K + TILE_SIZE - 1) / TILE_SIZE;
    for (int t = 0; t < numTiles; ++t) {
        // 协作加载：每个线程负责加载 tileA 和 tileB 各一个元素
        int a_col = t * TILE_SIZE + threadIdx.x;   // A[row][a_col]
        int b_row = t * TILE_SIZE + threadIdx.y;   // B[b_row][col]

        tileA[threadIdx.y][threadIdx.x] = (row < M && a_col < K)
            ? A[row * K + a_col] : 0.0f;
        tileB[threadIdx.y][threadIdx.x] = (b_row < K && col < N)
            ? B[b_row * N + col] : 0.0f;
        __syncthreads();   // 确保 Tile 加载完成再计算

        // 在 Tile 范围内计算点积
        #pragma unroll
        for (int k = 0; k < TILE_SIZE; ++k)
            acc += tileA[threadIdx.y][k] * tileB[k][threadIdx.x];

        __syncthreads();   // 确保本 Tile 计算完成再加载下一个 Tile
    }

    if (row < M && col < N)
        C[row * N + col] = acc;
}

// ── V3：每线程计算 2×2 输出（减少 Shared Memory 加载，提高计算强度）────────
// 每个 Block 处理 TILE_SIZE×2 × TILE_SIZE×2 的 C 子块
// 每个线程处理 2×2 = 4 个 C 元素，寄存器复用更充分
#define TM 2   // 每线程 y 方向输出数
#define TN 2   // 每线程 x 方向输出数

__global__ void gemmTiled2x2(const float* __restrict__ A,
                               const float* __restrict__ B,
                               float*       __restrict__ C,
                               int M, int N, int K) {
    __shared__ float tileA[TILE_SIZE][TILE_SIZE];
    __shared__ float tileB[TILE_SIZE][TILE_SIZE];

    // 每线程负责 TM×TN 个 C 元素
    int thread_row = blockIdx.y * (TILE_SIZE * TM) + threadIdx.y * TM;
    int thread_col = blockIdx.x * (TILE_SIZE * TN) + threadIdx.x * TN;

    float acc[TM][TN] = {};   // 累加寄存器，初始化为 0

    int numTiles = (K + TILE_SIZE - 1) / TILE_SIZE;
    for (int t = 0; t < numTiles; ++t) {
        // 协作加载（每线程加载多行以覆盖 2× 的行数）
        for (int m = 0; m < TM; ++m) {
            int row   = thread_row + m;
            int a_col = t * TILE_SIZE + threadIdx.x;
            tileA[threadIdx.y * TM + m][threadIdx.x] =
                (row < M && a_col < K) ? A[row * K + a_col] : 0.0f;
        }
        for (int n = 0; n < TN; ++n) {
            int b_row = t * TILE_SIZE + threadIdx.y;
            int col   = thread_col + n;
            tileB[threadIdx.y][threadIdx.x * TN + n] =
                (b_row < K && col < N) ? B[b_row * N + col] : 0.0f;
        }
        __syncthreads();

        #pragma unroll
        for (int k = 0; k < TILE_SIZE; ++k)
            for (int m = 0; m < TM; ++m)
                for (int n = 0; n < TN; ++n)
                    acc[m][n] += tileA[threadIdx.y * TM + m][k]
                               * tileB[k][threadIdx.x * TN + n];
        __syncthreads();
    }

    for (int m = 0; m < TM; ++m)
        for (int n = 0; n < TN; ++n)
            if (thread_row + m < M && thread_col + n < N)
                C[(thread_row + m) * N + (thread_col + n)] = acc[m][n];
}

// ── Host 封装 ─────────────────────────────────────────────────────────────
void gemm(const float* d_A, const float* d_B, float* d_C,
          int M, int N, int K) {
    dim3 block(TILE_SIZE, TILE_SIZE);
    dim3 grid((N + TILE_SIZE - 1) / TILE_SIZE,
              (M + TILE_SIZE - 1) / TILE_SIZE);
    gemmTiled<<<grid, block>>>(d_A, d_B, d_C, M, N, K);
}
```

### 1.2 性能分析

|版本|全局内存读次数（每元素）|关键瓶颈|
|---|---|---|
|朴素|$2K$（A 行 + B 列各 $K$ 次）|Memory Bound|
|Tiled|$2K / \text{TILE_SIZE}$|计算/内存均衡|
|2×2 Thread Tile|同 Tiled 但寄存器利用率更高|Compute Bound（接近）|

## 2. 二维矩阵转置（逐步优化）

### 2.1 问题定义

$B = A^\top$，$A \in \mathbb{R}^{R \times C}$，$B \in \mathbb{R}^{C \times R}$。 读 $A$ 是行优先（coalesced），写 $B$ 是列跳跃（non-coalesced）；或反之。 目标：使读写均 coalesced。

```cuda
#include <cuda_runtime.h>

#define TILE 32

// ── V1：朴素转置（写 B 时非 Coalesced）──────────────────────────────────
// 读 A：线程 (row, col) 读 A[row][col]，同一 Warp 内 col 连续 → Coalesced ✓
// 写 B：写 B[col][row]，同一 Warp 内 col 相同而 row 连续 → Stride=C → Non-Coalesced ✗
__global__ void transposeNaive(const float* __restrict__ A,
                                float*       __restrict__ B,
                                int R, int C) {
    int row = blockIdx.y * blockDim.y + threadIdx.y;
    int col = blockIdx.x * blockDim.x + threadIdx.x;
    if (row < R && col < C)
        B[col * R + row] = A[row * C + col];
}

// ── V2：Shared Memory 转置（读写均 Coalesced）────────────────────────────
// 策略：
//   1. 协作读 A 的一个 TILE×TILE 块到 smem（行连续读，Coalesced）
//   2. 从 smem 中转置读取后写到 B（列连续写，Coalesced）
__global__ void transposeShared(const float* __restrict__ A,
                                 float*       __restrict__ B,
                                 int R, int C) {
    // +1 消除 Bank Conflict：
    // 转置后同一列元素映射到同一 Bank，+1 使步长为质数，打散 Bank 访问
    __shared__ float smem[TILE][TILE + 1];

    // 读阶段：按 Block 坐标从 A 加载（行优先，Coalesced）
    int read_row = blockIdx.y * TILE + threadIdx.y;
    int read_col = blockIdx.x * TILE + threadIdx.x;
    if (read_row < R && read_col < C)
        smem[threadIdx.y][threadIdx.x] = A[read_row * C + read_col];
    __syncthreads();

    // 写阶段：Block 坐标互换（x↔y），从 smem 中转置读出后按列写 B
    // B 的目标 Tile：行起点 = blockIdx.x * TILE，列起点 = blockIdx.y * TILE
    int write_row = blockIdx.x * TILE + threadIdx.y;   // 注意：x↔y 互换
    int write_col = blockIdx.y * TILE + threadIdx.x;
    if (write_row < C && write_col < R)
        // 从 smem 转置读：smem[threadIdx.x][threadIdx.y]（x,y 互换）
        B[write_row * R + write_col] = smem[threadIdx.x][threadIdx.y];
}

// ── V3：向量化加载（float4，提高内存带宽利用率）──────────────────────────
// 每次加载 4 个 float（128 bit），需要 C 为 4 的倍数
__global__ void transposeVec4(const float4* __restrict__ A,
                               float*        __restrict__ B,
                               int R, int C4) {
    // C4 = C / 4，A 按 float4 访问，每线程加载 1 个 float4（4 个 float）
    __shared__ float smem[TILE][TILE * 4 + 4];   // 每行 TILE*4 个 float，+4 避 Bank Conflict

    int row = blockIdx.y * TILE + threadIdx.y;
    int col = blockIdx.x * TILE + threadIdx.x;   // col 是 float4 下标

    if (row < R && col < C4) {
        float4 v = A[row * C4 + col];
        // 将 4 个 float 展开存入 smem
        int base = threadIdx.x * 4;
        smem[threadIdx.y][base + 0] = v.x;
        smem[threadIdx.y][base + 1] = v.y;
        smem[threadIdx.y][base + 2] = v.z;
        smem[threadIdx.y][base + 3] = v.w;
    }
    __syncthreads();

    // 转置写出（写 B，行连续，Coalesced）
    int C = C4 * 4;
    int write_row = blockIdx.x * (TILE * 4) + threadIdx.y * 4;
    int write_col = blockIdx.y * TILE + threadIdx.x;
    if (write_col < R) {
        for (int k = 0; k < 4; ++k)
            if (write_row + k < C)
                B[(write_row + k) * R + write_col] = smem[threadIdx.x][threadIdx.y * 4 + k];
    }
}

// ── Host 封装（V2 推荐版）─────────────────────────────────────────────────
void transpose(const float* d_A, float* d_B, int R, int C) {
    dim3 block(TILE, TILE);
    dim3 grid((C + TILE - 1) / TILE, (R + TILE - 1) / TILE);
    transposeShared<<<grid, block>>>(d_A, d_B, R, C);
}
```

### 2.2 各版本性能对比

|版本|读 A|写 B|Bank Conflict|带宽利用率|
|---|---|---|---|---|
|V1 朴素|Coalesced ✓|Non-Coalesced ✗|无|~25%|
|V2 Shared Memory|Coalesced ✓|Coalesced ✓|有（未 padding）|~85%|
|V2 + padding|Coalesced ✓|Coalesced ✓|无 ✓|~95%|
|V3 Vec4|Coalesced ✓|Coalesced ✓|无 ✓|~峰值|

### 2.3 Bank Conflict 根因分析

Shared Memory 有 32 个 Bank，每个 Bank 宽度 4 字节。 转置时，Warp 内 32 线程读 `smem[threadIdx.x][threadIdx.y]`：

- `threadIdx.x` 在 $[0,31]$ 变化，`threadIdx.y` 相同
- 访问地址步长 = $(\text{TILE}+0) \times 4 = 128$ 字节 = 32 个 Bank 的整数倍 → 全部映射到同一 Bank → **32-way Bank Conflict**
- 加 `+1` padding 后步长 = $33 \times 4 = 132$ 字节，不再是 32 的整数倍 → 打散到不同 Bank → 消除

## 3. Concat（四维张量按指定轴拼接）

### 3.1 问题定义

将 $N$ 个四维张量 $T_0, T_1, \ldots, T_{N-1}$（形状各为 $[B, C_i, H, W]$ 或其他轴）沿 `axis` 拼接，输出张量形状在该轴上为 $\sum_i \text{dim}_i$。

以最常见的 **Channel 维度（axis=1）** 拼接为例： 输入 $A[B, C_A, H, W]$，$B_\text{tensor}[B, C_B, H, W]$，输出 $C_\text{out}[B, C_A+C_B, H, W]$。

```cuda
#include <cuda_runtime.h>

// ── 通用四维 Concat（沿任意轴）──────────────────────────────────────────
// 输入：多个 tensor（flat，row-major），形状均为 [d0, d1, d2, d3]，
//        仅 axis 维度可不同
// 输出：沿 axis 维度拼接的结果

// 辅助：给定输出的线性下标 out_idx，以及输出 shape 和各输入的 axis_offset，
//        找到对应的输入 tensor 编号及其内部线性下标
__device__ void resolveIndex(
    int out_idx,
    const int* out_shape,        // [4]
    const int* src_axis_sizes,   // 每个 src 在 axis 维度的大小
    int num_srcs,
    int axis,
    int* src_id_out,
    int* src_linear_out)
{
    // 将 out_idx 转为四维坐标
    int coords[4];
    int tmp = out_idx;
    for (int d = 3; d >= 0; --d) {
        coords[d]  = tmp % out_shape[d];
        tmp       /= out_shape[d];
    }

    // 找到 axis 坐标落在哪个 src 的范围内
    int axis_coord = coords[axis];
    int offset     = 0;
    int src_id     = 0;
    for (int i = 0; i < num_srcs; ++i) {
        if (axis_coord < offset + src_axis_sizes[i]) {
            src_id = i;
            break;
        }
        offset += src_axis_sizes[i];
    }
    *src_id_out = src_id;

    // 修正 axis 坐标为 src 内部坐标
    coords[axis] -= offset;

    // 计算 src 内的线性下标
    // src shape 与 out_shape 相同，只有 axis 维度不同
    int src_shape_axis = src_axis_sizes[src_id];
    int linear = 0;
    int stride = 1;
    for (int d = 3; d >= 0; --d) {
        int dim = (d == axis) ? src_shape_axis : out_shape[d];
        linear += coords[d] * stride;
        stride *= dim;
    }
    *src_linear_out = linear;
}

// ── 两输入 Channel-wise Concat（最常见场景，特化版）─────────────────────
// A: [B, CA, H, W]，B_in: [B, CB, H, W] → out: [B, CA+CB, H, W]
__global__ void concat2ChannelKernel(
    const float* __restrict__ A,
    const float* __restrict__ B_in,
    float*       __restrict__ out,
    int batch, int CA, int CB, int H, int W)
{
    int C_out = CA + CB;
    int HW    = H * W;
    int total = batch * C_out * HW;
    int idx   = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= total) return;

    // 将线性下标解码为 [b, c, h, w]
    int w = idx % W;
    int h = (idx / W)     % H;
    int c = (idx / HW)    % C_out;
    int b = (idx / (HW * C_out));

    if (c < CA)
        out[idx] = A   [b * CA * HW + c       * HW + h * W + w];
    else
        out[idx] = B_in[b * CB * HW + (c - CA)* HW + h * W + w];
}

// ── 通用多输入 Concat Kernel（最多 8 个输入）─────────────────────────────
// 使用设备端指针数组，避免多次 kernel 调用
__global__ void concatGeneralKernel(
    const float* const* srcs,          // 设备端指针数组（每个元素是一个 src 的数据指针）
    const int*          src_axis_sizes, // 设备端：每个 src 在 axis 维度的大小
    int                 num_srcs,
    float*              out,
    const int*          out_shape,      // 设备端 [4]
    int                 axis,
    int                 total)          // 输出元素总数
{
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= total) return;

    int src_id, src_linear;
    resolveIndex(idx, out_shape, src_axis_sizes, num_srcs, axis, &src_id, &src_linear);
    out[idx] = srcs[src_id][src_linear];
}

// ── Host 封装（两输入 Channel Concat）────────────────────────────────────
void concatChannel(const float* d_A, const float* d_B,
                   float* d_out,
                   int batch, int CA, int CB, int H, int W) {
    int total = batch * (CA + CB) * H * W;
    int block = 256;
    int grid  = (total + block - 1) / block;
    concat2ChannelKernel<<<grid, block>>>(d_A, d_B, d_out, batch, CA, CB, H, W);
}
```

### 3.2 性能注意事项

| 场景                     | 建议                                     |
| ---------------------- | -------------------------------------- |
| 两输入 Channel Concat     | 使用特化 kernel，无需解码通用坐标，分支更少              |
| 多输入通用 Concat           | 设备端指针数组，单次 kernel 调用                   |
| 输入已在显存连续且目标 layout 也连续 | 用 `cudaMemcpyAsync` 多次异步拷贝代替 kernel，更快 |
| Concat 后接 Conv         | 考虑 Kernel Fusion，避免中间 tensor 写回全局内存    |

## 1. RMSNorm

### 1.1 公式

$$ \text{RMS}(\mathbf{x}) = \sqrt{\frac{1}{d}\sum_{i=1}^{d} x_i^2 + \varepsilon}, \qquad y_i = \frac{x_i}{\text{RMS}(\mathbf{x})} \cdot \gamma_i $$

无均值中心化，LLaMA / Mistral / Qwen 均采用。

### 1.2 实现

每行（每个 token）分配一个 Block，Block 内做平方和归约得到 RMS，再做归一化写回。

```cuda
#include <cuda_runtime.h>

// ── Warp 级求和 ───────────────────────────────────────────────────────────
__device__ __forceinline__ float warpReduceSum(float val) {
    #pragma unroll
    for (int offset = 16; offset > 0; offset >>= 1)
        val += __shfl_down_sync(0xFFFFFFFF, val, offset);
    return val;
}

// ── Block 级求和 ──────────────────────────────────────────────────────────
__device__ float blockReduceSum(float val) {
    __shared__ float warp_sums[32];
    int lane    = threadIdx.x & 31;
    int warp_id = threadIdx.x >> 5;
    val = warpReduceSum(val);
    if (lane == 0) warp_sums[warp_id] = val;
    __syncthreads();
    int num_warps = (blockDim.x + 31) >> 5;
    val = (threadIdx.x < num_warps) ? warp_sums[threadIdx.x] : 0.0f;
    if (warp_id == 0) val = warpReduceSum(val);
    return val;
}

// ── RMSNorm Kernel ────────────────────────────────────────────────────────
// in/out: [num_tokens, hidden_dim]；gamma: [hidden_dim]
__global__ void rmsNormKernel(const float* __restrict__ in,
                               const float* __restrict__ gamma,
                               float*       __restrict__ out,
                               int hidden_dim, float eps) {
    int token_id = blockIdx.x;
    const float* x = in  + token_id * hidden_dim;
    float*       y = out + token_id * hidden_dim;

    // 1. 每线程步长遍历，累加平方和
    float sum_sq = 0.0f;
    for (int i = threadIdx.x; i < hidden_dim; i += blockDim.x)
        sum_sq += x[i] * x[i];

    // 2. Block 内归约得到总平方和
    sum_sq = blockReduceSum(sum_sq);

    // 3. 共享 inv_rms 给所有线程
    __shared__ float s_inv_rms;
    if (threadIdx.x == 0)
        s_inv_rms = rsqrtf(sum_sq / static_cast<float>(hidden_dim) + eps);
    __syncthreads();

    // 4. 归一化 + 仿射
    float inv_rms = s_inv_rms;
    for (int i = threadIdx.x; i < hidden_dim; i += blockDim.x)
        y[i] = x[i] * inv_rms * gamma[i];
}
```

## 2. Softmax

### 2.1 实现策略

每行分配一个 Block，三步：求 max → 求 sum_exp → 归一化。

```cuda
#include <cuda_runtime.h>
#include <cfloat>

// ── Warp / Block 级 max 归约 ──────────────────────────────────────────────
__device__ __forceinline__ float warpReduceMax(float val) {
    #pragma unroll
    for (int offset = 16; offset > 0; offset >>= 1)
        val = fmaxf(val, __shfl_down_sync(0xFFFFFFFF, val, offset));
    return val;
}

__device__ float blockReduceMax(float val) {
    __shared__ float warp_maxs[32];
    int lane = threadIdx.x & 31, wid = threadIdx.x >> 5;
    val = warpReduceMax(val);
    if (lane == 0) warp_maxs[wid] = val;
    __syncthreads();
    int nw = (blockDim.x + 31) >> 5;
    val = (threadIdx.x < nw) ? warp_maxs[threadIdx.x] : -FLT_MAX;
    if (wid == 0) val = warpReduceMax(val);
    return val;
}

// ── Softmax Kernel（三 Pass，数值稳定）───────────────────────────────────
__global__ void softmaxKernel(const float* __restrict__ in,
                               float*       __restrict__ out,
                               int row_len) {
    int row        = blockIdx.x;
    const float* x = in  + row * row_len;
    float*       y = out + row * row_len;

    // Pass 1：求 max
    float max_val = -FLT_MAX;
    for (int i = threadIdx.x; i < row_len; i += blockDim.x)
        max_val = fmaxf(max_val, x[i]);
    max_val = blockReduceMax(max_val);
    __shared__ float s_max;
    if (threadIdx.x == 0) s_max = max_val;
    __syncthreads();
    max_val = s_max;

    // Pass 2：计算 exp(x - max) 并写入 out（临时），同时求 sum
    float sum_exp = 0.0f;
    for (int i = threadIdx.x; i < row_len; i += blockDim.x) {
        float e = expf(x[i] - max_val);
        y[i]    = e;   // 临时存 exp 值
        sum_exp += e;
    }
    sum_exp = blockReduceSum(sum_exp);
    __shared__ float s_sum;
    if (threadIdx.x == 0) s_sum = sum_exp;
    __syncthreads();
    float inv_sum = 1.0f / s_sum;

    // Pass 3：归一化
    for (int i = threadIdx.x; i < row_len; i += blockDim.x)
        y[i] *= inv_sum;
}
```

## 3. Online Softmax（单 Pass）

### 3.1 (m, d) 对的跨 Warp 合并

```cuda
#include <cuda_runtime.h>
#include <cfloat>

// ── Warp 级 (m, d) 合并 ───────────────────────────────────────────────────
__device__ void warpReduceMD(float& m, float& d) {
    #pragma unroll
    for (int offset = 16; offset > 0; offset >>= 1) {
        float om = __shfl_down_sync(0xFFFFFFFF, m, offset);
        float od = __shfl_down_sync(0xFFFFFFFF, d, offset);
        if (om > m) { d = d * expf(m - om) + od; m = om; }
        else        { d = d + od * expf(om - m); }
    }
}

// ── Online Softmax Kernel ─────────────────────────────────────────────────
__global__ void onlineSoftmaxKernel(const float* __restrict__ in,
                                     float*       __restrict__ out,
                                     int row_len) {
    int row    = blockIdx.x;
    const float* x = in  + row * row_len;
    float*       y = out + row * row_len;

    // Pass 1：每线程单次遍历，维护局部 (m, d)
    float m = -FLT_MAX, d = 0.0f;
    for (int i = threadIdx.x; i < row_len; i += blockDim.x) {
        float xi    = x[i];
        float new_m = fmaxf(m, xi);
        d = d * expf(m - new_m) + expf(xi - new_m);
        m = new_m;
    }
    // Warp 内合并
    warpReduceMD(m, d);

    // 跨 Warp 合并
    __shared__ float s_m[32], s_d[32];
    int lane = threadIdx.x & 31, wid = threadIdx.x >> 5;
    if (lane == 0) { s_m[wid] = m; s_d[wid] = d; }
    __syncthreads();
    int nw = (blockDim.x + 31) >> 5;
    if (wid == 0) {
        m = threadIdx.x < nw ? s_m[threadIdx.x] : -FLT_MAX;
        d = threadIdx.x < nw ? s_d[threadIdx.x] : 0.0f;
        warpReduceMD(m, d);
        if (lane == 0) { s_m[0] = m; s_d[0] = d; }
    }
    __syncthreads();

    float gm = s_m[0], ginv = 1.0f / s_d[0];

    // Pass 2：输出归一化结果
    for (int i = threadIdx.x; i < row_len; i += blockDim.x)
        y[i] = expf(x[i] - gm) * ginv;
}
```

## 4. Layer Norm

### 4.1 公式

$$ y_i = \frac{x_i - \mu}{\sqrt{\sigma^2 + \varepsilon}} \cdot \gamma_i + \beta_i, \quad \mu = \frac{1}{d}\sum x_i,\quad \sigma^2 = \frac{1}{d}\sum(x_i - \mu)^2 $$

### 4.2 两 Pass 实现 + Welford 单 Pass 实现

```cuda
// ── LayerNorm 两 Pass 版 ──────────────────────────────────────────────────
__global__ void layerNormKernel(const float* __restrict__ in,
                                 const float* __restrict__ gamma,
                                 const float* __restrict__ beta,
                                 float*       __restrict__ out,
                                 int hidden_dim, float eps) {
    int token_id   = blockIdx.x;
    const float* x = in  + token_id * hidden_dim;
    float*       y = out + token_id * hidden_dim;
    float inv_d    = 1.0f / static_cast<float>(hidden_dim);

    // Pass 1：均值
    float sum = 0.0f;
    for (int i = threadIdx.x; i < hidden_dim; i += blockDim.x) sum += x[i];
    sum = blockReduceSum(sum);
    __shared__ float s_mean, s_inv_std;
    if (threadIdx.x == 0) s_mean = sum * inv_d;
    __syncthreads();
    float mean = s_mean;

    // Pass 2：方差
    float var = 0.0f;
    for (int i = threadIdx.x; i < hidden_dim; i += blockDim.x) {
        float diff = x[i] - mean;
        var += diff * diff;
    }
    var = blockReduceSum(var);
    if (threadIdx.x == 0) s_inv_std = rsqrtf(var * inv_d + eps);
    __syncthreads();
    float inv_std = s_inv_std;

    // Pass 3：归一化 + 仿射
    for (int i = threadIdx.x; i < hidden_dim; i += blockDim.x)
        y[i] = (x[i] - mean) * inv_std * gamma[i] + beta[i];
}

// ── Welford 在线单 Pass 版（数值更稳定）──────────────────────────────────
// Welford 更新：delta = x_n - mean; mean += delta/n; M2 += delta*(x_n - mean)
// 跨线程合并（Chan 并行算法）：
//   n_new  = n_a + n_b
//   mean_new = (n_a*mean_a + n_b*mean_b) / n_new
//   M2_new = M2_a + M2_b + delta^2 * n_a*n_b / n_new
__global__ void layerNormWelford(const float* __restrict__ in,
                                  const float* __restrict__ gamma,
                                  const float* __restrict__ beta,
                                  float*       __restrict__ out,
                                  int hidden_dim, float eps) {
    int token_id   = blockIdx.x;
    const float* x = in  + token_id * hidden_dim;
    float*       y = out + token_id * hidden_dim;

    // 局部 Welford 状态
    float mean = 0.0f, M2 = 0.0f;
    int   cnt  = 0;
    for (int i = threadIdx.x; i < hidden_dim; i += blockDim.x) {
        ++cnt;
        float delta = x[i] - mean;
        mean += delta / cnt;
        M2   += delta * (x[i] - mean);
    }

    // 存入 shared memory，进行 Block 内合并
    __shared__ float s_mean[256], s_M2[256];
    __shared__ int   s_cnt[256];
    s_mean[threadIdx.x] = mean;
    s_M2  [threadIdx.x] = M2;
    s_cnt [threadIdx.x] = cnt;
    __syncthreads();

    for (int s = blockDim.x / 2; s > 0; s >>= 1) {
        if (threadIdx.x < s) {
            int   nb = s_cnt [threadIdx.x + s];
            float mb = s_mean[threadIdx.x + s];
            float M2b= s_M2  [threadIdx.x + s];
            int   na = s_cnt [threadIdx.x];
            float ma = s_mean[threadIdx.x];
            int n_new   = na + nb;
            float delta2= mb - ma;
            s_mean[threadIdx.x] = (na * ma + nb * mb) / n_new;
            s_M2  [threadIdx.x] = s_M2[threadIdx.x] + M2b
                                 + delta2 * delta2 * (float)na * nb / n_new;
            s_cnt [threadIdx.x] = n_new;
        }
        __syncthreads();
    }

    float global_mean    = s_mean[0];
    float global_inv_std = rsqrtf(s_M2[0] / hidden_dim + eps);

    for (int i = threadIdx.x; i < hidden_dim; i += blockDim.x)
        y[i] = (x[i] - global_mean) * global_inv_std * gamma[i] + beta[i];
}
```

## 5. MHA（Multi-Head Attention）

### 5.1 公式

$$ \text{head}_h = \text{Softmax}!\left(\frac{Q_h K_h^\top}{\sqrt{d_k}}\right) V_h, \quad d_k = d_{\text{model}} / H $$

### 5.2 实现（每 Block 处理一行 Query，Online Softmax 合并）

```cuda
#include <cuda_runtime.h>
#include <cfloat>

// gridDim = batch * num_heads * seq_q；每 Block 处理 (batch, head, query_pos)
__global__ void mhaKernel(
    const float* __restrict__ Q,   // [batch, H, seq_q,  dk]
    const float* __restrict__ K,   // [batch, H, seq_kv, dk]
    const float* __restrict__ V,   // [batch, H, seq_kv, dk]
    float*       __restrict__ O,   // [batch, H, seq_q,  dk]
    int batch, int num_heads, int seq_q, int seq_kv, int head_dim,
    bool causal)
{
    int gid       = blockIdx.x;
    int query_pos = gid % seq_q;
    int head_id   = (gid / seq_q) % num_heads;
    int batch_id  = gid / (seq_q * num_heads);

    int hb = batch_id * num_heads + head_id;
    const float* q      = Q + hb * seq_q  * head_dim + query_pos * head_dim;
    const float* k_base = K + hb * seq_kv * head_dim;
    const float* v_base = V + hb * seq_kv * head_dim;
    float*       o      = O + hb * seq_q  * head_dim + query_pos * head_dim;
    float scale = rsqrtf((float)head_dim);

    // scores 临时存在 shared memory（朴素 MHA，seq_kv 不能太大）
    extern __shared__ float smem[];   // 大小 = seq_kv * 4 字节
    float* scores = smem;

    // 阶段 1：计算 Q·K^T / sqrt(dk)
    for (int kv = threadIdx.x; kv < seq_kv; kv += blockDim.x) {
        if (causal && kv > query_pos) { scores[kv] = -FLT_MAX; continue; }
        float s = 0.0f;
        for (int d = 0; d < head_dim; ++d)
            s += q[d] * k_base[kv * head_dim + d];
        scores[kv] = s * scale;
    }
    __syncthreads();

    // 阶段 2：Online Softmax 求全局 (m, d)
    float lm = -FLT_MAX, ld = 0.0f;
    for (int kv = threadIdx.x; kv < seq_kv; kv += blockDim.x) {
        float s = scores[kv];
        float nm = fmaxf(lm, s);
        ld = ld * expf(lm - nm) + expf(s - nm);
        lm = nm;
    }
    warpReduceMD(lm, ld);   // 复用前面定义的 warpReduceMD
    __shared__ float sm[32], sd[32];
    int lane = threadIdx.x & 31, wid = threadIdx.x >> 5;
    if (lane == 0) { sm[wid] = lm; sd[wid] = ld; }
    __syncthreads();
    int nw = (blockDim.x + 31) >> 5;
    if (wid == 0) {
        lm = threadIdx.x < nw ? sm[threadIdx.x] : -FLT_MAX;
        ld = threadIdx.x < nw ? sd[threadIdx.x] : 0.0f;
        warpReduceMD(lm, ld);
        if (lane == 0) { sm[0] = lm; sd[0] = ld; }
    }
    __syncthreads();
    float gm = sm[0], ginv = 1.0f / sd[0];

    // 阶段 3：加权求和 V
    for (int d = threadIdx.x; d < head_dim; d += blockDim.x) {
        float acc = 0.0f;
        for (int kv = 0; kv < seq_kv; ++kv)
            acc += expf(scores[kv] - gm) * ginv * v_base[kv * head_dim + d];
        o[d] = acc;
    }
}
```

## 6. GQA（Grouped Query Attention）

### 6.1 核心修改

GQA 中 $G$ 个 KV Head，$H$ 个 Q Head，每组 $H/G$ 个 Q 共享一对 KV：

$$ \text{kv_head_id} = \left\lfloor \frac{\text{q_head_id}}{H/G} \right\rfloor $$

MHA = $G=H$，MQA = $G=1$。

```cuda
// 在 mhaKernel 基础上，仅修改 KV 的 head 索引
__global__ void gqaKernel(
    const float* __restrict__ Q,   // [batch, H,  seq_q,  dk]
    const float* __restrict__ K,   // [batch, G,  seq_kv, dk]
    const float* __restrict__ V,   // [batch, G,  seq_kv, dk]
    float*       __restrict__ O,
    int batch, int num_q_heads, int num_kv_heads,
    int seq_q, int seq_kv, int head_dim, bool causal)
{
    int gid       = blockIdx.x;
    int query_pos = gid % seq_q;
    int q_head_id = (gid / seq_q) % num_q_heads;
    int batch_id  = gid / (seq_q * num_q_heads);

    // 关键：Q head 映射到 KV head
    int kv_head_id = q_head_id / (num_q_heads / num_kv_heads);

    const float* q      = Q + (batch_id * num_q_heads  + q_head_id)  * seq_q  * head_dim
                            + query_pos * head_dim;
    const float* k_base = K + (batch_id * num_kv_heads + kv_head_id) * seq_kv * head_dim;
    const float* v_base = V + (batch_id * num_kv_heads + kv_head_id) * seq_kv * head_dim;
    float* o = O + (batch_id * num_q_heads + q_head_id) * seq_q * head_dim
                 + query_pos * head_dim;
    float scale = rsqrtf((float)head_dim);

    extern __shared__ float smem[];
    float* scores = smem;

    // 以下逻辑与 mhaKernel 完全相同，仅 k_base/v_base 的 head offset 不同
    for (int kv = threadIdx.x; kv < seq_kv; kv += blockDim.x) {
        if (causal && kv > query_pos) { scores[kv] = -FLT_MAX; continue; }
        float s = 0.0f;
        for (int d = 0; d < head_dim; ++d)
            s += q[d] * k_base[kv * head_dim + d];
        scores[kv] = s * scale;
    }
    __syncthreads();

    float lm = -FLT_MAX, ld = 0.0f;
    for (int kv = threadIdx.x; kv < seq_kv; kv += blockDim.x) {
        float s = scores[kv]; float nm = fmaxf(lm, s);
        ld = ld * expf(lm - nm) + expf(s - nm); lm = nm;
    }
    warpReduceMD(lm, ld);
    __shared__ float sm[32], sd[32];
    int lane = threadIdx.x & 31, wid = threadIdx.x >> 5;
    if (lane == 0) { sm[wid] = lm; sd[wid] = ld; }
    __syncthreads();
    int nw = (blockDim.x + 31) >> 5;
    if (wid == 0) {
        lm = threadIdx.x < nw ? sm[threadIdx.x] : -FLT_MAX;
        ld = threadIdx.x < nw ? sd[threadIdx.x] : 0.0f;
        warpReduceMD(lm, ld);
        if (lane == 0) { sm[0] = lm; sd[0] = ld; }
    }
    __syncthreads();
    float gm = sm[0], ginv = 1.0f / sd[0];

    for (int d = threadIdx.x; d < head_dim; d += blockDim.x) {
        float acc = 0.0f;
        for (int kv = 0; kv < seq_kv; ++kv)
            acc += expf(scores[kv] - gm) * ginv * v_base[kv * head_dim + d];
        o[d] = acc;
    }
}
```

## 7. MLA（Multi-head Latent Attention）

### 7.1 核心思路

DeepSeek-V2/V3 采用低秩 KV 压缩，只缓存压缩向量 $c_t^{KV} \in \mathbb{R}^{d_c}$，$d_c \ll d_{\text{model}}$：

$$ K_t = W^{UK} c_t^{KV},\quad V_t = W^{UV} c_t^{KV} $$

推理时实时解压 K/V，KV Cache 从 $O(T \cdot H \cdot d_k)$ 降至 $O(T \cdot d_c)$。

```cuda
// MLA：实时解压 K/V 并计算 Attention
// c_kv:  [batch, seq_kv, dc]  (压缩 KV Cache，只需缓存这个)
// W_UK:  [H, dk, dc]          (K 解压矩阵)
// W_UV:  [H, dk, dc]          (V 解压矩阵)
// Q:     [batch, H, seq_q, dk]
__global__ void mlaKernel(
    const float* __restrict__ Q,
    const float* __restrict__ c_kv,
    const float* __restrict__ W_UK,
    const float* __restrict__ W_UV,
    float*       __restrict__ O,
    int batch, int num_heads, int seq_q, int seq_kv,
    int head_dim, int dc, bool causal)
{
    int gid       = blockIdx.x;
    int query_pos = gid % seq_q;
    int head_id   = (gid / seq_q) % num_heads;
    int batch_id  = gid / (seq_q * num_heads);

    const float* q    = Q + (batch_id * num_heads + head_id) * seq_q * head_dim
                          + query_pos * head_dim;
    const float* ckv  = c_kv + batch_id * seq_kv * dc;
    const float* wuk  = W_UK + head_id * head_dim * dc;
    const float* wuv  = W_UV + head_id * head_dim * dc;
    float* o = O + (batch_id * num_heads + head_id) * seq_q * head_dim
                 + query_pos * head_dim;
    float scale = rsqrtf((float)head_dim);

    extern __shared__ float smem[];
    float* scores = smem;   // [seq_kv]

    // 阶段 1：实时解压 K 并计算 score
    // K[kv][d] = sum_j c_kv[kv][j] * W_UK[d][j]
    for (int kv = threadIdx.x; kv < seq_kv; kv += blockDim.x) {
        if (causal && kv > query_pos) { scores[kv] = -FLT_MAX; continue; }
        const float* c = ckv + kv * dc;
        float s = 0.0f;
        for (int d = 0; d < head_dim; ++d) {
            float k_val = 0.0f;
            for (int j = 0; j < dc; ++j)
                k_val += c[j] * wuk[d * dc + j];
            s += q[d] * k_val;
        }
        scores[kv] = s * scale;
    }
    __syncthreads();

    // Online Softmax（与 MHA 相同）
    float lm = -FLT_MAX, ld = 0.0f;
    for (int kv = threadIdx.x; kv < seq_kv; kv += blockDim.x) {
        float s = scores[kv]; float nm = fmaxf(lm, s);
        ld = ld * expf(lm - nm) + expf(s - nm); lm = nm;
    }
    warpReduceMD(lm, ld);
    __shared__ float sm[32], sd[32];
    int lane = threadIdx.x & 31, wid = threadIdx.x >> 5;
    if (lane == 0) { sm[wid] = lm; sd[wid] = ld; }
    __syncthreads();
    int nw = (blockDim.x + 31) >> 5;
    if (wid == 0) {
        lm = threadIdx.x < nw ? sm[threadIdx.x] : -FLT_MAX;
        ld = threadIdx.x < nw ? sd[threadIdx.x] : 0.0f;
        warpReduceMD(lm, ld);
        if (lane == 0) { sm[0] = lm; sd[0] = ld; }
    }
    __syncthreads();
    float gm = sm[0], ginv = 1.0f / sd[0];

    // 阶段 2：实时解压 V 并加权求和
    // V[kv][d] = sum_j c_kv[kv][j] * W_UV[d][j]
    for (int d = threadIdx.x; d < head_dim; d += blockDim.x) {
        float acc = 0.0f;
        for (int kv = 0; kv < seq_kv; ++kv) {
            const float* c = ckv + kv * dc;
            float v_val = 0.0f;
            for (int j = 0; j < dc; ++j)
                v_val += c[j] * wuv[d * dc + j];
            acc += expf(scores[kv] - gm) * ginv * v_val;
        }
        o[d] = acc;
    }
}
```

## 8. Flash Attention v1

### 8.1 分块 Online Softmax 合并公式

分块 1 统计量 $(m_1, d_1, o_1)$，分块 2 统计量 $(m_2, d_2, o_2)$，合并：

$$ m = \max(m_1, m_2) $$ $$ d = d_1 e^{m_1 - m} + d_2 e^{m_2 - m} $$ $$ o = \frac{d_1 e^{m_1 - m} \cdot o_1 + d_2 e^{m_2 - m} \cdot o_2}{d} $$

### 8.2 实现

```cuda
// Flash Attention v1
// 每个 Block 处理一行 Q（query_pos），遍历所有 KV 分块，
// 全程在 SRAM 内维护 (row_m, row_d, row_o)，不写中间 Score 矩阵到 HBM
#define BLOCK_KV  64
#define HEAD_DIM  128

__global__ void flashAttentionV1(
    const float* __restrict__ Q,   // [batch, H, seq_q,  HEAD_DIM]
    const float* __restrict__ K,   // [batch, H, seq_kv, HEAD_DIM]
    const float* __restrict__ V,   // [batch, H, seq_kv, HEAD_DIM]
    float*       __restrict__ O,
    int batch, int num_heads, int seq_q, int seq_kv, bool causal)
{
    // blockIdx.x = batch * num_heads 展开；blockIdx.y = query_pos
    int hb_id     = blockIdx.x;
    int query_pos = blockIdx.y;

    const float* q      = Q + hb_id * seq_q  * HEAD_DIM + query_pos * HEAD_DIM;
    const float* k_base = K + hb_id * seq_kv * HEAD_DIM;
    const float* v_base = V + hb_id * seq_kv * HEAD_DIM;
    float*       o      = O + hb_id * seq_q  * HEAD_DIM + query_pos * HEAD_DIM;
    float scale = rsqrtf((float)HEAD_DIM);

    // Shared Memory：缓存当前 KV 分块
    __shared__ float s_K[BLOCK_KV][HEAD_DIM];
    __shared__ float s_V[BLOCK_KV][HEAD_DIM];

    // 行级累积状态（全程驻留寄存器）
    float row_m = -FLT_MAX;
    float row_d = 0.0f;
    float row_o[HEAD_DIM] = {};   // 累积输出（编译器展开为寄存器）

    int num_kv_blocks = (seq_kv + BLOCK_KV - 1) / BLOCK_KV;

    for (int blk = 0; blk < num_kv_blocks; ++blk) {
        int kv_start = blk * BLOCK_KV;
        int kv_end   = min(kv_start + BLOCK_KV, seq_kv);
        int kv_size  = kv_end - kv_start;

        // 协作加载 K / V 分块到 Shared Memory
        for (int i = threadIdx.x; i < kv_size * HEAD_DIM; i += blockDim.x) {
            int r = i / HEAD_DIM, c = i % HEAD_DIM;
            s_K[r][c] = k_base[(kv_start + r) * HEAD_DIM + c];
            s_V[r][c] = v_base[(kv_start + r) * HEAD_DIM + c];
        }
        __syncthreads();

        // 计算分块内的 scores + 局部 Online Softmax
        float blk_m = -FLT_MAX, blk_d = 0.0f;
        float blk_o[HEAD_DIM] = {};

        for (int kv = 0; kv < kv_size; ++kv) {
            int abs_kv = kv_start + kv;
            if (causal && abs_kv > query_pos) continue;

            // score = Q · K[kv] / sqrt(dk)
            float score = 0.0f;
            for (int d = 0; d < HEAD_DIM; ++d)
                score += q[d] * s_K[kv][d];
            score *= scale;

            // 更新分块局部 (m, d, o)
            float new_m = fmaxf(blk_m, score);
            float exp_s = expf(score - new_m);
            float corr  = (blk_m > -FLT_MAX) ? expf(blk_m - new_m) : 0.0f;
            blk_d = blk_d * corr + exp_s;
            for (int d = 0; d < HEAD_DIM; ++d)
                blk_o[d] = blk_o[d] * corr + exp_s * s_V[kv][d];
            blk_m = new_m;
        }
        __syncthreads();

        // 将分块结果合并到行级全局状态
        // 公式：合并 (row_m, row_d, row_o) 与 (blk_m, blk_d, blk_o)
        float new_m = fmaxf(row_m, blk_m);
        float c1 = (row_m  > -FLT_MAX) ? expf(row_m  - new_m) : 0.0f;
        float c2 = (blk_m  > -FLT_MAX) ? expf(blk_m  - new_m) : 0.0f;
        float new_d = row_d * c1 + blk_d * c2;
        for (int d = 0; d < HEAD_DIM; ++d)
            row_o[d] = row_o[d] * c1 + blk_o[d] * c2;
        row_m = new_m;
        row_d = new_d;
    }

    // 写出：除以归一化因子
    float inv_d = (row_d > 0.0f) ? 1.0f / row_d : 0.0f;
    for (int d = threadIdx.x; d < HEAD_DIM; d += blockDim.x)
        o[d] = row_o[d] * inv_d;
}
```

### 8.3 收益对比

|指标|标准 Attention|Flash Attention v1|
|---|---|---|
|HBM 读写（Score 矩阵）|$O(T^2)$|$O(T \cdot d / B_{KV})$，$B_{KV}$ 为分块大小|
|中间结果写回 HBM|$S, P$ 矩阵|无|
|SRAM 占用|无|$O(B_Q \cdot d + B_{KV} \cdot d)$|
|数值精度|标准|与标准等价|

## 9. Linear Attention（CUDA 版）

### 9.1 核心公式

$$ O = \frac{\phi(Q),[\phi(K)^\top V]}{\phi(Q),[\phi(K)^\top \mathbf{1}]}, \quad \phi(x) = \max(x, 0) + 1 $$

先算状态矩阵 $S = \phi(K)^\top V \in \mathbb{R}^{d \times d}$，复杂度 $O(Td^2)$，消除 $T^2$ 项。

```cuda
#include <cuda_runtime.h>

__device__ __forceinline__ float featureMap(float x) {
    return (x >= 0.0f) ? (x + 1.0f) : expf(x);   // ELU + 1
}

// ── 阶段 1：构建状态矩阵 S = φ(K)^T · V，[H, dk, dv] ────────────────────
// K: [batch, H, T, dk]；V: [batch, H, T, dv]
__global__ void linearAttnBuildState(
    const float* __restrict__ K,
    const float* __restrict__ V,
    float*       __restrict__ S,   // [batch, H, dk, dv]
    float*       __restrict__ z,   // [batch, H, dk]（归一化向量）
    int seq, int head_dim)
{
    int hb_id = blockIdx.x;   // batch * H 展开 ID
    int dk = head_dim, dv = head_dim;

    const float* k_base = K + hb_id * seq * dk;
    const float* v_base = V + hb_id * seq * dv;
    float*       s_out  = S + hb_id * dk * dv;
    float*       z_out  = z + hb_id * dk;

    // 初始化为 0
    for (int i = threadIdx.x; i < dk * dv; i += blockDim.x) s_out[i] = 0.0f;
    for (int i = threadIdx.x; i < dk;      i += blockDim.x) z_out[i] = 0.0f;
    __syncthreads();

    // 遍历序列：S += outer(φ(k_t), v_t)
    for (int t = 0; t < seq; ++t) {
        const float* kt = k_base + t * dk;
        const float* vt = v_base + t * dv;
        // 每线程负责若干行 i（外积的行）
        for (int i = threadIdx.x; i < dk; i += blockDim.x) {
            float phi_ki = featureMap(kt[i]);
            for (int j = 0; j < dv; ++j)
                atomicAdd(&s_out[i * dv + j], phi_ki * vt[j]);
            atomicAdd(&z_out[i], phi_ki);
        }
    }
}

// ── 阶段 2：查询 O = φ(Q) · S / (φ(Q) · z) ─────────────────────────────
__global__ void linearAttnQuery(
    const float* __restrict__ Q,   // [batch, H, seq_q, dk]
    const float* __restrict__ S,   // [batch, H, dk, dv]
    const float* __restrict__ z,   // [batch, H, dk]
    float*       __restrict__ O,
    int seq_q, int head_dim)
{
    int gid   = blockIdx.x;
    int q_pos = gid % seq_q;
    int hb_id = gid / seq_q;
    int dk = head_dim, dv = head_dim;

    const float* q  = Q + hb_id * seq_q * dk + q_pos * dk;
    const float* sv = S + hb_id * dk * dv;
    const float* zv = z + hb_id * dk;
    float*       o  = O + hb_id * seq_q * dv + q_pos * dv;

    // 归一化因子 norm = φ(Q[q_pos]) · z
    float norm = 0.0f;
    for (int i = threadIdx.x; i < dk; i += blockDim.x)
        norm += featureMap(q[i]) * zv[i];
    norm = blockReduceSum(norm);
    __shared__ float s_norm;
    if (threadIdx.x == 0) s_norm = fmaxf(norm, 1e-6f);
    __syncthreads();
    float inv_norm = 1.0f / s_norm;

    // o[j] = φ(Q) · S[:, j] * inv_norm
    for (int j = threadIdx.x; j < dv; j += blockDim.x) {
        float val = 0.0f;
        for (int i = 0; i < dk; ++i)
            val += featureMap(q[i]) * sv[i * dv + j];
        o[j] = val * inv_norm;
    }
}
```

## 1. uint8 数组第 K 大

### 1.1 核心思路

uint8 取值范围 $[0, 255]$，共 256 个桶。 **计数排序（Counting Sort）思路**：

1. **Kernel 1**：每线程遍历负责的元素，局部直方图写入 shared memory，Block 归约后 `atomicAdd` 到全局直方图。
2. **CPU 端**：对全局直方图做后缀累加，找到第 $K$ 大对应的桶编号（即值）。

全程 $O(N + 256)$，远优于通用排序的 $O(N \log N)$。

```cuda
#include <cuda_runtime.h>
#include <cstdint>

// ── Kernel 1：构建全局直方图 ──────────────────────────────────────────────
// 每个 Block 先在 shared memory 中建局部直方图，再 atomicAdd 到全局
__global__ void buildHistogram(const uint8_t* __restrict__ data,
                                uint32_t*     __restrict__ hist,   // [256]
                                int N) {
    // Shared Memory 局部直方图，256 个桶，初始化为 0
    __shared__ uint32_t local_hist[256];

    // 每线程负责初始化若干桶（blockDim.x 通常 256，一线程一桶）
    for (int i = threadIdx.x; i < 256; i += blockDim.x)
        local_hist[i] = 0;
    __syncthreads();

    // 每线程遍历自己负责的数据，以步长 gridDim.x * blockDim.x 跨越全局数组
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int stride = gridDim.x * blockDim.x;
    for (int i = idx; i < N; i += stride)
        atomicAdd(&local_hist[data[i]], 1);   // 局部 atomicAdd，比全局竞争小
    __syncthreads();

    // 将局部直方图归并到全局直方图
    for (int i = threadIdx.x; i < 256; i += blockDim.x)
        atomicAdd(&hist[i], local_hist[i]);
}

// ── Host 端：对直方图做后缀累加，找第 K 大 ───────────────────────────────
// 返回第 k 大的值（1-indexed，k=1 为最大值）
int kthLargestUint8(const uint8_t* d_data, int N, int k) {
    // 分配并清零全局直方图
    uint32_t* d_hist;
    cudaMalloc(&d_hist, 256 * sizeof(uint32_t));
    cudaMemset(d_hist, 0, 256 * sizeof(uint32_t));

    // 构建直方图（Block 256 线程，Grid 按 SM 数量选取，通常 128~256）
    int block = 256;
    int grid  = std::min((N + block - 1) / block, 256);
    buildHistogram<<<grid, block>>>(d_data, d_hist, N);

    // 拷回 CPU 做后缀累加
    uint32_t h_hist[256];
    cudaMemcpy(h_hist, d_hist, 256 * sizeof(uint32_t), cudaMemcpyDeviceToHost);
    cudaFree(d_hist);

    // 从最大值（255）向最小值（0）累加计数，直到累计数 >= k
    uint64_t cum = 0;
    for (int v = 255; v >= 0; --v) {
        cum += h_hist[v];
        if (cum >= static_cast<uint64_t>(k))
            return v;   // 第 k 大的值
    }
    return -1;   // 不应到达此处
}
```

### 1.2 性能分析

|步骤|复杂度|说明|
|---|---|---|
|构建直方图|$O(N / P)$|$P$ 为并行线程数，全局内存只读|
|局部→全局 `atomicAdd`|$O(256)$ per Block|256 个桶竞争低，远小于 $N$|
|CPU 后缀累加|$O(256)$|可忽略|

## 2. CUDA 快速排序

### 2.1 核心思路

GPU 上实现通用快速排序有天然困难：递归深度不固定，子问题大小不均衡，线程利用率低。 工程上常用 **Thrust**（`thrust::sort`，内部用 CUB Radix Sort）。

面试场景考察的是对 GPU 快排思路的理解，通常分两阶段：

1. **小数组**（≤ 阈值，如 1024 元素）：Block 内使用 **Bitonic Sort**（双调排序），完全并行，无递归。
2. **大数组**：使用 **基于 Partition 的 GPU Quicksort**，每轮 partition 输出两个子问题，递归调度。

此处实现面试最常考的 **Block 内 Bitonic Sort**（适合固定大小的局部排序）。

```cuda
#include <cuda_runtime.h>

// ── Bitonic Sort（Block 内原地排序，N 必须为 2 的幂）─────────────────────
// 原理：双调序列可以在 O(log²N) 步内排序，每步全部比较独立，可完全并行
// 每线程负责 2 个元素的比较交换（Block 处理 2 * blockDim.x 个元素）
__global__ void bitonicSortKernel(float* __restrict__ data, int N) {
    extern __shared__ float smem[];
    int tid = threadIdx.x;
    int base = blockIdx.x * blockDim.x * 2;

    // 加载 2 个元素到 shared memory
    smem[2 * tid]     = (base + 2 * tid     < N) ? data[base + 2 * tid]     : FLT_MAX;
    smem[2 * tid + 1] = (base + 2 * tid + 1 < N) ? data[base + 2 * tid + 1] : FLT_MAX;
    __syncthreads();

    int n = blockDim.x * 2;   // 本 Block 处理的元素数

    // Bitonic Sort 主循环：外层 k 控制当前子序列大小，内层 j 控制比较步长
    for (int k = 2; k <= n; k <<= 1) {
        for (int j = k >> 1; j >= 1; j >>= 1) {
            int idx = tid;
            while (idx < n / 2) {
                // 计算参与比较的两个元素下标
                int i    = (idx / j) * j * 2 + (idx % j);
                int pair = i ^ j;   // 异或得到比较对端

                // 判断当前是升序段还是降序段（Bitonic 网络方向控制）
                bool ascending = ((i & k) == 0);

                if (smem[i] > smem[pair] == ascending)
                    // 不满足目标顺序，交换
                    { float tmp = smem[i]; smem[i] = smem[pair]; smem[pair] = tmp; }

                idx += blockDim.x;
            }
            __syncthreads();
        }
    }

    // 写回全局内存
    if (base + 2 * tid     < N) data[base + 2 * tid]     = smem[2 * tid];
    if (base + 2 * tid + 1 < N) data[base + 2 * tid + 1] = smem[2 * tid + 1];
}

// ── 基于 Partition 的 GPU Quicksort（动态并行，Fermi+ 支持）────────────
// 利用 CUDA Dynamic Parallelism（CDP）在 GPU 端递归启动子 kernel
__global__ void gpuQuicksortKernel(float* data, int lo, int hi) {
    if (lo >= hi) return;

    // 在 GPU 端做 partition（单线程，仅用于示意；生产中需并行 partition）
    float pivot = data[lo + (hi - lo) / 2];
    int i = lo, j = hi;
    while (i <= j) {
        while (data[i] < pivot) ++i;
        while (data[j] > pivot) --j;
        if (i <= j) {
            float tmp = data[i]; data[i] = data[j]; data[j] = tmp;
            ++i; --j;
        }
    }

    // CDP：在当前 kernel 内启动子 kernel
    if (lo < j) gpuQuicksortKernel<<<1, 1>>>(data, lo, j);
    if (i < hi) gpuQuicksortKernel<<<1, 1>>>(data, i, hi);
}

// ── 推荐面试答法：分阶段混合策略 ─────────────────────────────────────────
// 1. 将大数组切成 Block 大小的片段，每片用 Bitonic Sort 独立排序（高并行）
// 2. 对排好的片段做 GPU Merge（归并排序合并阶段）
// 3. 实际生产：直接用 thrust::sort 或 cub::DeviceRadixSort（最高性能）
void cudaSort(float* d_data, int N) {
    // 阶段 1：Block 内 Bitonic Sort（每 Block 2*256=512 元素）
    int block = 256;
    int grid  = (N + block * 2 - 1) / (block * 2);
    bitonicSortKernel<<<grid, block, block * 2 * sizeof(float)>>>(d_data, N);
    // 阶段 2：跨 Block 归并（省略，见归并排序章节）
}
```

## 3. CUDA 线性插值

### 3.1 核心思路

对 $M$ 个查询点 ${x_i}$，每个查询点落在已知控制点表 ${(t_j, y_j)}$ 的某个区间内，做线性插值。 每个线程处理一个查询点：二分查找区间 + 插值计算，完全独立，无线程间依赖。

```cuda
#include <cuda_runtime.h>

// ── 设备端二分查找：在有序数组 t[0..n-1] 中找最后一个 <= x 的下标 ────────
__device__ int lowerBoundDevice(const float* __restrict__ t, int n, float x) {
    int lo = 0, hi = n - 1;
    while (lo < hi) {
        int mid = lo + (hi - lo + 1) / 2;
        if (t[mid] <= x) lo = mid;
        else             hi = mid - 1;
    }
    return lo;
}

// ── 线性插值 Kernel ───────────────────────────────────────────────────────
// t:      [n_ctrl]  控制点 x 坐标（升序）
// y:      [n_ctrl]  控制点 y 值
// x_query:[M]       查询点 x 坐标
// out:    [M]       插值结果
__global__ void linearInterpKernel(const float* __restrict__ t,
                                    const float* __restrict__ y,
                                    const float* __restrict__ x_query,
                                    float*       __restrict__ out,
                                    int n_ctrl, int M) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= M) return;

    float xq = x_query[idx];

    // 边界处理：超出控制点范围时夹紧（clamp）
    if (xq <= t[0])         { out[idx] = y[0];         return; }
    if (xq >= t[n_ctrl-1])  { out[idx] = y[n_ctrl-1];  return; }

    // 二分查找左端点
    int lo = lowerBoundDevice(t, n_ctrl, xq);
    // 确保 lo+1 不越界
    int hi = lo + 1;

    // lerp：y = y0 + (xq - t0) / (t1 - t0) * (y1 - y0)
    float t0 = t[lo], t1 = t[hi];
    float y0 = y[lo], y1 = y[hi];
    float frac = (xq - t0) / (t1 - t0);
    out[idx] = y0 + frac * (y1 - y0);
}

// ── 特化：均匀采样控制点（步长固定，可直接计算下标，避免二分查找）────────
// t[j] = t_start + j * step
__global__ void linearInterpUniformKernel(float t_start, float step,
                                           const float* __restrict__ y,
                                           const float* __restrict__ x_query,
                                           float*       __restrict__ out,
                                           int n_ctrl, int M) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= M) return;

    float xq = x_query[idx];

    // 直接计算落点（无需二分）
    float pos = (xq - t_start) / step;
    int   lo  = __float2int_rd(pos);   // 向下取整，= (int)floorf(pos)
    lo = max(0, min(lo, n_ctrl - 2));  // clamp 到 [0, n_ctrl-2]

    float frac = pos - (float)lo;
    out[idx] = y[lo] + frac * (y[lo + 1] - y[lo]);
}

// ── Host 封装 ─────────────────────────────────────────────────────────────
void linearInterp(const float* d_t, const float* d_y, int n_ctrl,
                  const float* d_xq, float* d_out, int M) {
    int block = 256;
    int grid  = (M + block - 1) / block;
    linearInterpKernel<<<grid, block>>>(d_t, d_y, d_xq, d_out, n_ctrl, M);
}
```

## 4. CUDA 双线性插值

### 4.1 公式回顾

对非整数坐标 $(u, v)$，从图像中采样：

$$ f(u,v) = (1-t_x)(1-t_y),Q_{00} + t_x(1-t_y),Q_{10} + (1-t_x),t_y,Q_{01} + t_x,t_y,Q_{11} $$

其中 $t_x = u - \lfloor u \rfloor$，$t_y = v - \lfloor v \rfloor$。

### 4.2 图像 Resize（最常见应用）

将 $[H_{in}, W_{in}]$ 的图像缩放到 $[H_{out}, W_{out}]$，每个输出像素的源坐标：

$$ u = \left(x + 0.5\right) \cdot \frac{W_{in}}{W_{out}} - 0.5, \quad v = \left(y + 0.5\right) \cdot \frac{H_{in}}{H_{out}} - 0.5 $$

（Align Corners = False，PyTorch 默认）

```cuda
#include <cuda_runtime.h>

// ── 单通道双线性插值 Resize Kernel ───────────────────────────────────────
// in:  [H_in,  W_in]  输入图像（单通道，float）
// out: [H_out, W_out] 输出图像
__global__ void bilinearResizeKernel(const float* __restrict__ in,
                                      float*       __restrict__ out,
                                      int H_in,  int W_in,
                                      int H_out, int W_out) {
    // 每线程处理一个输出像素
    int x_out = blockIdx.x * blockDim.x + threadIdx.x;   // 列
    int y_out = blockIdx.y * blockDim.y + threadIdx.y;   // 行
    if (x_out >= W_out || y_out >= H_out) return;

    // 计算源坐标（Align Corners = False，像素中心对齐）
    float u = (x_out + 0.5f) * (float)W_in  / (float)W_out - 0.5f;
    float v = (y_out + 0.5f) * (float)H_in  / (float)H_out - 0.5f;

    // 四邻域整数坐标，clamp 到有效范围
    int x0 = max(0, min((int)floorf(u),     W_in - 1));
    int x1 = max(0, min((int)floorf(u) + 1, W_in - 1));
    int y0 = max(0, min((int)floorf(v),     H_in - 1));
    int y1 = max(0, min((int)floorf(v) + 1, H_in - 1));

    // 小数部分（插值权重）
    float tx = u - floorf(u);
    float ty = v - floorf(v);

    // 四角像素值
    float Q00 = in[y0 * W_in + x0];
    float Q10 = in[y0 * W_in + x1];
    float Q01 = in[y1 * W_in + x0];
    float Q11 = in[y1 * W_in + x1];

    // 双线性插值：先 x 方向两次 lerp，再 y 方向一次 lerp
    float R0 = Q00 + tx * (Q10 - Q00);   // y = y0 边沿
    float R1 = Q01 + tx * (Q11 - Q01);   // y = y1 边沿
    out[y_out * W_out + x_out] = R0 + ty * (R1 - R0);
}

// ── 多通道版（NCHW layout）────────────────────────────────────────────────
// in:  [N, C, H_in,  W_in]
// out: [N, C, H_out, W_out]
__global__ void bilinearResizeNCHW(const float* __restrict__ in,
                                    float*       __restrict__ out,
                                    int N, int C,
                                    int H_in,  int W_in,
                                    int H_out, int W_out) {
    // 每线程处理 (n, c, y_out, x_out) 一个元素
    int x_out = blockIdx.x * blockDim.x + threadIdx.x;
    int y_out = blockIdx.y * blockDim.y + threadIdx.y;
    int nc    = blockIdx.z;   // n * C + c 展开
    if (x_out >= W_out || y_out >= H_out || nc >= N * C) return;

    int in_offset  = nc * H_in  * W_in;
    int out_offset = nc * H_out * W_out;

    float u = (x_out + 0.5f) * (float)W_in  / (float)W_out - 0.5f;
    float v = (y_out + 0.5f) * (float)H_in  / (float)H_out - 0.5f;

    int x0 = max(0, min((int)floorf(u),     W_in - 1));
    int x1 = max(0, min((int)floorf(u) + 1, W_in - 1));
    int y0 = max(0, min((int)floorf(v),     H_in - 1));
    int y1 = max(0, min((int)floorf(v) + 1, H_in - 1));

    float tx = u - floorf(u), ty = v - floorf(v);

    float Q00 = in[in_offset + y0 * W_in + x0];
    float Q10 = in[in_offset + y0 * W_in + x1];
    float Q01 = in[in_offset + y1 * W_in + x0];
    float Q11 = in[in_offset + y1 * W_in + x1];

    float R0 = Q00 + tx * (Q10 - Q00);
    float R1 = Q01 + tx * (Q11 - Q01);
    out[out_offset + y_out * W_out + x_out] = R0 + ty * (R1 - R0);
}

// ── Host 封装 ─────────────────────────────────────────────────────────────
void bilinearResize(const float* d_in,  float* d_out,
                    int H_in, int W_in, int H_out, int W_out) {
    dim3 block(16, 16);
    dim3 grid((W_out + 15) / 16, (H_out + 15) / 16);
    bilinearResizeKernel<<<grid, block>>>(d_in, d_out, H_in, W_in, H_out, W_out);
}
```

## 5. CUDA MoE（Mixture of Experts）

### 5.1 架构回顾

$$ \text{MoE}(\mathbf{x}) = \sum_{i \in \text{TopK}} g_i(\mathbf{x}), \text{FFN}_i(\mathbf{x}) $$

GPU 上 MoE 的核心挑战：**负载不均衡**——不同 Expert 被选中的 token 数不同，朴素实现会有大量空闲线程。

工程上使用 **Expert Parallelism + Token Dispatch（all-to-all）**，但单 GPU 实现的考察要点是：

1. **Token → Expert 路由**（TopK + Softmax）
2. **按 Expert 分组**（Scatter，将 token 按目标 Expert 重排）
3. **Expert 计算**（批量 GEMM，GroupGEMM / Segment GEMM）
4. **结果聚合**（Gather + 加权求和）

```cuda
#include <cuda_runtime.h>
#include <cstdint>

// ── Kernel 1：TopK 路由，计算每个 token 的 topk expert 和对应权重 ─────────
// logits:      [num_tokens, num_experts]（路由 logits）
// topk_ids:    [num_tokens, K]（输出：top-K expert 下标）
// topk_weights:[num_tokens, K]（输出：归一化后的 gate 权重）
__global__ void topkRouterKernel(const float* __restrict__ logits,
                                  int*         __restrict__ topk_ids,
                                  float*       __restrict__ topk_weights,
                                  int num_tokens, int num_experts, int K) {
    int token_id = blockIdx.x * blockDim.x + threadIdx.x;
    if (token_id >= num_tokens) return;

    const float* row = logits + token_id * num_experts;

    // 找 TopK（小堆，O(num_experts * K)）
    // 面试简化：用冒泡选 K 次
    // 生产：用 cub::BlockRadixSort 或 partial sort
    float vals[8];   // 假设 K <= 8
    int   ids[8];
    // 初始化为最小值
    for (int i = 0; i < K; ++i) { vals[i] = -FLT_MAX; ids[i] = -1; }

    for (int e = 0; e < num_experts; ++e) {
        float v = row[e];
        // 若大于当前第 K 小，插入并维护有序
        if (v > vals[K - 1]) {
            vals[K - 1] = v; ids[K - 1] = e;
            // 插入排序维护降序
            for (int i = K - 1; i > 0 && vals[i] > vals[i-1]; --i) {
                float tv = vals[i]; vals[i] = vals[i-1]; vals[i-1] = tv;
                int   ti = ids[i];  ids[i]  = ids[i-1];  ids[i-1]  = ti;
            }
        }
    }

    // Softmax 归一化 TopK logits（只对选中的 K 个做 softmax）
    float max_v = vals[0];
    float sum   = 0.0f;
    for (int i = 0; i < K; ++i) {
        vals[i] = expf(vals[i] - max_v);
        sum += vals[i];
    }
    float inv_sum = 1.0f / sum;

    for (int i = 0; i < K; ++i) {
        topk_ids    [token_id * K + i] = ids[i];
        topk_weights[token_id * K + i] = vals[i] * inv_sum;
    }
}

// ── Kernel 2：Expert 计算（每 Expert 独立的 FFN，两层线性）─────────────────
// 此处简化为单 Expert 的单层线性变换：out = ReLU(x · W1^T) · W2^T
// tokens_for_expert: [cnt, d_model]
// W1: [hidden_dim, d_model]，W2: [d_model, hidden_dim]
__global__ void expertFFNKernel(const float* __restrict__ x,
                                 const float* __restrict__ W1,
                                 const float* __restrict__ W2,
                                 float*       __restrict__ out,
                                 int cnt, int d_model, int hidden_dim) {
    // 每 Block 处理一个 token（行），线程负责 hidden_dim 的分片
    int token_local = blockIdx.x;
    if (token_local >= cnt) return;

    const float* xt = x   + token_local * d_model;
    float*       ot = out + token_local * d_model;

    // 上投影 + ReLU：h = ReLU(xt · W1^T)，shape [hidden_dim]
    extern __shared__ float smem[];   // [hidden_dim]
    for (int h = threadIdx.x; h < hidden_dim; h += blockDim.x) {
        float val = 0.0f;
        for (int d = 0; d < d_model; ++d)
            val += xt[d] * W1[h * d_model + d];
        smem[h] = fmaxf(val, 0.0f);   // ReLU
    }
    __syncthreads();

    // 下投影：ot = smem · W2^T，shape [d_model]
    for (int d = threadIdx.x; d < d_model; d += blockDim.x) {
        float val = 0.0f;
        for (int h = 0; h < hidden_dim; ++h)
            val += smem[h] * W2[d * hidden_dim + h];
        ot[d] = val;
    }
}

// ── Kernel 3：结果聚合（Gather + 加权求和）────────────────────────────────
// expert_out:   [num_tokens * K, d_model]（每个 (token, expert) 对的输出）
// topk_weights: [num_tokens, K]
// final_out:    [num_tokens, d_model]（加权聚合结果）
__global__ void gatherWeightedSum(const float* __restrict__ expert_out,
                                   const float* __restrict__ topk_weights,
                                   float*       __restrict__ final_out,
                                   int num_tokens, int K, int d_model) {
    int token_id = blockIdx.x;
    if (token_id >= num_tokens) return;

    // 每线程处理输出维度的若干分量
    for (int d = threadIdx.x; d < d_model; d += blockDim.x) {
        float acc = 0.0f;
        for (int k = 0; k < K; ++k) {
            float w    = topk_weights[token_id * K + k];
            float val  = expert_out[(token_id * K + k) * d_model + d];
            acc += w * val;
        }
        final_out[token_id * d_model + d] = acc;
    }
}

// ── 完整 MoE 前向 Host 封装 ───────────────────────────────────────────────
void moeFwd(
    const float* d_x,           // [num_tokens, d_model]
    const float* d_router_logits,  // [num_tokens, num_experts]
    // Expert 权重（简化：所有 expert 共享同一组权重用于演示）
    const float* d_W1,          // [hidden_dim, d_model]
    const float* d_W2,          // [d_model, hidden_dim]
    float*       d_out,         // [num_tokens, d_model]
    int num_tokens, int num_experts, int K,
    int d_model, int hidden_dim)
{
    // Step 1：TopK 路由
    int*   d_topk_ids;
    float* d_topk_weights;
    cudaMalloc(&d_topk_ids,     num_tokens * K * sizeof(int));
    cudaMalloc(&d_topk_weights, num_tokens * K * sizeof(float));

    int block = 128;
    int grid  = (num_tokens + block - 1) / block;
    topkRouterKernel<<<grid, block>>>(
        d_router_logits, d_topk_ids, d_topk_weights,
        num_tokens, num_experts, K);

    // Step 2：分配 expert 输出缓冲区（简化：每个 (token, k) 对独立存储）
    float* d_expert_out;
    cudaMalloc(&d_expert_out, num_tokens * K * d_model * sizeof(float));

    // Step 3：对每个 (token, k) 对调用对应 expert 的 FFN
    // （生产中：先 scatter 到 expert 桶，批量 GroupGEMM，再 gather）
    // 此处简化：直接为每个 (token, k) 对调用一次 FFN
    for (int tk = 0; tk < num_tokens * K; ++tk) {
        expertFFNKernel<<<1, 256, hidden_dim * sizeof(float)>>>(
            d_x + (tk / K) * d_model,       // 对应 token 的输入
            d_W1, d_W2,
            d_expert_out + tk * d_model,
            1, d_model, hidden_dim);
    }

    // Step 4：加权聚合
    gatherWeightedSum<<<num_tokens, 256>>>(
        d_expert_out, d_topk_weights, d_out,
        num_tokens, K, d_model);

    cudaFree(d_topk_ids);
    cudaFree(d_topk_weights);
    cudaFree(d_expert_out);
}
```

### 5.2 生产级 MoE 优化路径

| 优化点         | 方法                                    | 收益             |
| ----------- | ------------------------------------- | -------------- |
| Expert 分组计算 | Scatter token → Expert 桶，批量 GroupGEMM | 消除逐 token 启动开销 |
| 负载均衡        | 容量因子（Capacity Factor）+ 辅助损失           | 避免热 Expert 瓶颈  |
| 专家并行（EP）    | 每个 GPU 负责若干 Expert，all-to-all 路由      | 线性扩展 Expert 数量 |
| 共享 Expert   | 部分 Expert 对所有 token 可见（DeepSeek-V2）   | 降低路由损失         |
| Token Drop  | 超出 Expert 容量的 token 直接跳过              | 严格控制计算量        |
