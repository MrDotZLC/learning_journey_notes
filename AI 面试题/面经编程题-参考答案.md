### 1. 快速排序

#### 1.1 核心思路

选取一个 **pivot**，将数组分为「小于 pivot」和「大于 pivot」两部分，递归处理两侧。 平均时间复杂度 $O(n \log n)$，最坏（已排序 + 固定 pivot）退化为 $O(n^2)$。 三路划分（Dutch National Flag）处理大量重复元素时可避免退化。

#### 1.2 标准双指针 Lomuto / Hoare 分区

```cpp
#include <vector>
#include <algorithm>  // std::swap

// ── Hoare 分区方案（左右双指针，交换次数更少）────────────────────────────
// 返回最终 pivot 落定后左半段的最右下标（不保证 pivot 精确在该位置）
// [lo, hi] 均为闭区间
int partition(std::vector<int>& arr, int lo, int hi) {
    // 三数取中：避免已排序序列退化为 O(n²)
    int mid = lo + (hi - lo) / 2;
    if (arr[lo] > arr[mid]) std::swap(arr[lo], arr[mid]);
    if (arr[lo] > arr[hi]) std::swap(arr[lo], arr[hi]);
    if (arr[mid] > arr[hi]) std::swap(arr[mid], arr[hi]);
    // 此时 arr[mid] 为三者中值，将其放到 hi-1 作为 pivot
    std::swap(arr[mid], arr[hi - 1]);
    int pivot = arr[hi - 1];

    int i = lo, j = hi - 1;
    while (true) {
        while (arr[++i] < pivot) {}   // 找到左侧 ≥ pivot 的元素
        while (arr[--j] > pivot) {}   // 找到右侧 ≤ pivot 的元素
        if (i >= j) break;
        std::swap(arr[i], arr[j]);    // 交换使其各归其位
    }
    std::swap(arr[i], arr[hi - 1]);   // pivot 归位
    return i;                          // pivot 最终下标
}

void quickSort(std::vector<int>& arr, int lo, int hi) {
    if (hi - lo < 1) return;          // 区间长度 ≤ 1，无需排序
    int p = partition(arr, lo, hi);
    quickSort(arr, lo, p - 1);        // 递归左半段
    quickSort(arr, p + 1, hi);        // 递归右半段
}

// 对外接口
void quickSort(std::vector<int>& arr) {
    if (arr.size() > 1)
        quickSort(arr, 0, static_cast<int>(arr.size()) - 1);
}
```

#### 1.3 三路划分（处理大量重复元素）

```cpp
// 将 [lo, hi] 划分为三段：< pivot | == pivot | > pivot
// 递归只处理两端，中间段（等于 pivot）直接跳过
void quickSort3Way(std::vector<int>& arr, int lo, int hi) {
    if (lo >= hi) return;

    int pivot = arr[lo + (hi - lo) / 2];
    int lt = lo;      // arr[lo..lt-1]  < pivot
    int gt = hi;      // arr[gt+1..hi]  > pivot
    int i  = lo;      // arr[lt..i-1]  == pivot（已处理）

    while (i <= gt) {
        if      (arr[i] < pivot) std::swap(arr[lt++], arr[i++]);
        else if (arr[i] > pivot) std::swap(arr[i],    arr[gt--]);
        else                     ++i;                  // 等于 pivot，直接跳过
    }
    quickSort3Way(arr, lo, lt - 1);   // 递归 < pivot 段
    quickSort3Way(arr, gt + 1, hi);   // 递归 > pivot 段
}
```

### 2. 堆排序

#### 2.1 核心思路

利用 **最大堆** 性质：堆顶永远是当前区间最大值。

1. **建堆**（Build Heap）：从最后一个非叶节点向上逐一 `siftDown`，$O(n)$。
2. **排序**：反复将堆顶与末尾交换，缩小堆范围并 `siftDown`，$O(n \log n)$。

节点 $i$（0-indexed）的子节点为 $2i+1$（左）和 $2i+2$（右）。

```cpp
#include <vector>

// ── siftDown：将下标 i 的元素下沉到正确位置 ─────────────────────────────
// heap_size：当前堆的有效大小（堆尾已交换到数组末尾，不参与堆操作）
void siftDown(std::vector<int>& arr, int i, int heap_size) {
    while (true) {
        int largest = i;
        int left  = 2 * i + 1;
        int right = 2 * i + 2;

        // 找出 i、左孩子、右孩子中的最大值
        if (left  < heap_size && arr[left]  > arr[largest]) largest = left;
        if (right < heap_size && arr[right] > arr[largest]) largest = right;

        if (largest == i) break;           // 已满足堆性质，退出

        std::swap(arr[i], arr[largest]);   // 将当前节点下沉
        i = largest;                        // 继续向下检查
    }
}

void heapSort(std::vector<int>& arr) {
    int n = static_cast<int>(arr.size());

    // 阶段 1：建最大堆
    // 最后一个非叶节点下标为 n/2 - 1，从此向上逐一 siftDown
    for (int i = n / 2 - 1; i >= 0; --i)
        siftDown(arr, i, n);

    // 阶段 2：逐步将堆顶（最大值）交换到数组末尾
    for (int end = n - 1; end > 0; --end) {
        std::swap(arr[0], arr[end]);   // 堆顶放到已排好的区域
        siftDown(arr, 0, end);         // 对缩小后的堆重新维护堆性质
    }
}
```

### 3. 归并排序

#### 3.1 核心思路

**分治**：递归将数组二分，再将两个有序子数组合并（Merge）。 时间复杂度严格 $O(n \log n)$，空间 $O(n)$（需要辅助数组）。 是 **稳定排序**，适合链表排序和外排序。

```cpp
#include <vector>

// ── merge：将 arr[lo..mid] 和 arr[mid+1..hi] 两段有序子数组合并 ─────────
void merge(std::vector<int>& arr, int lo, int mid, int hi,
           std::vector<int>& tmp) {
    // 将当前区间拷贝到临时数组，避免原地覆盖
    for (int k = lo; k <= hi; ++k) tmp[k] = arr[k];

    int i = lo;       // 左半段游标
    int j = mid + 1;  // 右半段游标

    for (int k = lo; k <= hi; ++k) {
        if      (i > mid)              arr[k] = tmp[j++];  // 左段用尽
        else if (j > hi)               arr[k] = tmp[i++];  // 右段用尽
        else if (tmp[i] <= tmp[j])     arr[k] = tmp[i++];  // 取较小值（≤ 保证稳定）
        else                           arr[k] = tmp[j++];
    }
}

void mergeSort(std::vector<int>& arr, int lo, int hi,
               std::vector<int>& tmp) {
    if (lo >= hi) return;              // 只剩一个元素，已有序

    int mid = lo + (hi - lo) / 2;
    mergeSort(arr, lo, mid, tmp);      // 排序左半段
    mergeSort(arr, mid + 1, hi, tmp);  // 排序右半段
    merge(arr, lo, mid, hi, tmp);      // 合并两段
}

void mergeSort(std::vector<int>& arr) {
    std::vector<int> tmp(arr.size());
    mergeSort(arr, 0, static_cast<int>(arr.size()) - 1, tmp);
}
```

### 4. 拓扑排序

#### 4.1 核心思路

针对 **有向无环图（DAG）**，输出一个线性顺序使得每条边 $u \to v$ 中 $u$ 出现在 $v$ 之前。 两种标准实现：

|方法|原理|检测环|
|---|---|---|
|Kahn（BFS）|维护入度表，入度为 0 则入队|最终输出节点数 < 总节点数|
|DFS 后序|DFS 完成时压栈，逆序即为拓扑序|检测到灰色节点（back edge）|

#### 4.2 Kahn 算法（BFS，推荐面试写法）

```cpp
#include <vector>
#include <queue>
#include <stdexcept>

// 输入：n 个节点（0-indexed），edges[i] = {from, to}
// 返回：拓扑序列；若图含环则抛出异常
std::vector<int> topoSort(int n, const std::vector<std::pair<int,int>>& edges) {
    std::vector<std::vector<int>> adj(n);    // 邻接表
    std::vector<int> indegree(n, 0);          // 每个节点的入度

    for (auto& [u, v] : edges) {
        adj[u].push_back(v);
        ++indegree[v];                         // v 的入度 +1
    }

    // 将所有入度为 0 的节点加入队列（可以作为起点）
    std::queue<int> q;
    for (int i = 0; i < n; ++i)
        if (indegree[i] == 0) q.push(i);

    std::vector<int> order;
    order.reserve(n);

    while (!q.empty()) {
        int u = q.front(); q.pop();
        order.push_back(u);

        for (int v : adj[u]) {
            --indegree[v];                     // 移除 u 后，v 的入度减 1
            if (indegree[v] == 0) q.push(v);  // 入度归零，可以处理
        }
    }

    if (static_cast<int>(order.size()) != n)
        throw std::runtime_error("Graph has a cycle");

    return order;
}
```

#### 4.3 DFS 后序反转

```cpp
#include <vector>
#include <stack>
#include <stdexcept>

// color: 0=白(未访问), 1=灰(递归栈中), 2=黑(已完成)
void dfs(int u, std::vector<std::vector<int>>& adj,
         std::vector<int>& color, std::stack<int>& stk) {
    color[u] = 1;                               // 标记为"处理中"
    for (int v : adj[u]) {
        if (color[v] == 1)
            throw std::runtime_error("Cycle detected");  // 发现回边 → 有环
        if (color[v] == 0)
            dfs(v, adj, color, stk);
    }
    color[u] = 2;                               // 所有后继已处理完
    stk.push(u);                                // 后序压栈
}

std::vector<int> topoSortDFS(int n, const std::vector<std::pair<int,int>>& edges) {
    std::vector<std::vector<int>> adj(n);
    for (auto& [u, v] : edges) adj[u].push_back(v);

    std::vector<int> color(n, 0);
    std::stack<int> stk;

    for (int i = 0; i < n; ++i)
        if (color[i] == 0)
            dfs(i, adj, color, stk);

    std::vector<int> order;
    while (!stk.empty()) {
        order.push_back(stk.top());
        stk.pop();
    }
    return order;   // 弹栈顺序即为拓扑序
}
```

### 5. LRU Cache

#### 5.1 核心思路

**Least Recently Used**：容量满时淘汰最久未访问的条目。

经典实现：**双向链表 + 哈希表**。

- 链表维护访问顺序（头部最新，尾部最旧）。
- 哈希表 `key → 链表节点指针` 实现 $O(1)$ 查找。
- `get` / `put` 均摊 $O(1)$。

```cpp
#include <unordered_map>
#include <list>
#include <stdexcept>

class LRUCache {
public:
    explicit LRUCache(int capacity) : cap_(capacity) {}

    // 查询 key；不存在返回 -1，存在则将节点移到链表头（最新位置）
    int get(int key) {
        auto it = map_.find(key);
        if (it == map_.end()) return -1;

        // 将命中节点移到链表头部（表示最近访问）
        moveToFront(it->second);
        return it->second->second;   // 返回 value
    }

    // 插入或更新 key-value
    void put(int key, int value) {
        auto it = map_.find(key);
        if (it != map_.end()) {
            // key 已存在：更新值并移到头部
            it->second->second = value;
            moveToFront(it->second);
            return;
        }

        // 容量已满：移除链表尾部节点（最久未访问）
        if (static_cast<int>(map_.size()) >= cap_) {
            auto lru = std::prev(list_.end());  // 尾部节点
            map_.erase(lru->first);              // 从哈希表中删除
            list_.erase(lru);                    // 从链表中删除
        }

        // 插入新节点到链表头部
        list_.emplace_front(key, value);
        map_[key] = list_.begin();
    }

private:
    using KV   = std::pair<int, int>;
    using Iter = std::list<KV>::iterator;

    int                           cap_;
    std::list<KV>                 list_;   // front=最新, back=最旧
    std::unordered_map<int, Iter> map_;    // key → 链表节点迭代器

    // 将节点 it 移到链表头部（splice 不会使其他迭代器失效）
    void moveToFront(Iter it) {
        list_.splice(list_.begin(), list_, it);
    }
};
```

### 6. 快速幂

#### 6.1 核心思路

反复平方法（Exponentiation by Squaring）：

$$ x^n = \begin{cases} 1 & n = 0 \ (x^{n/2})^2 & n \text{ 为偶数} \ x \cdot (x^{(n-1)/2})^2 & n \text{ 为奇数} \end{cases} $$

时间复杂度 $O(\log n)$，空间 $O(1)$（迭代版）。

```cpp
#include <cstdint>

// ── 整数快速幂（模运算版，防溢出）─────────────────────────────────────────
// 计算 base^exp mod modulus
long long fastPow(long long base, long long exp, long long modulus) {
    long long result = 1;
    base %= modulus;              // 先对底数取模，避免初始溢出

    while (exp > 0) {
        if (exp & 1)              // 当前位为 1：将当前 base 乘入结果
            result = result * base % modulus;
        base = base * base % modulus;  // base 自平方，对应指数右移一位
        exp >>= 1;                // 指数右移
    }
    return result;
}

// ── 浮点快速幂（用于 pow(x, n) 类题目，需处理负指数）───────────────────
double myPow(double x, int n) {
    // 用 long long 防止 n = INT_MIN 取反时溢出
    long long N = n;
    if (N < 0) { x = 1.0 / x; N = -N; }

    double result = 1.0;
    while (N > 0) {
        if (N & 1) result *= x;
        x *= x;
        N >>= 1;
    }
    return result;
}
```

### 7. 求平方根

#### 7.1 核心思路

两种方法并存：

|方法|特点|
|---|---|
|二分查找|整数平方根，精确，$O(\log n)$|
|牛顿迭代|浮点平方根，二次收敛，$O(\log(1/\varepsilon))$|

**牛顿迭代**：求 $f(x) = x^2 - S = 0$ 的根，迭代公式：

$$ x_{k+1} = \frac{1}{2}\left(x_k + \frac{S}{x_k}\right) $$

每次迭代有效位数翻倍（二次收敛）。

```cpp
#include <cmath>    // std::abs（浮点版）
#include <cstdint>

// ── 方法 1：二分查找（整数平方根，向下取整）──────────────────────────────
// 返回 ⌊√n⌋
int sqrtBinarySearch(int n) {
    if (n < 2) return n;
    int lo = 1, hi = n / 2;    // 结果一定 ≤ n/2（n ≥ 4 时）
    while (lo <= hi) {
        long long mid = lo + (hi - lo) / 2;
        long long sq  = mid * mid;
        if      (sq == n) return static_cast<int>(mid);
        else if (sq < n)  lo = static_cast<int>(mid) + 1;
        else              hi = static_cast<int>(mid) - 1;
    }
    return hi;   // hi 是最后一个满足 hi² ≤ n 的值
}

// ── 方法 2：牛顿迭代（浮点平方根）────────────────────────────────────────
// 精度由 eps 控制
double sqrtNewton(double S, double eps = 1e-9) {
    if (S < 0) return -1.0;    // 非法输入
    if (S == 0) return 0.0;

    double x = S;              // 初始猜测值（也可用 S/2 或更好的估计）
    while (true) {
        double next = 0.5 * (x + S / x);   // 牛顿迭代公式
        if (std::abs(next - x) < eps)       // 收敛判断
            break;
        x = next;
    }
    return x;
}
```

### 8. 大数乘法

#### 8.1 核心思路

用字符串模拟竖式乘法。两个长度分别为 $m$、$n$ 的数相乘，结果最多 $m+n$ 位。

关键观察：$\text{num1}[i] \times \text{num2}[j]$ 的结果，**个位**落在结果数组的 $[i+j+1]$，**十位**落在 $[i+j]$。

```cpp
#include <string>
#include <algorithm>

// 输入：两个非负整数字符串（不含前导零，除了 "0" 本身）
// 输出：乘积字符串
std::string multiply(const std::string& num1, const std::string& num2) {
    int m = static_cast<int>(num1.size());
    int n = static_cast<int>(num2.size());

    // 结果最多 m+n 位，初始化为 0
    std::vector<int> pos(m + n, 0);

    // 从低位到高位逐位相乘（从字符串末尾开始）
    for (int i = m - 1; i >= 0; --i) {
        for (int j = n - 1; j >= 0; --j) {
            int mul  = (num1[i] - '0') * (num2[j] - '0');
            int p1   = i + j;       // 十位落点
            int p2   = i + j + 1;   // 个位落点
            int sum  = mul + pos[p2];

            pos[p2] = sum % 10;     // 个位存入结果
            pos[p1] += sum / 10;    // 进位加到高位
        }
    }

    // 将 pos 数组转换为字符串，跳过前导零
    std::string result;
    for (int digit : pos) {
        if (result.empty() && digit == 0) continue;  // 跳过前导零
        result += static_cast<char>('0' + digit);
    }

    return result.empty() ? "0" : result;
}
```

### 9. Hamming Weight（位 1 的个数）

#### 9.1 核心思路

统计一个 32 位无符号整数的二进制表示中 `1` 的个数（popcount）。

三种常见实现：

```cpp
#include <cstdint>

// ── 方法 1：逐位检查，O(32) ──────────────────────────────────────────────
int hammingWeightNaive(uint32_t n) {
    int count = 0;
    while (n) {
        count += n & 1;   // 检查最低位
        n >>= 1;
    }
    return count;
}

// ── 方法 2：Brian Kernighan 技巧，O(popcount) ────────────────────────────
// n & (n-1) 每次消去最低位的 1，循环次数等于 1 的个数
int hammingWeightKernighan(uint32_t n) {
    int count = 0;
    while (n) {
        n &= (n - 1);   // 清除最低位的 1（例：1010 → 1000）
        ++count;
    }
    return count;
}

// ── 方法 3：SWAR（并行位计数），O(1)，面试展示用 ─────────────────────────
// 原理：将 32 位分为多组，组内并行累加，最后汇总
int hammingWeightSWAR(uint32_t n) {
    n = n - ((n >> 1) & 0x55555555u);          // 每 2 位：计算该 2 位内的 1 数量
    n = (n & 0x33333333u) + ((n >> 2) & 0x33333333u);  // 每 4 位合并
    n = (n + (n >> 4)) & 0x0f0f0f0fu;          // 每 8 位合并
    return static_cast<int>((n * 0x01010101u) >> 24);   // 横向累加所有字节
}

// ── 方法 4：编译器内建（实际生产代码首选）────────────────────────────────
int hammingWeightBuiltin(uint32_t n) {
    return __builtin_popcount(n);   // GCC/Clang 直接映射到 POPCNT 指令
}
```

### 10. K-coverage Intervals

#### 10.1 问题定义

给定 $n$ 个区间 $[l_i, r_i]$，找出所有被 **至少 $k$ 个区间覆盖** 的点（或连续区间段）。

#### 10.2 差分数组解法，$O(n \log n)$

```cpp
#include <vector>
#include <algorithm>
#include <utility>

// 返回所有被至少 k 个区间覆盖的连续段，每段用 [start, end] 表示（闭区间）
// 输入区间为整数闭区间 [l, r]
std::vector<std::pair<int,int>>
kCoverageIntervals(const std::vector<std::pair<int,int>>& intervals, int k) {
    // 差分数组：在 l 处 +1，在 r+1 处 -1
    // 扫描线扫过时，前缀和即为当前点被覆盖的区间数
    std::vector<std::pair<int,int>> events;
    events.reserve(2 * intervals.size());

    for (auto& [l, r] : intervals) {
        events.emplace_back(l,     +1);   // 区间开始
        events.emplace_back(r + 1, -1);   // 区间结束（开区间端点）
    }

    // 按坐标排序；同坐标时先处理 +1（保证端点处计数正确）
    std::sort(events.begin(), events.end(), [](const auto& a, const auto& b) {
        return a.first != b.first ? a.first < b.first : a.second > b.second;
    });

    std::vector<std::pair<int,int>> result;
    int coverage = 0;
    int segStart = -1;   // 当前 ≥k 覆盖段的起点，-1 表示未开始

    for (auto& [pos, delta] : events) {
        int prevCoverage = coverage;
        coverage += delta;

        // 从 < k 变为 ≥ k：段开始
        if (prevCoverage < k && coverage >= k) {
            segStart = pos;
        }
        // 从 ≥ k 变为 < k：段结束
        if (prevCoverage >= k && coverage < k) {
            result.emplace_back(segStart, pos - 1);
        }
    }

    return result;
}
```

### 11. 按指定概率采样

#### 11.1 问题定义

给定 $n$ 个选项，概率分别为 $p_0, p_1, \ldots, p_{n-1}$（$\sum p_i = 1$），每次采样返回下标 $i$，使得 $P(\text{返回} i) = p_i$。

#### 11.2 前缀和 + 二分查找，$O(n)$ 构建，$O(\log n)$ 查询

```cpp
#include <vector>
#include <random>
#include <algorithm>
#include <stdexcept>

class WeightedSampler {
public:
    // weights[i] 为第 i 项的权重（不要求归一化）
    explicit WeightedSampler(const std::vector<double>& weights)
        : gen_(std::random_device{}())
    {
        if (weights.empty())
            throw std::invalid_argument("Empty weights");

        // 构建前缀和数组（累积分布函数 CDF）
        prefix_.resize(weights.size());
        prefix_[0] = weights[0];
        for (size_t i = 1; i < weights.size(); ++i) {
            prefix_[i] = prefix_[i-1] + weights[i];
            if (weights[i] < 0)
                throw std::invalid_argument("Negative weight");
        }
        // prefix_.back() 为总权重，uniform 分布上界
        dist_ = std::uniform_real_distribution<double>(0.0, prefix_.back());
    }

    // 返回采样下标
    int sample() {
        double r = dist_(gen_);    // 在 [0, total_weight) 均匀采样
        // lower_bound 找到第一个 prefix_[i] >= r 的位置，即所属区间
        auto it = std::lower_bound(prefix_.begin(), prefix_.end(), r);
        return static_cast<int>(it - prefix_.begin());
    }

private:
    std::vector<double>                        prefix_;  // 前缀权重和
    std::mt19937                               gen_;     // Mersenne Twister PRNG
    std::uniform_real_distribution<double>    dist_;
};
```

#### 11.3 别名方法（Alias Method），$O(n)$ 构建，$O(1)$ 查询

```cpp
// Alias Method：适合高频采样场景
// 将 n 个概率"重分配"为 n 个等面积柱，每柱最多两个原始分类
class AliasSampler {
public:
    explicit AliasSampler(std::vector<double> probs)
        : n_(probs.size()), gen_(std::random_device{}()),
          prob_(probs.size()), alias_(probs.size()),
          coin_(0.0, 1.0), dice_(0, static_cast<int>(probs.size()) - 1)
    {
        double n = static_cast<double>(n_);
        // 将概率归一化为均匀分布的倍数（每格期望面积 = 1）
        for (auto& p : probs) p *= n;

        std::vector<int> small, large;
        for (int i = 0; i < n_; ++i)
            (probs[i] < 1.0 ? small : large).push_back(i);

        // 将"小格"用"大格"填满
        while (!small.empty() && !large.empty()) {
            int s = small.back(); small.pop_back();
            int l = large.back(); large.pop_back();
            prob_[s]  = probs[s];
            alias_[s] = l;
            probs[l]  = (probs[l] + probs[s]) - 1.0;   // 减去分给 s 的部分
            (probs[l] < 1.0 ? small : large).push_back(l);
        }
        // 剩余数值接近 1.0，浮点误差导致分类不准确，强制置 1
        for (int i : small) prob_[i] = 1.0;
        for (int i : large) prob_[i] = 1.0;
    }

    int sample() {
        int i = dice_(gen_);                    // 随机选一列
        return (coin_(gen_) < prob_[i]) ? i : alias_[i];  // 按列内概率决定取原或别名
    }

private:
    int                                     n_;
    std::mt19937                            gen_;
    std::vector<double>                     prob_;   // 每列原概率
    std::vector<int>                        alias_;  // 每列的别名（填充列）
    std::uniform_real_distribution<double>  coin_;
    std::uniform_int_distribution<int>      dice_;
};
```

### 12. 循环 Buffer / 循环队列

#### 12.1 核心思路

固定大小数组实现的 FIFO 队列，头尾指针循环利用空间。 区分**满**与**空**：常见方案有两种——

1. **浪费一格**：`(tail + 1) % cap == head` 为满。
2. **额外计数器**（本文采用）：用 `size_` 变量，无歧义。

```cpp
#include <vector>
#include <stdexcept>

template<typename T>
class CircularBuffer {
public:
    explicit CircularBuffer(int capacity)
        : buf_(capacity), cap_(capacity), head_(0), tail_(0), size_(0) {}

    // 尾部入队
    void push(const T& val) {
        if (full())
            throw std::overflow_error("CircularBuffer is full");
        buf_[tail_] = val;
        tail_ = (tail_ + 1) % cap_;   // 尾指针循环前进
        ++size_;
    }

    // 头部出队，返回队首元素
    T pop() {
        if (empty())
            throw std::underflow_error("CircularBuffer is empty");
        T val = buf_[head_];
        head_ = (head_ + 1) % cap_;   // 头指针循环前进
        --size_;
        return val;
    }

    // 查看队首，不出队
    T& front() {
        if (empty()) throw std::underflow_error("CircularBuffer is empty");
        return buf_[head_];
    }

    bool empty() const { return size_ == 0; }
    bool full()  const { return size_ == cap_; }
    int  size()  const { return size_; }
    int  capacity() const { return cap_; }

private:
    std::vector<T> buf_;   // 底层存储
    int cap_;              // 总容量
    int head_;             // 队首下标（下一个出队位置）
    int tail_;             // 队尾下标（下一个入队位置）
    int size_;             // 当前元素数量
};
```

#### 12.2 无锁单生产者单消费者（SPSC）版

```cpp
#include <vector>
#include <atomic>
#include <optional>

// 无锁 SPSC 循环队列：生产者写 tail，消费者写 head，无需互斥锁
// 注意：cap 必须为 2 的幂，利用位掩码替代取模（更快）
template<typename T>
class SPSCQueue {
public:
    explicit SPSCQueue(size_t cap)
        : buf_(cap), mask_(cap - 1), head_(0), tail_(0) {
        // cap 必须是 2 的幂
        if ((cap & mask_) != 0)
            throw std::invalid_argument("Capacity must be a power of 2");
    }

    // 生产者调用：入队，失败返回 false（队满）
    bool push(const T& val) {
        size_t tail = tail_.load(std::memory_order_relaxed);
        size_t next = (tail + 1) & mask_;
        // 队满判断：下一个 tail 追上 head
        if (next == head_.load(std::memory_order_acquire)) return false;
        buf_[tail] = val;
        tail_.store(next, std::memory_order_release);   // release 保证写入可见
        return true;
    }

    // 消费者调用：出队，失败返回 nullopt（队空）
    std::optional<T> pop() {
        size_t head = head_.load(std::memory_order_relaxed);
        if (head == tail_.load(std::memory_order_acquire)) return std::nullopt;
        T val = buf_[head];
        head_.store((head + 1) & mask_, std::memory_order_release);
        return val;
    }

private:
    std::vector<T>         buf_;
    size_t                 mask_;                      // cap - 1，用于位与取模
    std::atomic<size_t>    head_;                      // 消费者端写
    char                   pad_[64 - sizeof(size_t)];  // 避免 head/tail 共享缓存行
    std::atomic<size_t>    tail_;                      // 生产者端写
};
```
