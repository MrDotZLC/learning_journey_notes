### 1. 快速排序

#### 1.1 核心思路

选取一个 **pivot**，将数组分为「小于 pivot」和「大于 pivot」两部分，递归处理两侧。 平均时间复杂度 $O(n \log n)$，最坏（已排序 + 固定 pivot）退化为 $O(n^2)$。 三路划分（Dutch National Flag）处理大量重复元素时可避免退化。

#### 1.2 标准双指针 Lomuto / Hoare 分区

```cpp
#include <vector>
#include <algorithm>  // std::swap

// ── Hoare 分区方案（左右双指针，交换次数更少）────────────────────────────
// 返回最终 pivot 落定后左半段的最右下标（不保证 pivot 精确在该位置）
// [lo, hi] 均为闭区间
int partition(std::vector<int>& arr, int lo, int hi) {
    // 三数取中：避免已排序序列退化为 O(n²)
    int mid = lo + (hi - lo) / 2;
    if (arr[lo] > arr[mid]) std::swap(arr[lo], arr[mid]);
    if (arr[lo] > arr[hi]) std::swap(arr[lo], arr[hi]);
    if (arr[mid] > arr[hi]) std::swap(arr[mid], arr[hi]);
    // 此时 arr[mid] 为三者中值，将其放到 hi-1 作为 pivot
    std::swap(arr[mid], arr[hi - 1]);
    int pivot = arr[hi - 1];

    int i = lo, j = hi - 1;
    while (true) {
        while (arr[++i] < pivot) {}   // 找到左侧 ≥ pivot 的元素
        while (arr[--j] > pivot) {}   // 找到右侧 ≤ pivot 的元素
        if (i >= j) break;
        std::swap(arr[i], arr[j]);    // 交换使其各归其位
    }
    std::swap(arr[i], arr[hi - 1]);   // pivot 归位
    return i;                          // pivot 最终下标
}

void quickSort(std::vector<int>& arr, int lo, int hi) {
    if (hi - lo < 1) return;          // 区间长度 ≤ 1，无需排序
    int p = partition(arr, lo, hi);
    quickSort(arr, lo, p - 1);        // 递归左半段
    quickSort(arr, p + 1, hi);        // 递归右半段
}

// 对外接口
void quickSort(std::vector<int>& arr) {
    if (arr.size() > 1)
        quickSort(arr, 0, static_cast<int>(arr.size()) - 1);
}
```

#### 1.3 三路划分（处理大量重复元素）

```cpp
// 将 [lo, hi] 划分为三段：< pivot | == pivot | > pivot
// 递归只处理两端，中间段（等于 pivot）直接跳过
void quickSort3Way(std::vector<int>& arr, int lo, int hi) {
    if (lo >= hi) return;

    int pivot = arr[lo + (hi - lo) / 2];
    int lt = lo;      // arr[lo..lt-1]  < pivot
    int gt = hi;      // arr[gt+1..hi]  > pivot
    int i  = lo;      // arr[lt..i-1]  == pivot（已处理）

    while (i <= gt) {
        if      (arr[i] < pivot) std::swap(arr[lt++], arr[i++]);
        else if (arr[i] > pivot) std::swap(arr[i],    arr[gt--]);
        else                     ++i;                  // 等于 pivot，直接跳过
    }
    quickSort3Way(arr, lo, lt - 1);   // 递归 < pivot 段
    quickSort3Way(arr, gt + 1, hi);   // 递归 > pivot 段
}
```

### 2. 堆排序

#### 2.1 核心思路

利用 **最大堆** 性质：堆顶永远是当前区间最大值。

1. **建堆**（Build Heap）：从最后一个非叶节点向上逐一 `siftDown`，$O(n)$。
2. **排序**：反复将堆顶与末尾交换，缩小堆范围并 `siftDown`，$O(n \log n)$。

节点 $i$（0-indexed）的子节点为 $2i+1$（左）和 $2i+2$（右）。

```cpp
#include <vector>

// ── siftDown：将下标 i 的元素下沉到正确位置 ─────────────────────────────
// heap_size：当前堆的有效大小（堆尾已交换到数组末尾，不参与堆操作）
void siftDown(std::vector<int>& arr, int i, int heap_size) {
    while (true) {
        int largest = i;
        int left  = 2 * i + 1;
        int right = 2 * i + 2;

        // 找出 i、左孩子、右孩子中的最大值
        if (left  < heap_size && arr[left]  > arr[largest]) largest = left;
        if (right < heap_size && arr[right] > arr[largest]) largest = right;

        if (largest == i) break;           // 已满足堆性质，退出

        std::swap(arr[i], arr[largest]);   // 将当前节点下沉
        i = largest;                        // 继续向下检查
    }
}

void heapSort(std::vector<int>& arr) {
    int n = static_cast<int>(arr.size());

    // 阶段 1：建最大堆
    // 最后一个非叶节点下标为 n/2 - 1，从此向上逐一 siftDown
    for (int i = n / 2 - 1; i >= 0; --i)
        siftDown(arr, i, n);

    // 阶段 2：逐步将堆顶（最大值）交换到数组末尾
    for (int end = n - 1; end > 0; --end) {
        std::swap(arr[0], arr[end]);   // 堆顶放到已排好的区域
        siftDown(arr, 0, end);         // 对缩小后的堆重新维护堆性质
    }
}
```

### 3. 归并排序

#### 3.1 核心思路

**分治**：递归将数组二分，再将两个有序子数组合并（Merge）。 时间复杂度严格 $O(n \log n)$，空间 $O(n)$（需要辅助数组）。 是 **稳定排序**，适合链表排序和外排序。

```cpp
#include <vector>

// ── merge：将 arr[lo..mid] 和 arr[mid+1..hi] 两段有序子数组合并 ─────────
void merge(std::vector<int>& arr, int lo, int mid, int hi,
           std::vector<int>& tmp) {
    // 将当前区间拷贝到临时数组，避免原地覆盖
    for (int k = lo; k <= hi; ++k) tmp[k] = arr[k];

    int i = lo;       // 左半段游标
    int j = mid + 1;  // 右半段游标

    for (int k = lo; k <= hi; ++k) {
        if      (i > mid)              arr[k] = tmp[j++];  // 左段用尽
        else if (j > hi)               arr[k] = tmp[i++];  // 右段用尽
        else if (tmp[i] <= tmp[j])     arr[k] = tmp[i++];  // 取较小值（≤ 保证稳定）
        else                           arr[k] = tmp[j++];
    }
}

void mergeSort(std::vector<int>& arr, int lo, int hi,
               std::vector<int>& tmp) {
    if (lo >= hi) return;              // 只剩一个元素，已有序

    int mid = lo + (hi - lo) / 2;
    mergeSort(arr, lo, mid, tmp);      // 排序左半段
    mergeSort(arr, mid + 1, hi, tmp);  // 排序右半段
    merge(arr, lo, mid, hi, tmp);      // 合并两段
}

void mergeSort(std::vector<int>& arr) {
    std::vector<int> tmp(arr.size());
    mergeSort(arr, 0, static_cast<int>(arr.size()) - 1, tmp);
}
```

### 4. 拓扑排序

#### 4.1 核心思路

针对 **有向无环图（DAG）**，输出一个线性顺序使得每条边 $u \to v$ 中 $u$ 出现在 $v$ 之前。 两种标准实现：

|方法|原理|检测环|
|---|---|---|
|Kahn（BFS）|维护入度表，入度为 0 则入队|最终输出节点数 < 总节点数|
|DFS 后序|DFS 完成时压栈，逆序即为拓扑序|检测到灰色节点（back edge）|

#### 4.2 Kahn 算法（BFS，推荐面试写法）

```cpp
#include <vector>
#include <queue>
#include <stdexcept>

// 输入：n 个节点（0-indexed），edges[i] = {from, to}
// 返回：拓扑序列；若图含环则抛出异常
std::vector<int> topoSort(int n, const std::vector<std::pair<int,int>>& edges) {
    std::vector<std::vector<int>> adj(n);    // 邻接表
    std::vector<int> indegree(n, 0);          // 每个节点的入度

    for (auto& [u, v] : edges) {
        adj[u].push_back(v);
        ++indegree[v];                         // v 的入度 +1
    }

    // 将所有入度为 0 的节点加入队列（可以作为起点）
    std::queue<int> q;
    for (int i = 0; i < n; ++i)
        if (indegree[i] == 0) q.push(i);

    std::vector<int> order;
    order.reserve(n);

    while (!q.empty()) {
        int u = q.front(); q.pop();
        order.push_back(u);

        for (int v : adj[u]) {
            --indegree[v];                     // 移除 u 后，v 的入度减 1
            if (indegree[v] == 0) q.push(v);  // 入度归零，可以处理
        }
    }

    if (static_cast<int>(order.size()) != n)
        throw std::runtime_error("Graph has a cycle");

    return order;
}
```

#### 4.3 DFS 后序反转

```cpp
#include <vector>
#include <stack>
#include <stdexcept>

// color: 0=白(未访问), 1=灰(递归栈中), 2=黑(已完成)
void dfs(int u, std::vector<std::vector<int>>& adj,
         std::vector<int>& color, std::stack<int>& stk) {
    color[u] = 1;                               // 标记为"处理中"
    for (int v : adj[u]) {
        if (color[v] == 1)
            throw std::runtime_error("Cycle detected");  // 发现回边 → 有环
        if (color[v] == 0)
            dfs(v, adj, color, stk);
    }
    color[u] = 2;                               // 所有后继已处理完
    stk.push(u);                                // 后序压栈
}

std::vector<int> topoSortDFS(int n, const std::vector<std::pair<int,int>>& edges) {
    std::vector<std::vector<int>> adj(n);
    for (auto& [u, v] : edges) adj[u].push_back(v);

    std::vector<int> color(n, 0);
    std::stack<int> stk;

    for (int i = 0; i < n; ++i)
        if (color[i] == 0)
            dfs(i, adj, color, stk);

    std::vector<int> order;
    while (!stk.empty()) {
        order.push_back(stk.top());
        stk.pop();
    }
    return order;   // 弹栈顺序即为拓扑序
}
```

### 5. LRU Cache

#### 5.1 核心思路

**Least Recently Used**：容量满时淘汰最久未访问的条目。

经典实现：**双向链表 + 哈希表**。

- 链表维护访问顺序（头部最新，尾部最旧）。
- 哈希表 `key → 链表节点指针` 实现 $O(1)$ 查找。
- `get` / `put` 均摊 $O(1)$。

```cpp
#include <unordered_map>
#include <list>
#include <stdexcept>

class LRUCache {
public:
    explicit LRUCache(int capacity) : cap_(capacity) {}

    // 查询 key；不存在返回 -1，存在则将节点移到链表头（最新位置）
    int get(int key) {
        auto it = map_.find(key);
        if (it == map_.end()) return -1;

        // 将命中节点移到链表头部（表示最近访问）
        moveToFront(it->second);
        return it->second->second;   // 返回 value
    }

    // 插入或更新 key-value
    void put(int key, int value) {
        auto it = map_.find(key);
        if (it != map_.end()) {
            // key 已存在：更新值并移到头部
            it->second->second = value;
            moveToFront(it->second);
            return;
        }

        // 容量已满：移除链表尾部节点（最久未访问）
        if (static_cast<int>(map_.size()) >= cap_) {
            auto lru = std::prev(list_.end());  // 尾部节点
            map_.erase(lru->first);              // 从哈希表中删除
            list_.erase(lru);                    // 从链表中删除
        }

        // 插入新节点到链表头部
        list_.emplace_front(key, value);
        map_[key] = list_.begin();
    }

private:
    using KV   = std::pair<int, int>;
    using Iter = std::list<KV>::iterator;

    int                           cap_;
    std::list<KV>                 list_;   // front=最新, back=最旧
    std::unordered_map<int, Iter> map_;    // key → 链表节点迭代器

    // 将节点 it 移到链表头部（splice 不会使其他迭代器失效）
    void moveToFront(Iter it) {
        list_.splice(list_.begin(), list_, it);
    }
};
```

### 6. 快速幂

#### 6.1 核心思路

反复平方法（Exponentiation by Squaring）：

$$ x^n = \begin{cases} 1 & n = 0 \ (x^{n/2})^2 & n \text{ 为偶数} \ x \cdot (x^{(n-1)/2})^2 & n \text{ 为奇数} \end{cases} $$

时间复杂度 $O(\log n)$，空间 $O(1)$（迭代版）。

```cpp
#include <cstdint>

// ── 整数快速幂（模运算版，防溢出）─────────────────────────────────────────
// 计算 base^exp mod modulus
long long fastPow(long long base, long long exp, long long modulus) {
    long long result = 1;
    base %= modulus;              // 先对底数取模，避免初始溢出

    while (exp > 0) {
        if (exp & 1)              // 当前位为 1：将当前 base 乘入结果
            result = result * base % modulus;
        base = base * base % modulus;  // base 自平方，对应指数右移一位
        exp >>= 1;                // 指数右移
    }
    return result;
}

// ── 浮点快速幂（用于 pow(x, n) 类题目，需处理负指数）───────────────────
double myPow(double x, int n) {
    // 用 long long 防止 n = INT_MIN 取反时溢出
    long long N = n;
    if (N < 0) { x = 1.0 / x; N = -N; }

    double result = 1.0;
    while (N > 0) {
        if (N & 1) result *= x;
        x *= x;
        N >>= 1;
    }
    return result;
}
```

### 7. 求平方根

#### 7.1 核心思路

两种方法并存：

|方法|特点|
|---|---|
|二分查找|整数平方根，精确，$O(\log n)$|
|牛顿迭代|浮点平方根，二次收敛，$O(\log(1/\varepsilon))$|

**牛顿迭代**：求 $f(x) = x^2 - S = 0$ 的根，迭代公式：

$$ x_{k+1} = \frac{1}{2}\left(x_k + \frac{S}{x_k}\right) $$

每次迭代有效位数翻倍（二次收敛）。

```cpp
#include <cmath>    // std::abs（浮点版）
#include <cstdint>

// ── 方法 1：二分查找（整数平方根，向下取整）──────────────────────────────
// 返回 ⌊√n⌋
int sqrtBinarySearch(int n) {
    if (n < 2) return n;
    int lo = 1, hi = n / 2;    // 结果一定 ≤ n/2（n ≥ 4 时）
    while (lo <= hi) {
        long long mid = lo + (hi - lo) / 2;
        long long sq  = mid * mid;
        if      (sq == n) return static_cast<int>(mid);
        else if (sq < n)  lo = static_cast<int>(mid) + 1;
        else              hi = static_cast<int>(mid) - 1;
    }
    return hi;   // hi 是最后一个满足 hi² ≤ n 的值
}

// ── 方法 2：牛顿迭代（浮点平方根）────────────────────────────────────────
// 精度由 eps 控制
double sqrtNewton(double S, double eps = 1e-9) {
    if (S < 0) return -1.0;    // 非法输入
    if (S == 0) return 0.0;

    double x = S;              // 初始猜测值（也可用 S/2 或更好的估计）
    while (true) {
        double next = 0.5 * (x + S / x);   // 牛顿迭代公式
        if (std::abs(next - x) < eps)       // 收敛判断
            break;
        x = next;
    }
    return x;
}
```

### 8. 大数乘法

#### 8.1 核心思路

用字符串模拟竖式乘法。两个长度分别为 $m$、$n$ 的数相乘，结果最多 $m+n$ 位。

关键观察：$\text{num1}[i] \times \text{num2}[j]$ 的结果，**个位**落在结果数组的 $[i+j+1]$，**十位**落在 $[i+j]$。

```cpp
#include <string>
#include <algorithm>

// 输入：两个非负整数字符串（不含前导零，除了 "0" 本身）
// 输出：乘积字符串
std::string multiply(const std::string& num1, const std::string& num2) {
    int m = static_cast<int>(num1.size());
    int n = static_cast<int>(num2.size());

    // 结果最多 m+n 位，初始化为 0
    std::vector<int> pos(m + n, 0);

    // 从低位到高位逐位相乘（从字符串末尾开始）
    for (int i = m - 1; i >= 0; --i) {
        for (int j = n - 1; j >= 0; --j) {
            int mul  = (num1[i] - '0') * (num2[j] - '0');
            int p1   = i + j;       // 十位落点
            int p2   = i + j + 1;   // 个位落点
            int sum  = mul + pos[p2];

            pos[p2] = sum % 10;     // 个位存入结果
            pos[p1] += sum / 10;    // 进位加到高位
        }
    }

    // 将 pos 数组转换为字符串，跳过前导零
    std::string result;
    for (int digit : pos) {
        if (result.empty() && digit == 0) continue;  // 跳过前导零
        result += static_cast<char>('0' + digit);
    }

    return result.empty() ? "0" : result;
}
```

### 9. Hamming Weight（位 1 的个数）

#### 9.1 核心思路

统计一个 32 位无符号整数的二进制表示中 `1` 的个数（popcount）。

三种常见实现：

```cpp
#include <cstdint>

// ── 方法 1：逐位检查，O(32) ──────────────────────────────────────────────
int hammingWeightNaive(uint32_t n) {
    int count = 0;
    while (n) {
        count += n & 1;   // 检查最低位
        n >>= 1;
    }
    return count;
}

// ── 方法 2：Brian Kernighan 技巧，O(popcount) ────────────────────────────
// n & (n-1) 每次消去最低位的 1，循环次数等于 1 的个数
int hammingWeightKernighan(uint32_t n) {
    int count = 0;
    while (n) {
        n &= (n - 1);   // 清除最低位的 1（例：1010 → 1000）
        ++count;
    }
    return count;
}

// ── 方法 3：SWAR（并行位计数），O(1)，面试展示用 ─────────────────────────
// 原理：将 32 位分为多组，组内并行累加，最后汇总
int hammingWeightSWAR(uint32_t n) {
    n = n - ((n >> 1) & 0x55555555u);          // 每 2 位：计算该 2 位内的 1 数量
    n = (n & 0x33333333u) + ((n >> 2) & 0x33333333u);  // 每 4 位合并
    n = (n + (n >> 4)) & 0x0f0f0f0fu;          // 每 8 位合并
    return static_cast<int>((n * 0x01010101u) >> 24);   // 横向累加所有字节
}

// ── 方法 4：编译器内建（实际生产代码首选）────────────────────────────────
int hammingWeightBuiltin(uint32_t n) {
    return __builtin_popcount(n);   // GCC/Clang 直接映射到 POPCNT 指令
}
```

### 10. K-coverage Intervals

#### 10.1 问题定义

给定 $n$ 个区间 $[l_i, r_i]$，找出所有被 **至少 $k$ 个区间覆盖** 的点（或连续区间段）。

#### 10.2 差分数组解法，$O(n \log n)$

```cpp
#include <vector>
#include <algorithm>
#include <utility>

// 返回所有被至少 k 个区间覆盖的连续段，每段用 [start, end] 表示（闭区间）
// 输入区间为整数闭区间 [l, r]
std::vector<std::pair<int,int>>
kCoverageIntervals(const std::vector<std::pair<int,int>>& intervals, int k) {
    // 差分数组：在 l 处 +1，在 r+1 处 -1
    // 扫描线扫过时，前缀和即为当前点被覆盖的区间数
    std::vector<std::pair<int,int>> events;
    events.reserve(2 * intervals.size());

    for (auto& [l, r] : intervals) {
        events.emplace_back(l,     +1);   // 区间开始
        events.emplace_back(r + 1, -1);   // 区间结束（开区间端点）
    }

    // 按坐标排序；同坐标时先处理 +1（保证端点处计数正确）
    std::sort(events.begin(), events.end(), [](const auto& a, const auto& b) {
        return a.first != b.first ? a.first < b.first : a.second > b.second;
    });

    std::vector<std::pair<int,int>> result;
    int coverage = 0;
    int segStart = -1;   // 当前 ≥k 覆盖段的起点，-1 表示未开始

    for (auto& [pos, delta] : events) {
        int prevCoverage = coverage;
        coverage += delta;

        // 从 < k 变为 ≥ k：段开始
        if (prevCoverage < k && coverage >= k) {
            segStart = pos;
        }
        // 从 ≥ k 变为 < k：段结束
        if (prevCoverage >= k && coverage < k) {
            result.emplace_back(segStart, pos - 1);
        }
    }

    return result;
}
```

### 11. 按指定概率采样

#### 11.1 问题定义

给定 $n$ 个选项，概率分别为 $p_0, p_1, \ldots, p_{n-1}$（$\sum p_i = 1$），每次采样返回下标 $i$，使得 $P(\text{返回} i) = p_i$。

#### 11.2 前缀和 + 二分查找，$O(n)$ 构建，$O(\log n)$ 查询

```cpp
#include <vector>
#include <random>
#include <algorithm>
#include <stdexcept>

class WeightedSampler {
public:
    // weights[i] 为第 i 项的权重（不要求归一化）
    explicit WeightedSampler(const std::vector<double>& weights)
        : gen_(std::random_device{}())
    {
        if (weights.empty())
            throw std::invalid_argument("Empty weights");

        // 构建前缀和数组（累积分布函数 CDF）
        prefix_.resize(weights.size());
        prefix_[0] = weights[0];
        for (size_t i = 1; i < weights.size(); ++i) {
            prefix_[i] = prefix_[i-1] + weights[i];
            if (weights[i] < 0)
                throw std::invalid_argument("Negative weight");
        }
        // prefix_.back() 为总权重，uniform 分布上界
        dist_ = std::uniform_real_distribution<double>(0.0, prefix_.back());
    }

    // 返回采样下标
    int sample() {
        double r = dist_(gen_);    // 在 [0, total_weight) 均匀采样
        // lower_bound 找到第一个 prefix_[i] >= r 的位置，即所属区间
        auto it = std::lower_bound(prefix_.begin(), prefix_.end(), r);
        return static_cast<int>(it - prefix_.begin());
    }

private:
    std::vector<double>                        prefix_;  // 前缀权重和
    std::mt19937                               gen_;     // Mersenne Twister PRNG
    std::uniform_real_distribution<double>    dist_;
};
```

#### 11.3 别名方法（Alias Method），$O(n)$ 构建，$O(1)$ 查询

```cpp
// Alias Method：适合高频采样场景
// 将 n 个概率"重分配"为 n 个等面积柱，每柱最多两个原始分类
class AliasSampler {
public:
    explicit AliasSampler(std::vector<double> probs)
        : n_(probs.size()), gen_(std::random_device{}()),
          prob_(probs.size()), alias_(probs.size()),
          coin_(0.0, 1.0), dice_(0, static_cast<int>(probs.size()) - 1)
    {
        double n = static_cast<double>(n_);
        // 将概率归一化为均匀分布的倍数（每格期望面积 = 1）
        for (auto& p : probs) p *= n;

        std::vector<int> small, large;
        for (int i = 0; i < n_; ++i)
            (probs[i] < 1.0 ? small : large).push_back(i);

        // 将"小格"用"大格"填满
        while (!small.empty() && !large.empty()) {
            int s = small.back(); small.pop_back();
            int l = large.back(); large.pop_back();
            prob_[s]  = probs[s];
            alias_[s] = l;
            probs[l]  = (probs[l] + probs[s]) - 1.0;   // 减去分给 s 的部分
            (probs[l] < 1.0 ? small : large).push_back(l);
        }
        // 剩余数值接近 1.0，浮点误差导致分类不准确，强制置 1
        for (int i : small) prob_[i] = 1.0;
        for (int i : large) prob_[i] = 1.0;
    }

    int sample() {
        int i = dice_(gen_);                    // 随机选一列
        return (coin_(gen_) < prob_[i]) ? i : alias_[i];  // 按列内概率决定取原或别名
    }

private:
    int                                     n_;
    std::mt19937                            gen_;
    std::vector<double>                     prob_;   // 每列原概率
    std::vector<int>                        alias_;  // 每列的别名（填充列）
    std::uniform_real_distribution<double>  coin_;
    std::uniform_int_distribution<int>      dice_;
};
```

### 12. 循环 Buffer / 循环队列

#### 12.1 核心思路

固定大小数组实现的 FIFO 队列，头尾指针循环利用空间。 区分**满**与**空**：常见方案有两种——

1. **浪费一格**：`(tail + 1) % cap == head` 为满。
2. **额外计数器**（本文采用）：用 `size_` 变量，无歧义。

```cpp
#include <vector>
#include <stdexcept>

template<typename T>
class CircularBuffer {
public:
    explicit CircularBuffer(int capacity)
        : buf_(capacity), cap_(capacity), head_(0), tail_(0), size_(0) {}

    // 尾部入队
    void push(const T& val) {
        if (full())
            throw std::overflow_error("CircularBuffer is full");
        buf_[tail_] = val;
        tail_ = (tail_ + 1) % cap_;   // 尾指针循环前进
        ++size_;
    }

    // 头部出队，返回队首元素
    T pop() {
        if (empty())
            throw std::underflow_error("CircularBuffer is empty");
        T val = buf_[head_];
        head_ = (head_ + 1) % cap_;   // 头指针循环前进
        --size_;
        return val;
    }

    // 查看队首，不出队
    T& front() {
        if (empty()) throw std::underflow_error("CircularBuffer is empty");
        return buf_[head_];
    }

    bool empty() const { return size_ == 0; }
    bool full()  const { return size_ == cap_; }
    int  size()  const { return size_; }
    int  capacity() const { return cap_; }

private:
    std::vector<T> buf_;   // 底层存储
    int cap_;              // 总容量
    int head_;             // 队首下标（下一个出队位置）
    int tail_;             // 队尾下标（下一个入队位置）
    int size_;             // 当前元素数量
};
```

#### 12.2 无锁单生产者单消费者（SPSC）版

```cpp
#include <vector>
#include <atomic>
#include <optional>

// 无锁 SPSC 循环队列：生产者写 tail，消费者写 head，无需互斥锁
// 注意：cap 必须为 2 的幂，利用位掩码替代取模（更快）
template<typename T>
class SPSCQueue {
public:
    explicit SPSCQueue(size_t cap)
        : buf_(cap), mask_(cap - 1), head_(0), tail_(0) {
        // cap 必须是 2 的幂
        if ((cap & mask_) != 0)
            throw std::invalid_argument("Capacity must be a power of 2");
    }

    // 生产者调用：入队，失败返回 false（队满）
    bool push(const T& val) {
        size_t tail = tail_.load(std::memory_order_relaxed);
        size_t next = (tail + 1) & mask_;
        // 队满判断：下一个 tail 追上 head
        if (next == head_.load(std::memory_order_acquire)) return false;
        buf_[tail] = val;
        tail_.store(next, std::memory_order_release);   // release 保证写入可见
        return true;
    }

    // 消费者调用：出队，失败返回 nullopt（队空）
    std::optional<T> pop() {
        size_t head = head_.load(std::memory_order_relaxed);
        if (head == tail_.load(std::memory_order_acquire)) return std::nullopt;
        T val = buf_[head];
        head_.store((head + 1) & mask_, std::memory_order_release);
        return val;
    }

private:
    std::vector<T>         buf_;
    size_t                 mask_;                      // cap - 1，用于位与取模
    std::atomic<size_t>    head_;                      // 消费者端写
    char                   pad_[64 - sizeof(size_t)];  // 避免 head/tail 共享缓存行
    std::atomic<size_t>    tail_;                      // 生产者端写
};
```

### 1. 生产者-消费者

#### 1.1 核心思路

经典并发模型：生产者向共享缓冲区写数据，消费者从中读取。 核心问题：

- **互斥**：同一时刻只有一个线程访问缓冲区（`std::mutex`）。
- **同步**：缓冲区满时生产者阻塞；缓冲区空时消费者阻塞（`std::condition_variable`）。

两个条件变量 `not_full` / `not_empty` 精确唤醒，避免虚假唤醒（`while` 而非 `if`）。

```cpp
#include <queue>
#include <mutex>
#include <condition_variable>
#include <thread>
#include <iostream>
#include <optional>

template<typename T>
class BoundedQueue {
public:
    explicit BoundedQueue(size_t cap) : cap_(cap) {}

    // 生产者调用：阻塞直到有空位
    void push(T val) {
        std::unique_lock<std::mutex> lk(mu_);
        // 用 while 防止虚假唤醒（spurious wakeup）
        not_full_.wait(lk, [this]{ return q_.size() < cap_ || stopped_; });
        if (stopped_) return;
        q_.push(std::move(val));
        not_empty_.notify_one();   // 通知至少一个消费者
    }

    // 消费者调用：阻塞直到有数据；返回 nullopt 表示已关闭且队列为空
    std::optional<T> pop() {
        std::unique_lock<std::mutex> lk(mu_);
        not_empty_.wait(lk, [this]{ return !q_.empty() || stopped_; });
        if (q_.empty()) return std::nullopt;   // 关闭且无剩余数据
        T val = std::move(q_.front());
        q_.pop();
        not_full_.notify_one();    // 通知至少一个生产者
        return val;
    }

    // 关闭队列，唤醒所有阻塞线程
    void stop() {
        {
            std::lock_guard<std::mutex> lk(mu_);
            stopped_ = true;
        }
        not_full_.notify_all();
        not_empty_.notify_all();
    }

private:
    std::queue<T>           q_;
    size_t                  cap_;
    bool                    stopped_ = false;
    std::mutex              mu_;
    std::condition_variable not_full_;
    std::condition_variable not_empty_;
};

// ── 使用示例 ─────────────────────────────────────────────────────────────
void producerConsumerDemo() {
    BoundedQueue<int> bq(4);   // 容量为 4 的有界队列

    // 生产者线程：生产 10 个数字后关闭队列
    std::thread producer([&]{
        for (int i = 0; i < 10; ++i) {
            bq.push(i);
            std::cout << "[Producer] pushed " << i << "\n";
        }
        bq.stop();
    });

    // 消费者线程：持续消费直到队列关闭且为空
    std::thread consumer([&]{
        while (auto val = bq.pop()) {
            std::cout << "[Consumer] got " << *val << "\n";
        }
    });

    producer.join();
    consumer.join();
}
```

#### 1.2 多消费者扩展

上述 `BoundedQueue` 天然支持多生产者多消费者（MPMC）：

- `notify_one()` 保证每次只唤醒一个等待线程，避免惊群（thundering herd）。
- `notify_all()` 仅在 `stop()` 时调用，确保所有线程能退出。

### 2. 多线程循环按顺序打印递增数字

#### 2.1 问题定义

$N$ 个线程，线程 $i$（$0$-indexed）负责打印满足 $n \equiv i \pmod{N}$ 的数字，要求严格按 $1, 2, 3, \ldots$ 的顺序打印，总共打印到 $M$。

核心挑战：强制线程间的**全局有序执行**。

#### 2.2 方案一：条件变量 + 轮转令牌

```cpp
#include <thread>
#include <mutex>
#include <condition_variable>
#include <vector>
#include <iostream>

// N 个线程按序打印 1..M
// 使用"令牌"变量 turn 标记当前轮到哪个线程
void printInOrder(int N, int M) {
    std::mutex              mu;
    std::condition_variable cv;
    int                     turn    = 0;   // 当前应执行的线程编号
    int                     current = 1;   // 下一个要打印的数字

    std::vector<std::thread> threads;
    for (int id = 0; id < N; ++id) {
        threads.emplace_back([&, id] {
            while (true) {
                std::unique_lock<std::mutex> lk(mu);
                // 等待直到轮到自己，且还有数字可打印
                cv.wait(lk, [&]{ return turn == id || current > M; });
                if (current > M) break;

                std::cout << "Thread-" << id << " prints: " << current++ << "\n";
                turn = (turn + 1) % N;   // 令牌传给下一个线程
                cv.notify_all();          // 唤醒所有等待线程，让下一个线程接力
            }
        });
    }

    for (auto& t : threads) t.join();
}
```

#### 2.3 方案二：信号量链（更低唤醒开销）

```cpp
#include <thread>
#include <vector>
#include <semaphore>   // C++20
#include <iostream>

// 每个线程有自己的信号量，形成环形"接力棒"
// thread[i] 完成后 release thread[(i+1)%N] 的信号量
void printInOrderSemaphore(int N, int M) {
    // 初始化：thread-0 的信号量为 1（先行），其余为 0（等待）
    std::vector<std::counting_semaphore<1>> sems;
    sems.reserve(N);
    for (int i = 0; i < N; ++i)
        sems.emplace_back(i == 0 ? 1 : 0);

    int current = 1;   // 共享计数器（由信号量保证互斥，无需额外锁）
    std::vector<std::thread> threads;

    for (int id = 0; id < N; ++id) {
        threads.emplace_back([&, id] {
            while (true) {
                sems[id].acquire();               // 等待自己的令牌
                if (current > M) {
                    sems[(id + 1) % N].release(); // 唤醒下一个使其也能退出
                    break;
                }
                std::cout << "Thread-" << id << ": " << current++ << "\n";
                sems[(id + 1) % N].release();     // 传递令牌给下一个线程
            }
        });
    }

    for (auto& t : threads) t.join();
}
```

### 3. 手撕内存池（支持 `new[]` / `delete[]`）

#### 3.1 核心思路

标准 `new`/`delete` 每次调用系统分配器（`malloc`/`free`），有较高的系统调用开销和内存碎片风险。 **内存池**预先申请一大块内存，在其上手动管理分配/释放：

- **固定大小内存池**：所有对象大小相同，用侵入式空闲链表（free list）管理空闲块，分配/释放均 $O(1)$。
- 支持 `operator new[]` / `operator delete[]`：重载类级别运算符，池内管理数组分配。

```cpp
#include <cstddef>
#include <cstdlib>
#include <stdexcept>
#include <new>

// ── 固定对象大小内存池 ────────────────────────────────────────────────────
// 用法：替换特定类型的 new/delete，消除 malloc 开销
// 模板参数 ObjSize：每个对象的字节数；Capacity：池中最大对象数
template<size_t ObjSize, size_t Capacity>
class FixedMemoryPool {
    // 保证对象足够容纳一个指针（用于空闲链表节点）
    static constexpr size_t BlockSize = ObjSize > sizeof(void*) ? ObjSize : sizeof(void*);

public:
    FixedMemoryPool() {
        // 将原始内存切割成 Capacity 个等大块，串成链表
        // 每块的前 sizeof(void*) 字节存储下一个空闲块的指针
        for (size_t i = 0; i < Capacity - 1; ++i) {
            // 将当前块的起始位置解释为指针，指向下一块
            *reinterpret_cast<void**>(&arena_[i * BlockSize]) =
                &arena_[(i + 1) * BlockSize];
        }
        // 最后一块的 next 指针为 nullptr（链表尾）
        *reinterpret_cast<void**>(&arena_[(Capacity - 1) * BlockSize]) = nullptr;
        free_list_ = &arena_[0];   // 空闲链表头指向第一块
    }

    // 分配一个对象大小的内存块，O(1)
    void* allocate() {
        if (!free_list_)
            throw std::bad_alloc();       // 池已耗尽

        void* block = free_list_;         // 取出链表头
        free_list_ = *reinterpret_cast<void**>(free_list_);  // 头指针后移
        return block;
    }

    // 归还内存块到池，O(1)
    void deallocate(void* ptr) noexcept {
        // 将归还的块插回链表头（LIFO，有利于缓存局部性）
        *reinterpret_cast<void**>(ptr) = free_list_;
        free_list_ = ptr;
    }

    // 查询剩余可用块数（调试用）
    size_t available() const noexcept {
        size_t count = 0;
        void* p = free_list_;
        while (p) { ++count; p = *reinterpret_cast<void**>(p); }
        return count;
    }

private:
    alignas(std::max_align_t) char arena_[BlockSize * Capacity];  // 原始内存
    void* free_list_;   // 空闲链表头
};

// ── 将内存池集成到类，支持 new[] / delete[] ──────────────────────────────
// 每个 Foo 对象大小 16 字节，池容量 1024
struct Foo {
    int  data[4];   // 占 16 字节

    // 重载 operator new：从池中分配
    static void* operator new(size_t size) {
        if (size != sizeof(Foo)) return ::operator new(size);  // 非标准大小走系统分配
        return pool_.allocate();
    }

    // 重载 operator delete：归还到池
    static void operator delete(void* ptr) noexcept {
        pool_.deallocate(ptr);
    }

    // 重载 operator new[]：数组分配（需要额外存储元素数量，用于析构）
    static void* operator new[](size_t total_size) {
        // total_size = sizeof(Foo) * n + 可能的对齐填充（实现相关）
        // 简单方案：直接走系统分配器（数组 new 语义复杂，生产中用容器代替）
        return ::operator new(total_size);
    }

    // 重载 operator delete[]
    static void operator delete[](void* ptr) noexcept {
        ::operator delete(ptr);
    }

private:
    static FixedMemoryPool<sizeof(Foo), 1024> pool_;
};

// 类静态成员定义（放在 .cpp 中）
FixedMemoryPool<sizeof(Foo), 1024> Foo::pool_{};
```

#### 3.2 通用多 Size 分级内存池（面试进阶）

```cpp
#include <array>
#include <vector>
#include <cstddef>
#include <cstdlib>

// 类似 tcmalloc 的思路：按对象大小分级，每级维护独立空闲链表
// 此处简化为 8 个 Size Class：8, 16, 32, 64, 128, 256, 512, 1024 字节
class TieredMemoryPool {
    static constexpr size_t kNumClasses = 8;
    // 每级对应的块大小
    static constexpr size_t kSizes[kNumClasses] = {8,16,32,64,128,256,512,1024};

public:
    void* allocate(size_t size) {
        int cls = sizeClass(size);
        if (cls < 0) return std::malloc(size);   // 超大对象走系统分配

        if (free_lists_[cls]) {
            // 从对应级别的空闲链表取出一块
            void* block = free_lists_[cls];
            free_lists_[cls] = *reinterpret_cast<void**>(block);
            return block;
        }
        // 链表为空：从系统申请一批（批量摊还系统调用开销）
        return refill(cls);
    }

    void deallocate(void* ptr, size_t size) noexcept {
        int cls = sizeClass(size);
        if (cls < 0) { std::free(ptr); return; }
        // 归还：插回对应级别链表头
        *reinterpret_cast<void**>(ptr) = free_lists_[cls];
        free_lists_[cls] = ptr;
    }

private:
    void* free_lists_[kNumClasses] = {};   // 每级空闲链表头

    // 找到能容纳 size 的最小 Size Class，找不到返回 -1
    static int sizeClass(size_t size) {
        for (int i = 0; i < kNumClasses; ++i)
            if (size <= kSizes[i]) return i;
        return -1;
    }

    // 向系统申请一批块（默认 16 块），串成链表返回头部
    void* refill(int cls) {
        constexpr int kBatch = 16;
        size_t block_size = kSizes[cls];
        char* mem = static_cast<char*>(std::malloc(block_size * kBatch));
        if (!mem) throw std::bad_alloc();

        // 将 [1, kBatch-1] 串成链表
        for (int i = 0; i < kBatch - 1; ++i)
            *reinterpret_cast<void**>(mem + i * block_size) =
                mem + (i + 1) * block_size;
        *reinterpret_cast<void**>(mem + (kBatch-1) * block_size) = free_lists_[cls];
        free_lists_[cls] = mem + block_size;   // [1..] 进链表

        return mem;   // 返回第 0 块给调用方
    }
};
```

### 4. 智能指针

#### 4.1 `unique_ptr` 实现

**核心语义**：独占所有权，不可拷贝，可移动，析构时自动释放。 关键：禁用拷贝构造/赋值，实现移动构造/赋值，支持自定义 Deleter。

```cpp
#include <utility>    // std::exchange, std::forward
#include <functional> // std::default_delete（参考）

// ── MyUniquePtr<T, Deleter> ───────────────────────────────────────────────
template<typename T, typename Deleter = std::default_delete<T>>
class MyUniquePtr {
public:
    // 构造：接管裸指针（不隐式转换，explicit 防止意外赋值）
    explicit MyUniquePtr(T* ptr = nullptr, Deleter d = Deleter{})
        : ptr_(ptr), deleter_(std::move(d)) {}

    // 析构：调用 deleter 释放资源
    ~MyUniquePtr() {
        if (ptr_) deleter_(ptr_);
    }

    // ── 禁止拷贝（独占所有权）────────────────────────────────────────────
    MyUniquePtr(const MyUniquePtr&)            = delete;
    MyUniquePtr& operator=(const MyUniquePtr&) = delete;

    // ── 移动构造：转移所有权，来源置空 ──────────────────────────────────
    MyUniquePtr(MyUniquePtr&& other) noexcept
        : ptr_(std::exchange(other.ptr_, nullptr)),
          deleter_(std::move(other.deleter_)) {}

    // ── 移动赋值：先释放自身，再接管来源 ────────────────────────────────
    MyUniquePtr& operator=(MyUniquePtr&& other) noexcept {
        if (this != &other) {
            if (ptr_) deleter_(ptr_);                   // 释放当前持有资源
            ptr_     = std::exchange(other.ptr_, nullptr);
            deleter_ = std::move(other.deleter_);
        }
        return *this;
    }

    // ── 访问接口 ──────────────────────────────────────────────────────────
    T* get()           const noexcept { return ptr_; }
    T& operator*()     const          { return *ptr_; }
    T* operator->()    const noexcept { return ptr_; }
    explicit operator bool() const noexcept { return ptr_ != nullptr; }

    // 释放所有权（返回裸指针，调用方负责删除）
    T* release() noexcept { return std::exchange(ptr_, nullptr); }

    // 重置：释放当前资源，持有新指针
    void reset(T* ptr = nullptr) noexcept {
        if (ptr_) deleter_(ptr_);
        ptr_ = ptr;
    }

private:
    T*      ptr_;
    Deleter deleter_;
};

// make_unique（C++14 风格）
template<typename T, typename... Args>
MyUniquePtr<T> makeUnique(Args&&... args) {
    return MyUniquePtr<T>(new T(std::forward<Args>(args)...));
}
```

#### 4.2 `shared_ptr` 实现

**核心语义**：共享所有权，引用计数。引用计数归零时释放对象；`weak_ptr` 用弱引用，不影响计数。

```cpp
#include <atomic>
#include <utility>
#include <stdexcept>

// ── 控制块：存储引用计数 ────────────────────────────────────────────────
struct ControlBlock {
    std::atomic<int> strong_count{1};   // shared_ptr 数量
    std::atomic<int> weak_count{1};     // weak_ptr 数量（+1 表示 shared 本身的弱引用）

    virtual void destroyObject() noexcept = 0;  // 销毁被管理对象
    virtual void destroySelf()   noexcept = 0;  // 销毁控制块自身
    virtual ~ControlBlock() = default;
};

template<typename T>
struct ControlBlockImpl : ControlBlock {
    T* ptr;
    explicit ControlBlockImpl(T* p) : ptr(p) {}
    void destroyObject() noexcept override { delete ptr; ptr = nullptr; }
    void destroySelf()   noexcept override { delete this; }
};

// ── 前向声明 ──────────────────────────────────────────────────────────────
template<typename T> class MyWeakPtr;

// ── MySharedPtr<T> ────────────────────────────────────────────────────────
template<typename T>
class MySharedPtr {
    friend class MyWeakPtr<T>;
public:
    // 默认构造：空指针
    MySharedPtr() noexcept : ptr_(nullptr), cb_(nullptr) {}

    // 接管裸指针，创建控制块
    explicit MySharedPtr(T* ptr)
        : ptr_(ptr), cb_(ptr ? new ControlBlockImpl<T>(ptr) : nullptr) {}

    // 拷贝构造：共享所有权，strong_count +1
    MySharedPtr(const MySharedPtr& other) noexcept
        : ptr_(other.ptr_), cb_(other.cb_) {
        if (cb_) cb_->strong_count.fetch_add(1, std::memory_order_relaxed);
    }

    // 移动构造：转移所有权，来源置空（不改变引用计数）
    MySharedPtr(MySharedPtr&& other) noexcept
        : ptr_(std::exchange(other.ptr_, nullptr)),
          cb_ (std::exchange(other.cb_,  nullptr)) {}

    // 析构：strong_count -1；归零时销毁对象；weak_count 也归零时销毁控制块
    ~MySharedPtr() { release(); }

    // 拷贝赋值
    MySharedPtr& operator=(const MySharedPtr& other) noexcept {
        if (this != &other) {
            release();
            ptr_ = other.ptr_;
            cb_  = other.cb_;
            if (cb_) cb_->strong_count.fetch_add(1, std::memory_order_relaxed);
        }
        return *this;
    }

    // 移动赋值
    MySharedPtr& operator=(MySharedPtr&& other) noexcept {
        if (this != &other) {
            release();
            ptr_ = std::exchange(other.ptr_, nullptr);
            cb_  = std::exchange(other.cb_,  nullptr);
        }
        return *this;
    }

    T* get()           const noexcept { return ptr_; }
    T& operator*()     const          { return *ptr_; }
    T* operator->()    const noexcept { return ptr_; }
    explicit operator bool() const noexcept { return ptr_ != nullptr; }

    int use_count() const noexcept {
        return cb_ ? cb_->strong_count.load(std::memory_order_relaxed) : 0;
    }

private:
    T*            ptr_;
    ControlBlock* cb_;

    // 供 weak_ptr::lock() 调用，从控制块直接构造（不再 new）
    MySharedPtr(T* ptr, ControlBlock* cb) : ptr_(ptr), cb_(cb) {
        if (cb_) cb_->strong_count.fetch_add(1, std::memory_order_relaxed);
    }

    void release() noexcept {
        if (!cb_) return;
        // strong_count 减 1，若归零则销毁对象
        if (cb_->strong_count.fetch_sub(1, std::memory_order_acq_rel) == 1) {
            cb_->destroyObject();
            // weak_count 减 1（对应 strong 持有的那个弱引用），若归零则销毁控制块
            if (cb_->weak_count.fetch_sub(1, std::memory_order_acq_rel) == 1)
                cb_->destroySelf();
        }
        ptr_ = nullptr;
        cb_  = nullptr;
    }
};

// ── MyWeakPtr<T> ──────────────────────────────────────────────────────────
template<typename T>
class MyWeakPtr {
public:
    MyWeakPtr() noexcept : ptr_(nullptr), cb_(nullptr) {}

    // 从 shared_ptr 构造：weak_count +1（不影响 strong_count）
    MyWeakPtr(const MySharedPtr<T>& sp) noexcept
        : ptr_(sp.ptr_), cb_(sp.cb_) {
        if (cb_) cb_->weak_count.fetch_add(1, std::memory_order_relaxed);
    }

    ~MyWeakPtr() { release(); }

    MyWeakPtr(const MyWeakPtr& other) noexcept : ptr_(other.ptr_), cb_(other.cb_) {
        if (cb_) cb_->weak_count.fetch_add(1, std::memory_order_relaxed);
    }

    MyWeakPtr& operator=(const MyWeakPtr& other) noexcept {
        if (this != &other) {
            release();
            ptr_ = other.ptr_; cb_ = other.cb_;
            if (cb_) cb_->weak_count.fetch_add(1, std::memory_order_relaxed);
        }
        return *this;
    }

    bool expired() const noexcept {
        return !cb_ || cb_->strong_count.load(std::memory_order_acquire) == 0;
    }

    // 尝试提升为 shared_ptr；若对象已销毁则返回空 shared_ptr
    MySharedPtr<T> lock() const noexcept {
        if (expired()) return MySharedPtr<T>{};
        // 用 CAS 原子地将 strong_count 从非零值 +1，防止竞态
        int cnt = cb_->strong_count.load(std::memory_order_relaxed);
        while (cnt > 0) {
            if (cb_->strong_count.compare_exchange_weak(
                    cnt, cnt + 1,
                    std::memory_order_acq_rel,
                    std::memory_order_relaxed))
                return MySharedPtr<T>(ptr_, cb_);  // 使用私有构造，不再 +1
        }
        return MySharedPtr<T>{};   // 对象已销毁（strong_count 已降至 0）
    }

private:
    T*            ptr_;
    ControlBlock* cb_;

    void release() noexcept {
        if (!cb_) return;
        if (cb_->weak_count.fetch_sub(1, std::memory_order_acq_rel) == 1)
            cb_->destroySelf();   // 最后一个 weak_ptr 销毁时回收控制块
        ptr_ = nullptr; cb_ = nullptr;
    }
};

// make_shared（合并对象和控制块为一次 new，减少内存碎片）
template<typename T, typename... Args>
MySharedPtr<T> makeShared(Args&&... args) {
    return MySharedPtr<T>(new T(std::forward<Args>(args)...));
    // 注：真正的 make_shared 将对象嵌入控制块一次性分配，此处为简化版
}
```

#### 4.3 常见面试追问

|问题|答案|
|---|---|
|`shared_ptr` 引用计数为何用 `atomic`？|多线程同时拷贝/析构时并发修改计数，需要原子操作保证不撕裂|
|为何 `shared_ptr` 本身不是线程安全的？|引用计数是原子的，但 `ptr_` 的读写不是；同时读写同一 `shared_ptr` 对象需外部加锁|
|`weak_ptr` 解决什么问题？|打破循环引用（A→B→A 导致内存泄漏）|
|`make_shared` vs `new` 传入 `shared_ptr`？|`make_shared` 一次分配合并对象和控制块，减少 1 次 `malloc`，且异常安全|
|为何 `unique_ptr` 不需要原子计数？|独占所有权，同一时刻只有一个所有者，无并发竞争|

### 1. 比较两个 `float` 是否相等

#### 1.1 核心思路

IEEE 754 浮点数不可用 `==` 直接比较。两种方案：

- **绝对误差**：$|a - b| < \varepsilon$，适合接近 0 的数。
- **相对误差**：$\dfrac{|a-b|}{\max(|a|,|b|)} < \varepsilon$，适合量级较大的数。
- **ULP（Units in the Last Place）**：比较二进制表示的距离，最通用。

```cpp
#include <cmath>
#include <cstdint>
#include <cstring>
#include <algorithm>
#include <limits>

// ── 方法 1：绝对误差（适合接近 0 的场景）────────────────────────────────
bool floatEqAbsolute(float a, float b, float eps = 1e-6f) {
    return std::fabs(a - b) < eps;
}

// ── 方法 2：相对 + 绝对组合（推荐通用方案）──────────────────────────────
// rel_eps：相对误差阈值；abs_eps：兜底绝对误差（防止除以接近 0 的数）
bool floatEqRelative(float a, float b,
                     float rel_eps = 1e-5f,
                     float abs_eps = 1e-8f) {
    float diff = std::fabs(a - b);
    if (diff < abs_eps) return true;   // 绝对意义上足够近
    float largest = std::max(std::fabs(a), std::fabs(b));
    return diff < rel_eps * largest;   // 相对误差检查
}

// ── 方法 3：ULP 比较（最精确，符合 IEEE 754 语义）────────────────────────
// 将 float 的二进制表示转为有符号整数，差值即为 ULP 距离
bool floatEqULP(float a, float b, int max_ulp = 4) {
    // NaN 不等于任何值（包括自身）
    if (std::isnan(a) || std::isnan(b)) return false;
    // 符号不同且均非零时，不相等（+0 和 -0 的 bit 表示不同但应视为相等）
    if (std::signbit(a) != std::signbit(b))
        return a == b;   // 仅 +0 == -0 返回 true

    // 通过 memcpy 将 float 的位模式无 UB 地转为 int32
    int32_t ia, ib;
    std::memcpy(&ia, &a, sizeof(float));
    std::memcpy(&ib, &b, sizeof(float));

    return std::abs(ia - ib) <= max_ulp;
}
```

### 2. 线性插值

#### 2.1 公式

在两点 $(x_0, y_0)$、$(x_1, y_1)$ 之间，对给定 $x$ 的插值：

$$ y = y_0 + \frac{x - x_0}{x_1 - x_0}(y_1 - y_0) = \text{lerp}(y_0,\ y_1,\ t), \quad t = \frac{x - x_0}{x_1 - x_0} $$

```cpp
// ── 基本 lerp（C++20 std::lerp 的等价实现）───────────────────────────────
// lerp(a, b, t)：当 t=0 返回 a，t=1 返回 b，t∈(0,1) 线性插值
// 精度写法：a + t*(b-a) 在 t=1 时有误差；下式在端点处精确
inline float lerp(float a, float b, float t) {
    return (1.0f - t) * a + t * b;   // 等价 a*(1-t) + b*t，保证 t=0/1 精确
}

// ── 根据 x 坐标插值 ──────────────────────────────────────────────────────
float linearInterp(float x0, float y0, float x1, float y1, float x) {
    if (x0 == x1) return y0;   // 避免除以零
    float t = (x - x0) / (x1 - x0);
    return lerp(y0, y1, t);
}
```

### 3. 双线性插值

#### 3.1 公式

在矩形网格四角 $(x_0,y_0),(x_1,y_0),(x_0,y_1),(x_1,y_1)$ 已知函数值 $Q_{00},Q_{10},Q_{01},Q_{11}$ 的条件下，对内部点 $(x,y)$ 插值：

$$ f(x,y) \approx \frac{1}{(x_1-x_0)(y_1-y_0)} \Big[(x_1-x)(y_1-y),Q_{00} + (x-x_0)(y_1-y),Q_{10}

- (x_1-x)(y-y_0),Q_{01} + (x-x_0)(y-y_0),Q_{11}\Big] $$

等价为先在 $x$ 方向做两次线性插值，再在 $y$ 方向做一次：

$$ R_0 = \text{lerp}(Q_{00}, Q_{10},\ t_x), \quad R_1 = \text{lerp}(Q_{01}, Q_{11},\ t_x), \quad f = \text{lerp}(R_0, R_1,\ t_y) $$

```cpp
// Q00=左下, Q10=右下, Q01=左上, Q11=右上
// tx = (x - x0)/(x1 - x0), ty = (y - y0)/(y1 - y0)
float bilinearInterp(float Q00, float Q10, float Q01, float Q11,
                     float tx,  float ty) {
    float R0 = lerp(Q00, Q10, tx);   // 下边沿插值（y = y0）
    float R1 = lerp(Q01, Q11, tx);   // 上边沿插值（y = y1）
    return lerp(R0, R1, ty);          // 在 y 方向合并
}

// 更常见的图像采样形式：输入 (u, v) ∈ [0, W-1] × [0, H-1]（非整数坐标）
float sampleBilinear(const float* img, int W, int H, float u, float v) {
    // 取整得到四个相邻像素坐标，clamp 防越界
    int x0 = std::max(0, std::min((int)u,     W - 1));
    int x1 = std::max(0, std::min((int)u + 1, W - 1));
    int y0 = std::max(0, std::min((int)v,     H - 1));
    int y1 = std::max(0, std::min((int)v + 1, H - 1));

    float tx = u - (int)u;   // 小数部分
    float ty = v - (int)v;

    float Q00 = img[y0 * W + x0];
    float Q10 = img[y0 * W + x1];
    float Q01 = img[y1 * W + x0];
    float Q11 = img[y1 * W + x1];

    return bilinearInterp(Q00, Q10, Q01, Q11, tx, ty);
}
```

## 4. 向量外积（Cross Product）

### 4.1 公式

三维向量 $\mathbf{a} = (a_x, a_y, a_z)$，$\mathbf{b} = (b_x, b_y, b_z)$：

$$ \mathbf{a} \times \mathbf{b} = \begin{vmatrix} \mathbf{i} & \mathbf{j} & \mathbf{k} \ a_x & a_y & a_z \ b_x & b_y & b_z \end{vmatrix} = (a_y b_z - a_z b_y,; a_z b_x - a_x b_z,; a_x b_y - a_y b_x) $$

结果向量垂直于 $\mathbf{a}$、$\mathbf{b}$ 所在平面，模长 $= |\mathbf{a}||\mathbf{b}|\sin\theta$。

```cpp
#include <array>
#include <cmath>

using Vec3 = std::array<float, 3>;

Vec3 cross(const Vec3& a, const Vec3& b) {
    return {
        a[1]*b[2] - a[2]*b[1],   // i 分量
        a[2]*b[0] - a[0]*b[2],   // j 分量
        a[0]*b[1] - a[1]*b[0]    // k 分量
    };
}

float dot(const Vec3& a, const Vec3& b) {
    return a[0]*b[0] + a[1]*b[1] + a[2]*b[2];
}

float norm(const Vec3& a) {
    return std::sqrt(dot(a, a));
}
```

## 5. 矩阵链乘：$(A_{1\times256} \cdot B_{256\times128}) \cdot C_{128\times256}$

### 5.1 最优括号顺序分析

设矩阵维度 $A_{1\times256}$，$B_{256\times128}$，$C_{128\times256}$。

**方案一**：$(A \cdot B) \cdot C$

$$ \text{cost}_1 = 1 \times 256 \times 128 + 1 \times 128 \times 256 = 32768 + 32768 = 65536 $$

**方案二**：$A \cdot (B \cdot C)$

$$ \text{cost}_2 = 256 \times 128 \times 256 + 1 \times 256 \times 256 = 8388608 + 65536 \gg \text{cost}_1 $$

**结论**：先算 $A \cdot B$，代价 $65536$，远优于先算 $B \cdot C$（$8M+$）。

```cpp
#include <vector>
#include <cassert>
#include <cstring>

using Matrix = std::vector<std::vector<float>>;

// 矩阵乘法：C = A(m×k) · B(k×n)，返回 m×n 矩阵
Matrix matMul(const Matrix& A, const Matrix& B) {
    int m = static_cast<int>(A.size());
    int k = static_cast<int>(B.size());
    int n = static_cast<int>(B[0].size());
    assert(static_cast<int>(A[0].size()) == k);

    Matrix C(m, std::vector<float>(n, 0.0f));
    for (int i = 0; i < m; ++i)
        for (int p = 0; p < k; ++p)       // 循环顺序 i-p-j 对缓存友好（A行连续）
            for (int j = 0; j < n; ++j)
                C[i][j] += A[i][p] * B[p][j];
    return C;
}

// 链乘：先 (A·B)，再结果·C
Matrix chainMul(const Matrix& A, const Matrix& B, const Matrix& C) {
    auto AB = matMul(A, B);    // A(1×256) · B(256×128) → AB(1×128)
    return matMul(AB, C);      // AB(1×128) · C(128×256) → result(1×256)
}
```

## 6. 交叉熵损失（Cross-Entropy Loss）

### 6.1 公式

对于 $C$ 类分类，单样本损失：

$$ \mathcal{L} = -\sum_{c=1}^{C} y_c \log(\hat{p}_c) $$

其中 $y_c \in {0,1}$ 为 one-hot 标签，$\hat{p}_c = \text{Softmax}(\mathbf{z})_c$。 数值稳定写法（log-sum-exp）：

$$ \log \hat{p}_c = z_c - \log!\sum_{j} e^{z_j} = z_c - \left(z_{\max} + \log!\sum_{j} e^{z_j - z_{\max}}\right) $$

```cpp
#include <vector>
#include <cmath>
#include <algorithm>
#include <numeric>

// logits：模型原始输出（未经 softmax）；label：真实类别下标
// 返回单样本交叉熵损失（数值稳定版）
float crossEntropyLoss(const std::vector<float>& logits, int label) {
    // 数值稳定：减去最大值后再做 exp，防止上溢
    float max_logit = *std::max_element(logits.begin(), logits.end());

    float log_sum_exp = 0.0f;
    for (float z : logits)
        log_sum_exp += std::exp(z - max_logit);
    log_sum_exp = std::log(log_sum_exp) + max_logit;   // log Σ exp(z_j)

    // loss = -z_label + log_sum_exp（等价于 -log softmax(label)）
    return log_sum_exp - logits[label];
}

// Batch 版本：对多个样本取平均
float batchCrossEntropyLoss(const std::vector<std::vector<float>>& batch_logits,
                             const std::vector<int>& labels) {
    float total = 0.0f;
    for (size_t i = 0; i < batch_logits.size(); ++i)
        total += crossEntropyLoss(batch_logits[i], labels[i]);
    return total / static_cast<float>(batch_logits.size());
}
```

## 7. 线性回归 SGD

### 7.1 模型与梯度

模型：$\hat{y} = \mathbf{w}^\top \mathbf{x} + b$

MSE 损失：$\mathcal{L} = \dfrac{1}{2N}\sum_{i=1}^{N}(\hat{y}_i - y_i)^2$

梯度（单样本）：

$$ \frac{\partial \mathcal{L}}{\partial \mathbf{w}} = (\hat{y} - y),\mathbf{x}, \qquad \frac{\partial \mathcal{L}}{\partial b} = (\hat{y} - y) $$

SGD 更新（学习率 $\eta$）：

$$ \mathbf{w} \leftarrow \mathbf{w} - \eta(\hat{y}-y)\mathbf{x}, \qquad b \leftarrow b - \eta(\hat{y}-y) $$

```cpp
#include <vector>
#include <random>
#include <numeric>
#include <cmath>

class LinearRegression {
public:
    explicit LinearRegression(int features, float lr = 0.01f)
        : w_(features, 0.0f), b_(0.0f), lr_(lr) {}

    // 前向推理：ŷ = w·x + b
    float predict(const std::vector<float>& x) const {
        float y = b_;
        for (size_t i = 0; i < w_.size(); ++i)
            y += w_[i] * x[i];
        return y;
    }

    // 单样本 SGD 更新
    void sgdStep(const std::vector<float>& x, float y_true) {
        float y_hat = predict(x);
        float err   = y_hat - y_true;   // 残差 (ŷ - y)

        // 梯度下降：w -= lr * err * x
        for (size_t i = 0; i < w_.size(); ++i)
            w_[i] -= lr_ * err * x[i];
        b_ -= lr_ * err;
    }

    // 训练 epochs 轮（每轮随机打乱顺序）
    void fit(const std::vector<std::vector<float>>& X,
             const std::vector<float>& Y, int epochs = 100) {
        std::mt19937 rng(42);
        std::vector<int> idx(X.size());
        std::iota(idx.begin(), idx.end(), 0);

        for (int e = 0; e < epochs; ++e) {
            std::shuffle(idx.begin(), idx.end(), rng);  // 随机打乱，减少梯度相关性
            for (int i : idx)
                sgdStep(X[i], Y[i]);
        }
    }

    const std::vector<float>& weights() const { return w_; }
    float bias() const { return b_; }

private:
    std::vector<float> w_;
    float              b_;
    float              lr_;
};
```

## 8. Softmax

### 8.1 公式与数值稳定性

$$ \text{Softmax}(z_i) = \frac{e^{z_i}}{\sum_{j=1}^{C} e^{z_j}} $$

数值稳定写法：令 $m = \max_j z_j$，则：

$$ \text{Softmax}(z_i) = \frac{e^{z_i - m}}{\sum_{j} e^{z_j - m}} $$

减去 $m$ 不改变结果（分子分母同乘 $e^{-m}$），但避免 $e^{z_i}$ 上溢。

```cpp
#include <vector>
#include <cmath>
#include <algorithm>
#include <numeric>

// in-place Softmax（数值稳定版）
void softmax(std::vector<float>& x) {
    // 1. 找最大值，做数值稳定偏移
    float max_val = *std::max_element(x.begin(), x.end());

    // 2. exp(x_i - max)
    float sum = 0.0f;
    for (float& v : x) {
        v    = std::exp(v - max_val);
        sum += v;
    }

    // 3. 归一化
    for (float& v : x) v /= sum;
}

// 返回新 vector 的版本
std::vector<float> softmaxCopy(const std::vector<float>& logits) {
    std::vector<float> out(logits);
    softmax(out);
    return out;
}
```

## 9. Online Softmax

### 9.1 核心思路

**问题**：标准 Softmax 需要两次遍历（第一次找 max，第二次算 exp 和归一化）。 **Online Softmax** 在一次遍历中同时维护最大值 $m$ 和修正后的归一化因子 $d$，实现单 Pass 计算。这是 Flash Attention 的数学基础。

**维护不变式**：处理到元素 $x_j$ 时，

$$ m_j = \max_{i \le j} x_i, \qquad d_j = \sum_{i \le j} e^{x_i - m_j} $$

当新元素 $x_{j+1}$ 到来时，若 $x_{j+1} > m_j$，需修正旧的 $d_j$：

$$ m_{j+1} = x_{j+1}, \qquad d_{j+1} = d_j \cdot e^{m_j - m_{j+1}} + 1 $$

否则（$x_{j+1} \le m_j$）：

$$ m_{j+1} = m_j, \qquad d_{j+1} = d_j + e^{x_{j+1} - m_j} $$

合并写法：

$$ m_{j+1} = \max(m_j, x_{j+1}), \qquad d_{j+1} = d_j \cdot e^{m_j - m_{j+1}} + e^{x_{j+1} - m_{j+1}} $$

```cpp
#include <vector>
#include <cmath>
#include <limits>

// Online Softmax：单次遍历，输出 softmax 概率向量
// 同时输出 log_sum_exp（供 log-softmax 等场景使用）
std::vector<float> onlineSoftmax(const std::vector<float>& x) {
    float m = -std::numeric_limits<float>::infinity();   // 当前最大值
    float d = 0.0f;   // 修正后的 exp 累加和（对应当前 m 的归一化因子）

    // Pass 1：单次遍历，同时维护 m 和 d
    for (float xi : x) {
        float new_m = std::max(m, xi);
        // 旧的 d 是基于 m 计算的，新 m 变化需要修正：乘以 e^(m - new_m)
        d = d * std::exp(m - new_m) + std::exp(xi - new_m);
        m = new_m;
    }

    // Pass 2：输出每个元素的 softmax 值
    std::vector<float> out(x.size());
    for (size_t i = 0; i < x.size(); ++i)
        out[i] = std::exp(x[i] - m) / d;

    return out;
}

// 完全单 Pass 版本（同时输出结果，需要额外存 exp 中间值）
std::vector<float> onlineSoftmaxSinglePass(const std::vector<float>& x) {
    int n = static_cast<int>(x.size());
    std::vector<float> exp_vals(n);

    float m = -std::numeric_limits<float>::infinity();
    float d = 0.0f;

    for (int i = 0; i < n; ++i) {
        float new_m = std::max(m, x[i]);
        // 修正之前存储的 exp 值（基准从 m 改为 new_m，需同比缩放）
        if (new_m != m) {
            float scale = std::exp(m - new_m);
            for (int j = 0; j < i; ++j) exp_vals[j] *= scale;
            d *= scale;
        }
        exp_vals[i] = std::exp(x[i] - new_m);
        d += exp_vals[i];
        m  = new_m;
    }

    for (float& v : exp_vals) v /= d;
    return exp_vals;
}
```

## 10. Layer Norm

### 10.1 公式

对长度为 $d$ 的向量 $\mathbf{x}$（一个样本的隐层激活）：

$$ \mu = \frac{1}{d}\sum_{i=1}^{d} x_i, \qquad \sigma^2 = \frac{1}{d}\sum_{i=1}^{d}(x_i - \mu)^2 $$

$$ \hat{x}_i = \frac{x_i - \mu}{\sqrt{\sigma^2 + \varepsilon}}, \qquad y_i = \gamma_i \hat{x}_i + \beta_i $$

其中 $\gamma, \beta$ 为可学习的缩放和偏移（Scale & Shift），$\varepsilon$ 防除零。

```cpp
#include <vector>
#include <cmath>
#include <stdexcept>

// in-place Layer Norm
// gamma, beta：可学习参数（shape 与 x 相同）；eps：数值稳定项
void layerNorm(std::vector<float>& x,
               const std::vector<float>& gamma,
               const std::vector<float>& beta,
               float eps = 1e-5f) {
    int d = static_cast<int>(x.size());
    if (d == 0) return;

    // 1. 计算均值 μ
    float mean = 0.0f;
    for (float v : x) mean += v;
    mean /= static_cast<float>(d);

    // 2. 计算方差 σ²（两次遍历，数值更稳定；Welford 算法也可）
    float var = 0.0f;
    for (float v : x) {
        float diff = v - mean;
        var += diff * diff;
    }
    var /= static_cast<float>(d);

    // 3. 归一化 + 仿射变换
    float inv_std = 1.0f / std::sqrt(var + eps);
    for (int i = 0; i < d; ++i)
        x[i] = gamma[i] * ((x[i] - mean) * inv_std) + beta[i];
}

// RMSNorm（无均值中心化，更高效，LLaMA 使用）
// y_i = (x_i / RMS(x)) * γ_i, RMS(x) = sqrt(1/d * Σ x_i²)
void rmsNorm(std::vector<float>& x,
             const std::vector<float>& gamma,
             float eps = 1e-6f) {
    int d = static_cast<int>(x.size());
    float rms = 0.0f;
    for (float v : x) rms += v * v;
    rms = std::sqrt(rms / static_cast<float>(d) + eps);
    float inv_rms = 1.0f / rms;
    for (int i = 0; i < d; ++i)
        x[i] = x[i] * inv_rms * gamma[i];
}
```

## 11. Attention 计算

### 11.1 公式

给定查询矩阵 $Q \in \mathbb{R}^{T_q \times d_k}$，键 $K \in \mathbb{R}^{T_k \times d_k}$，值 $V \in \mathbb{R}^{T_k \times d_v}$：

$$ \text{Attention}(Q,K,V) = \text{Softmax}!\left(\frac{QK^\top}{\sqrt{d_k}}\right) V $$

```cpp
#include <vector>
#include <cmath>
#include <algorithm>

using Mat = std::vector<std::vector<float>>;

// 矩阵乘法 C = A(m×k) · B(k×n)
Mat matMulBasic(const Mat& A, const Mat& B) {
    int m = A.size(), k = B.size(), n = B[0].size();
    Mat C(m, std::vector<float>(n, 0.0f));
    for (int i = 0; i < m; ++i)
        for (int p = 0; p < k; ++p)
            for (int j = 0; j < n; ++j)
                C[i][j] += A[i][p] * B[p][j];
    return C;
}

// 矩阵转置
Mat transpose(const Mat& A) {
    int m = A.size(), n = A[0].size();
    Mat B(n, std::vector<float>(m));
    for (int i = 0; i < m; ++i)
        for (int j = 0; j < n; ++j)
            B[j][i] = A[i][j];
    return B;
}

// 行 Softmax（in-place）
void rowSoftmax(Mat& A) {
    for (auto& row : A) {
        float mx = *std::max_element(row.begin(), row.end());
        float sum = 0.0f;
        for (float& v : row) { v = std::exp(v - mx); sum += v; }
        for (float& v : row)   v /= sum;
    }
}

// Scaled Dot-Product Attention
// Q: [Tq, dk], K: [Tk, dk], V: [Tk, dv]
// 返回: [Tq, dv]
Mat scaledDotProductAttention(const Mat& Q, const Mat& K, const Mat& V,
                               bool causal_mask = false) {
    int Tq = Q.size(), dk = Q[0].size(), Tk = K.size();
    float scale = 1.0f / std::sqrt(static_cast<float>(dk));

    // 1. 计算 scores = Q · K^T / sqrt(dk)，shape [Tq, Tk]
    Mat KT     = transpose(K);
    Mat scores = matMulBasic(Q, KT);
    for (auto& row : scores)
        for (float& v : row) v *= scale;

    // 2. 因果掩码（自回归场景）：position j > i 的位置置为 -inf
    if (causal_mask) {
        for (int i = 0; i < Tq; ++i)
            for (int j = i + 1; j < Tk; ++j)
                scores[i][j] = -std::numeric_limits<float>::infinity();
    }

    // 3. Softmax（逐行）
    rowSoftmax(scores);

    // 4. 加权求和：output = scores · V，shape [Tq, dv]
    return matMulBasic(scores, V);
}
```

## 12. Linear Attention

### 12.1 核心思路

标准 Attention 复杂度 $O(T^2 d)$，Linear Attention 通过核函数近似将复杂度降至 $O(Td^2)$。

**核技巧**：将 $\text{softmax}(QK^\top)$ 近似为 $\phi(Q)\phi(K)^\top$，其中 $\phi(\cdot)$ 为特征映射（常用 ELU+1 或随机特征）。

利用矩阵结合律：

$$ \text{Linear-Attention}(Q,K,V) = \frac{\phi(Q),[\phi(K)^\top V]}{\phi(Q),[\phi(K)^\top \mathbf{1}]} $$

先算 $\phi(K)^\top V$（$d \times d_v$ 矩阵），消去序列维度 $T$。

```cpp
#include <vector>
#include <cmath>
#include <algorithm>

// 特征映射 φ(x)_i = max(x_i, 0) + 1（ELU+1，保证非负）
std::vector<float> featureMap(const std::vector<float>& x) {
    std::vector<float> out(x.size());
    for (size_t i = 0; i < x.size(); ++i)
        out[i] = (x[i] >= 0.0f) ? (x[i] + 1.0f) : (std::exp(x[i]));
    return out;
}

// Linear Attention（简化版，单头）
// Q,K,V 均为 [T, d] 二维向量
Mat linearAttention(const Mat& Q, const Mat& K, const Mat& V) {
    int T = Q.size(), d = Q[0].size(), dv = V[0].size();

    // 1. 对 Q, K 逐行应用特征映射
    Mat phiQ(T), phiK(T);
    for (int i = 0; i < T; ++i) {
        phiQ[i] = featureMap(Q[i]);
        phiK[i] = featureMap(K[i]);
    }

    // 2. 计算 S = φ(K)^T · V，shape [d, dv]（与 T 无关）
    Mat S(d, std::vector<float>(dv, 0.0f));
    for (int t = 0; t < T; ++t)
        for (int i = 0; i < d;  ++i)
            for (int j = 0; j < dv; ++j)
                S[i][j] += phiK[t][i] * V[t][j];

    // 3. 计算归一化因子 z = φ(K)^T · 1，shape [d]
    std::vector<float> z(d, 0.0f);
    for (int t = 0; t < T; ++t)
        for (int i = 0; i < d; ++i)
            z[i] += phiK[t][i];

    // 4. 输出：output[t] = φ(Q[t]) · S / (φ(Q[t]) · z)
    Mat out(T, std::vector<float>(dv, 0.0f));
    for (int t = 0; t < T; ++t) {
        float norm = 0.0f;
        for (int i = 0; i < d; ++i) norm += phiQ[t][i] * z[i];
        if (norm < 1e-6f) norm = 1e-6f;   // 防除零
        for (int j = 0; j < dv; ++j) {
            float val = 0.0f;
            for (int i = 0; i < d; ++i) val += phiQ[t][i] * S[i][j];
            out[t][j] = val / norm;
        }
    }
    return out;
}
```

## 13. MoE（Mixture of Experts）前向推理

### 13.1 架构概述

$$ \text{MoE}(\mathbf{x}) = \sum_{i=1}^{K} g_i(\mathbf{x}), E_i(\mathbf{x}) $$

- **Router（门控网络）**：$g(\mathbf{x}) = \text{TopK-Softmax}(W_g \mathbf{x})$，仅激活 $K$ 个 Expert（通常 $K=2$）。
- **Expert**：每个 Expert 为独立的 FFN 子网络。
- **稀疏激活**：每个 token 只经过 $K/N$ 的参数量，推理计算量大幅降低。

```cpp
#include <vector>
#include <algorithm>
#include <numeric>
#include <cmath>

// 模拟单个 Expert（线性层 + ReLU）
std::vector<float> expertForward(const std::vector<float>& x,
                                  const Mat& W1, const Mat& W2) {
    // W1: [hidden_dim, d_model], W2: [d_model, hidden_dim]
    int hidden = W1.size(), d = x.size();
    std::vector<float> h(hidden, 0.0f);

    // 上投影 + ReLU
    for (int i = 0; i < hidden; ++i) {
        for (int j = 0; j < d; ++j) h[i] += W1[i][j] * x[j];
        h[i] = std::max(0.0f, h[i]);   // ReLU
    }

    // 下投影
    std::vector<float> out(d, 0.0f);
    for (int j = 0; j < d; ++j)
        for (int i = 0; i < hidden; ++i)
            out[j] += W2[j][i] * h[i];
    return out;
}

// MoE 前向：TopK 路由 + Expert 加权求和
// x: [d_model]; router_w: [num_experts, d_model]; experts: num_experts 个 FFN
std::vector<float> moeForward(
    const std::vector<float>& x,
    const Mat& router_w,                   // 路由权重矩阵
    const std::vector<std::pair<Mat,Mat>>& experts,  // 每个 Expert 的 (W1, W2)
    int top_k = 2)
{
    int num_experts = static_cast<int>(experts.size());
    int d = static_cast<int>(x.size());

    // 1. 计算路由 logits = router_w · x
    std::vector<float> logits(num_experts, 0.0f);
    for (int e = 0; e < num_experts; ++e)
        for (int j = 0; j < d; ++j)
            logits[e] += router_w[e][j] * x[j];

    // 2. TopK 选择：找最大的 top_k 个 Expert
    std::vector<int> indices(num_experts);
    std::iota(indices.begin(), indices.end(), 0);
    std::partial_sort(indices.begin(), indices.begin() + top_k, indices.end(),
                      [&](int a, int b){ return logits[a] > logits[b]; });
    indices.resize(top_k);

    // 3. Softmax 归一化所选 Expert 的权重（只对 top_k 个做 softmax）
    float max_logit = -std::numeric_limits<float>::infinity();
    for (int i : indices) max_logit = std::max(max_logit, logits[i]);
    float sum = 0.0f;
    for (int i : indices) { logits[i] = std::exp(logits[i] - max_logit); sum += logits[i]; }
    for (int i : indices)   logits[i] /= sum;

    // 4. 加权聚合各 Expert 输出
    std::vector<float> out(d, 0.0f);
    for (int i : indices) {
        auto expert_out = expertForward(x, experts[i].first, experts[i].second);
        for (int j = 0; j < d; ++j)
            out[j] += logits[i] * expert_out[j];
    }
    return out;
}
```

## 14. Flow Matching Model 采样

### 14.1 核心思路

Flow Matching 学习一个速度场 $v_\theta(\mathbf{x}, t)$，使得从噪声 $\mathbf{x}_0 \sim \mathcal{N}(0,I)$ 积分 ODE 到 $t=1$ 得到数据样本：

$$ \frac{d\mathbf{x}}{dt} = v_\theta(\mathbf{x}_t,\ t) $$

采样过程即求解此 ODE：常用 **Euler 法**（一阶）或 **Heun 法**（二阶，精度更高）。

**Euler 法**：$\mathbf{x}_{t+h} = \mathbf{x}_t + h \cdot v_\theta(\mathbf{x}_t, t)$

**Heun 法（Predictor-Corrector）**：

$$ \tilde{\mathbf{x}}_{t+h} = \mathbf{x}_t + h \cdot v_\theta(\mathbf{x}_t, t) $$ $$ \mathbf{x}_{t+h} = \mathbf{x}_t + \frac{h}{2}\left[v_\theta(\mathbf{x}_t, t) + v_\theta(\tilde{\mathbf{x}}_{t+h}, t+h)\right] $$

```cpp
#include <vector>
#include <functional>
#include <random>

using VelocityFn = std::function<std::vector<float>(
    const std::vector<float>& x, float t)>;

// ── Euler 采样器 ──────────────────────────────────────────────────────────
// velocity_fn: 模型预测的速度场 v_θ(x, t)
// x0: 初始噪声；steps: 积分步数（越多越准确）
std::vector<float> flowMatchingSampleEuler(
    VelocityFn velocity_fn,
    std::vector<float> x,      // 初始噪声 x_0
    int steps = 50)
{
    float dt = 1.0f / static_cast<float>(steps);  // 等步长，t 从 0 积分到 1

    for (int i = 0; i < steps; ++i) {
        float t = i * dt;
        auto v = velocity_fn(x, t);   // 查询当前速度
        for (size_t j = 0; j < x.size(); ++j)
            x[j] += dt * v[j];        // Euler 步进
    }
    return x;
}

// ── Heun 采样器（二阶，精度更高，每步需调用 2 次模型）────────────────────
std::vector<float> flowMatchingSampleHeun(
    VelocityFn velocity_fn,
    std::vector<float> x,
    int steps = 20)
{
    float dt = 1.0f / static_cast<float>(steps);

    for (int i = 0; i < steps; ++i) {
        float t  = i * dt;
        float t1 = t + dt;

        // Predictor：Euler 预测下一步
        auto v0 = velocity_fn(x, t);
        std::vector<float> x_pred(x.size());
        for (size_t j = 0; j < x.size(); ++j)
            x_pred[j] = x[j] + dt * v0[j];

        // Corrector：用预测值再算一次速度，取均值
        auto v1 = velocity_fn(x_pred, t1);
        for (size_t j = 0; j < x.size(); ++j)
            x[j] += 0.5f * dt * (v0[j] + v1[j]);   // 梯形公式
    }
    return x;
}
```

## 15. 支持 Broadcast 的 4D Tensor Elementwise Mul

### 15.1 Broadcast 规则

对齐后，各维度满足：相等，或其中一个为 1（自动扩展）。

```cpp
#include <vector>
#include <array>
#include <stdexcept>
#include <numeric>

using Shape4 = std::array<int, 4>;

// 计算广播后的输出 shape；不可广播时抛出异常
Shape4 broadcastShape(const Shape4& a, const Shape4& b) {
    Shape4 out;
    for (int i = 0; i < 4; ++i) {
        if (a[i] == b[i])         out[i] = a[i];
        else if (a[i] == 1)       out[i] = b[i];
        else if (b[i] == 1)       out[i] = a[i];
        else throw std::invalid_argument("Shapes not broadcastable");
    }
    return out;
}

// 将线性下标 idx（基于 out_shape）转换为在 src_shape 中的线性下标
// Broadcast 维度（src_shape[d]=1）在 src 中始终取 0
int broadcastIndex(int idx, const Shape4& out_shape, const Shape4& src_shape) {
    int src_idx = 0;
    int src_stride = 1;
    for (int d = 3; d >= 0; --d) {
        int pos     = idx % out_shape[d];    // 当前维度的坐标
        idx        /= out_shape[d];
        int src_pos = (src_shape[d] == 1) ? 0 : pos;  // Broadcast 维度强制取 0
        src_idx    += src_pos * src_stride;
        src_stride *= src_shape[d];
    }
    return src_idx;
}

// 4D Broadcast Elementwise 乘法
// 返回 out_shape 的 flat vector
std::vector<float> broadcastMul(const std::vector<float>& A, const Shape4& shapeA,
                                  const std::vector<float>& B, const Shape4& shapeB) {
    Shape4 out_shape = broadcastShape(shapeA, shapeB);
    int total = 1;
    for (int d : out_shape) total *= d;

    std::vector<float> out(total);
    for (int i = 0; i < total; ++i) {
        int ia = broadcastIndex(i, out_shape, shapeA);
        int ib = broadcastIndex(i, out_shape, shapeB);
        out[i] = A[ia] * B[ib];
    }
    return out;
}
```

## 16. Embedding Sparse Feature Pooling

### 16.1 问题定义

给定索引数组 $A$（长度 $M$）和嵌入向量 $B$（形状 $[M, d]$），计算：

$$ C[i] = \sum_{j:, A[j]=i} B[j], \quad i = 0, 1, \ldots, N-1 $$

即对每个 Embedding ID $i$，将所有属于它的行求和（稀疏 Reduce）。

```cpp
#include <vector>
#include <stdexcept>

// A: 索引数组，长度 M；B: 嵌入矩阵，[M, dim]；N: 目标嵌入数
// 返回 C: [N, dim]，C[i] = sum of B[j] for all j where A[j] == i
std::vector<std::vector<float>> embeddingPooling(
    const std::vector<int>& A,
    const std::vector<std::vector<float>>& B,
    int N)
{
    int M   = static_cast<int>(A.size());
    int dim = B.empty() ? 0 : static_cast<int>(B[0].size());

    // 输出矩阵，初始化为 0
    std::vector<std::vector<float>> C(N, std::vector<float>(dim, 0.0f));

    for (int j = 0; j < M; ++j) {
        int i = A[j];
        if (i < 0 || i >= N)
            throw std::out_of_range("Index out of range");
        // 累加第 j 行的嵌入向量到 C[i]
        for (int d = 0; d < dim; ++d)
            C[i][d] += B[j][d];
    }
    return C;
}

// 平坦化版本（嵌入矩阵以 flat vector 存储，行优先）
std::vector<float> embeddingPoolingFlat(
    const std::vector<int>& A,
    const std::vector<float>& B_flat,   // [M * dim]
    int M, int N, int dim)
{
    std::vector<float> C(N * dim, 0.0f);
    for (int j = 0; j < M; ++j) {
        int i = A[j];
        for (int d = 0; d < dim; ++d)
            C[i * dim + d] += B_flat[j * dim + d];
    }
    return C;
}
```

## 17. 井字棋判胜

### 17.1 判断逻辑

棋盘 $3\times3$，检查 8 条线（3 行 + 3 列 + 2 对角线）是否全为同一方。

```cpp
#include <array>

// board[i][j]：0=空, 1=玩家1(X), 2=玩家2(O)
// 返回 0=无胜者, 1=玩家1胜, 2=玩家2胜, -1=平局
int checkWinner(const std::array<std::array<int,3>,3>& board) {
    // 定义 8 条线，每条线为 3 个格坐标
    constexpr int lines[8][3][2] = {
        {{0,0},{0,1},{0,2}},  // 第 0 行
        {{1,0},{1,1},{1,2}},  // 第 1 行
        {{2,0},{2,1},{2,2}},  // 第 2 行
        {{0,0},{1,0},{2,0}},  // 第 0 列
        {{0,1},{1,1},{2,1}},  // 第 1 列
        {{0,2},{1,2},{2,2}},  // 第 2 列
        {{0,0},{1,1},{2,2}},  // 主对角线
        {{0,2},{1,1},{2,0}},  // 副对角线
    };

    for (auto& line : lines) {
        int a = board[line[0][0]][line[0][1]];
        int b = board[line[1][0]][line[1][1]];
        int c = board[line[2][0]][line[2][1]];
        if (a != 0 && a == b && b == c)
            return a;   // 找到胜者
    }

    // 检查是否平局（棋盘已满，无空格）
    for (auto& row : board)
        for (int cell : row)
            if (cell == 0) return 0;   // 仍有空格，游戏继续

    return -1;   // 平局
}
```

## 18. 单例模式

### 18.1 核心要求

- 全局唯一实例。
- 线程安全。
- 禁止拷贝/移动。

### 18.2 Meyer's Singleton（C++11，推荐）

```cpp
class Singleton {
public:
    // 获取唯一实例：C++11 保证局部静态变量初始化是线程安全的
    static Singleton& getInstance() {
        static Singleton instance;   // 首次调用时构造，程序结束时析构
        return instance;
    }

    // 禁止拷贝和移动
    Singleton(const Singleton&)            = delete;
    Singleton& operator=(const Singleton&) = delete;
    Singleton(Singleton&&)                 = delete;
    Singleton& operator=(Singleton&&)      = delete;

    void doSomething() { /* ... */ }

private:
    Singleton()  = default;   // 构造函数私有化
    ~Singleton() = default;
};

// 使用：
// Singleton::getInstance().doSomething();
```

### 18.3 双重检查锁（DCLP，C++11 atomic 版，需跨编译器兼容时使用）

```cpp
#include <atomic>
#include <mutex>

class SingletonDCLP {
public:
    static SingletonDCLP* getInstance() {
        // 第一次检查（无锁，避免每次都加锁）
        SingletonDCLP* p = instance_.load(std::memory_order_acquire);
        if (!p) {
            std::lock_guard<std::mutex> lk(mu_);
            // 第二次检查（加锁后再次确认，防止多线程同时通过第一次检查）
            p = instance_.load(std::memory_order_relaxed);
            if (!p) {
                p = new SingletonDCLP();
                instance_.store(p, std::memory_order_release);  // release 保证初始化可见
            }
        }
        return p;
    }

    SingletonDCLP(const SingletonDCLP&)            = delete;
    SingletonDCLP& operator=(const SingletonDCLP&) = delete;

private:
    SingletonDCLP() = default;
    static std::atomic<SingletonDCLP*> instance_;
    static std::mutex                  mu_;
};

std::atomic<SingletonDCLP*> SingletonDCLP::instance_{nullptr};
std::mutex                  SingletonDCLP::mu_;
```

> **面试注意**：C++11 及以后优先推荐 Meyer's Singleton，局部静态变量的线程安全性由标准保证（§6.7），无需手动加锁。DCLP 方案仅在需要指针语义（延迟销毁、动态多态）时考虑。
